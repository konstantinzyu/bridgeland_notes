	\section{Stability conditions on an abelian category}
	
	\subsection{Conventions}
	
	This section follows \cite[\S 4]{MS17}; the notion of a stability function on an abelian category goes back to Rudakov \cite{Rud97}.
	Throughout, $\calA$ is an essentially small abelian category and $\Kgroup(\calA)$ is its Grothendieck group, i.e.\ the abelian group generated by the classes $[E]$ of objects $E \in \calA$ subject to the relations $[E] = [E'] + [E'']$ for every short exact sequence $0 \to E' \to E \to E'' \to 0$. For an object $E$ and a group homomorphism $Z \colon \Kgroup(\calA) \to \CC$ we write $Z(E) := Z([E])$.
	
	Let
	\[
	\HH := \defineset{ r e^{i \pi \varphi} }{ r > 0,\ \varphi \in (0, 1] }
	= \defineset{ z \in \CC }{ \im z > 0 } \cup \RR_{<0}
	\]
	be the upper half-plane together with the negative real ray. For $z \in \HH$ the number $\varphi \in (0, 1]$ with $z \in \RR_{>0} e^{i \pi \varphi}$ is uniquely determined; we call it the \emph{phase} of $z$. Note that $\HH$ is closed under addition and under multiplication by positive real numbers.
	
	The notation $F \subset E$ and $E \twoheadrightarrow G$ always refers to a subobject and a quotient object in the abelian category $\calA$. A subobject $F \subset E$ is \emph{proper} if $F \neq E$ and \emph{nontrivial} if $F \neq 0$.
	
	\subsection{Stability functions}
	
	A group homomorphism $Z \colon \Kgroup(\calA) \to \CC$ is called a \emph{stability function} on $\calA$ (\cite[Def.~4.1]{MS17}, \cite[Def.~2.1]{Bri07}) if
	\begin{itemize}
		\item[] \condlabel{Z}{cond:Z} $Z(E) \in \HH$ for every nonzero object $E \in \calA$.
	\end{itemize}
	Equivalently: $\im Z(E) \geqslant 0$, and $\im Z(E) = 0$ implies $\re Z(E) < 0$.
	
	Let $Z$ be a stability function. For a nonzero object $E \in \calA$ the \emph{phase} of $E$ is the number $\varphi(E) = \varphi_Z(E) \in (0, 1]$ equal to the phase of the complex number $Z(E) \in \HH$, so that
	\[
	Z(E) = m(E) \, e^{i \pi \varphi(E)}, \qquad m(E) := |Z(E)| > 0 .
	\]
	The number $m(E)$ is called the \emph{mass} of $E$.
	
	\begin{remark}
		In \cite[Def.~4.1]{MS17} one works with the generalized rank $R := \im Z$ and the generalized degree $D := -\re Z$; the phase corresponds to the generalized slope $M := D / R = -\cot(\pi \varphi) \in \RR \cup \{ +\infty \}$, with $M = +\infty$ for $\varphi = 1$. Since $\varphi \mapsto -\cot(\pi \varphi)$ is strictly increasing on $(0, 1)$, comparing slopes and comparing phases is the same thing: $M(F) \leqslant M(E)$ if and only if $\varphi(F) \leqslant \varphi(E)$, and likewise for strict inequalities.
		We shall mostly work with phases;
		the rank $R$, the degree $D$ and the slope $M$ reappear in the proof of existence of HN filtrations, where the geometry of the plane is more conveniently expressed through them.
	\end{remark}
	
	Let $Z$ be a stability function on $\calA$. A nonzero object $E \in \calA$ is called \emph{$(\calA, Z)$-semistable} (\cite[Def.~4.2]{MS17}, \cite[Def.~2.2]{Bri07}) if
	\begin{itemize}
		\item[] \condlabel{AZ-SS}{cond:AZ-SS} $\varphi(F) \leqslant \varphi(E)$ for every nontrivial subobject $F \subset E$,
	\end{itemize}
	and \emph{$(\calA, Z)$-stable} if
	\begin{itemize}
		\item[] \condlabel{AZ-S}{cond:AZ-S} $\varphi(F) < \varphi(E)$ for every nontrivial proper subobject $F \subset E$.
	\end{itemize}
	When $\calA$ and $Z$ are clear from the context we simply speak of semistable and stable objects.
	
	\subsection{Harder--Narasimhan filtrations and stability conditions}
	
	Let $Z$ be a stability function on $\calA$ and let $E \in \calA$ be a nonzero object. A \emph{Harder--Narasimhan filtration} (\emph{HN filtration} for short) of $E$ with respect to $(\calA, Z)$ is a chain of subobjects
	\[
	0 = E_0 \subset E_1 \subset \ldots \subset E_{n-1} \subset E_n = E
	\]
	such that 
	\begin{itemize}
		\item the factors $A_i := E_i / E_{i-1}$, $i = 1, \ldots, n$, are nonzero $(\calA, Z)$-semistable objects;
		\item phases $\varphi(A_1) > \varphi(A_2) > \ldots > \varphi(A_n)$ are strictly decreasing.
	\end{itemize}
	The objects $A_i$ are called the \emph{HN factors} of $E$, and $\varphi^+(E) := \varphi(A_1)$, $\varphi^-(E) := \varphi(A_n)$ are called the \emph{maximal} and the \emph{minimal phase} of $E$.
	
	\begin{center}
		\begin{tikzcd}[column sep=large, row sep=large]
			0 = E_0 \arrow[r, hook] 
			& E_1 \arrow[r, hook] \arrow[d, two heads] 
			& E_2 \arrow[r, hook] \arrow[d, two heads] 
			& \cdots \arrow[r, hook] 
			& E_{n-1} \arrow[r, hook] \arrow[d, two heads] 
			& E_n = E \arrow[d, two heads] \\
			& A_1 & A_2 & & A_{n-1} & A_n
		\end{tikzcd}
	\end{center}
	Each column $E_{i-1} \hookrightarrow E_i \twoheadrightarrow A_i$ of the diagram is a short exact sequence;
	the horizontal arrows are the inclusions of the filtration.

	An $(\calA, Z)$-semistable object $E$ has the HN filtration $0 \subset E$ of length $n = 1$; conversely, if an HN filtration of $E$ has length $1$, then $E$ is $(\calA, Z)$-semistable. Uniqueness of HN filtrations will be proved below; until then the words ``maximal phase'' refer to a chosen filtration.
	
	A \emph{stability condition} on the abelian category $\calA$ (\cite[Def.~4.6]{MS17}, \cite[Def.~2.3]{Bri07}) is a pair $(\calA, Z)$, where $Z \colon \Kgroup(\calA) \to \CC$ is a group homomorphism, such that
	\begin{itemize}
		\item[] \condlabel{AZ-SC1}{cond:AZ-SC1} $Z$ is a stability function on $\calA$, i.e.\ condition \condref{cond:Z} holds;
		\item[] \condlabel{AZ-SC2}{cond:AZ-SC2} (\emph{Harder--Narasimhan property}) every nonzero object $E \in \calA$ admits an HN filtration with respect to $(\calA, Z)$.
	\end{itemize}
	The homomorphism $Z$ is called the \emph{central charge} of the stability condition.
	
	\begin{remark}
		Condition \condref{cond:AZ-SC2} does not follow from \condref{cond:AZ-SC1}: in Example~\ref{ex:no_HN} we shall give a stability function for which some object has no nonzero semistable subobject at all, hence no HN filtration.
		Sufficient conditions for the existence of HN filtrations (noetherianity of $\calA$ together with discreteness of the image of $\im Z$, or finite length of $\calA$) will be discussed separately.
	\end{remark}
	
	
	\subsection{Basic properties}
	
	Throughout this subsection $Z$ is a stability function on $\calA$.
	
	\begin{lemma}[see-saw, {\cite[Lemmas~1.2, 1.3]{Rud97}}]
		\label{lem:seesaw}
		Let $0 = E_0 \subset E_1 \subset \ldots \subset E_m = E$ be a chain of subobjects with nonzero factors $B_j := E_j / E_{j-1}$, $j = 1, \ldots, m$. Then
		\[
		\min_j \varphi(B_j) \leqslant \varphi(E) \leqslant \max_j \varphi(B_j),
		\]
		and equality holds on the left or on the right if and only if all the phases $\varphi(B_j)$ are equal.
		
		In particular, for a short exact sequence $0 \to F \to E \to G \to 0$ with $F, G \neq 0$ the following are equivalent: 
		\[ 
		\varphi(F) < \varphi(E); \quad \varphi(E) < \varphi(G); \quad \varphi(F) < \varphi(G).
		\]
		The same holds with $<$ replaced by $=$ or by $>$.
	\end{lemma}
		
	\begin{center}
		\begin{tikzpicture}[scale=1.2, >=stealth]
			% real axis
			\draw[gray] (-3.6,0) -- (2.6,0);
			\fill (0,0) circle (1.4pt) node[below] {$0$};
			% vectors
			\coordinate (F) at (1.8,1.4);
			\coordinate (G) at (-2.4,1.6);
			\coordinate (E) at (-0.6,3.0);
			\draw[thick, ->] (0,0) -- (F) node[right] {$Z(F)$};
			\draw[thick, ->] (0,0) -- (G) node[left] {$Z(G)$};
			\draw[very thick, ->] (0,0) -- (E) node[above] {$Z(E) = Z(F) + Z(G)$};
			% parallelogram
			\draw[dashed] (F) -- (E) -- (G);
		\end{tikzpicture}
	\end{center}
	The picture shows the case of a short exact sequence $0 \to F \to E \to G \to 0$:
	since $Z(F)$ and $Z(G)$ both lie in $\HH$, their sum $Z(E)$ lies in the sector between them,
	so the phase of $E$ is caught between the phases of $F$ and $G$.

	\begin{proof}
		By additivity, $Z(E) = \sum_j Z(B_j) = \sum_j r_j e^{i \pi \psi_j}$, where $r_j = m(B_j) > 0$ and $\psi_j = \varphi(B_j) \in (0, 1]$. Put $\psi := \max_j \psi_j$. Then
		\[
		e^{-i \pi \psi} Z(E) = \sum_j r_j e^{i \pi (\psi_j - \psi)}, \qquad \psi_j - \psi \in (-1, 0],
		\]
		so every summand has non-positive imaginary part, with equality if and only if $\psi_j = \psi$, in which case the summand is the positive real number $r_j$. 
		Hence $\im \big( e^{-i \pi \psi} Z(E) \big) \leqslant 0$, with equality if and only if all $\psi_j$ equal $\psi$. On the other hand $e^{-i \pi \psi} Z(E) = m(E) e^{i \pi \theta}$ with $\theta := \varphi(E) - \psi \in (-1, 1)$, and $\sin(\pi \theta) \leqslant 0$ on this interval exactly when $\theta \in (-1, 0]$, with $\sin(\pi \theta) = 0$ exactly when $\theta = 0$. This proves the upper bound and its equality case; the lower bound is obtained in the same way with $\min_j \psi_j$ in place of $\psi$.
		
		For a short exact sequence apply the above to the chain $0 \subset F \subset E$ with factors $F, G$: if, say, $\varphi(F) \leqslant \varphi(G)$, then $\varphi(F) \leqslant \varphi(E) \leqslant \varphi(G)$ and $\varphi(E)$ equals one of the endpoints only if $\varphi(F) = \varphi(G)$; the case $\varphi(F) \geqslant \varphi(G)$ is symmetric.
	\end{proof}
	
	\begin{lemma}[quotient characterization, {\cite[Exercise~4.4]{MS17}}]
		\label{lem:quotients}
		A nonzero object $E$ is $(\calA, Z)$-semistable 
		if and only if $\varphi(E) \leqslant \varphi(G)$ for every nonzero quotient $E \twoheadrightarrow G$, 
		and $(\calA, Z)$-stable if and only if $\varphi(E) < \varphi(G)$ for every nonzero proper quotient.
	\end{lemma}
	
	\begin{proof}
		Nontrivial proper subobjects $F \subset E$ correspond to nonzero proper quotients $G = E / F$, 
		and by Lemma~\ref{lem:seesaw} 
		the inequality $\varphi(F) \leqslant \varphi(E)$ is equivalent to $\varphi(E) \leqslant \varphi(G)$, 
		and similarly for strict inequalities. 
		The subobject $F = E$ and the quotient $G = E$ impose no condition.
	\end{proof}
	
	\begin{lemma}[Schur, {\cite[Thm~1]{Rud97}}, {\cite[Lemma~4.5]{MS17}}]
		\label{lem:schur}
		Let $E, F$ be $(\calA, Z)$-semistable objects.
		\begin{enumerate}
			\item If $\varphi(E) > \varphi(F)$, then $\Hom(E, F) = 0$.
			\item If $\varphi(E) = \varphi(F)$ and $f \colon E \to F$ is nonzero, then $f$ is a monomorphism provided $E$ is stable, an epimorphism provided $F$ is stable, and an isomorphism provided both are stable.
		\end{enumerate}
	\end{lemma}
	
	\begin{proof}
		Let $f \colon E \to F$ be nonzero, with image $I \neq 0$ and kernel $K$. 
		Since $I$ is a quotient of $E$, Lemma~\ref{lem:quotients} gives $\varphi(E) \leqslant \varphi(I)$; 
		since $I \subset F$, semistability of $F$ gives $\varphi(I) \leqslant \varphi(F)$. 
		Thus $\varphi(E) \leqslant \varphi(F)$, which proves (1). 
		In the situation of (2) all these inequalities are equalities. 
		If $E$ is stable and $K \neq 0$, 
		then $K$ is a nontrivial proper subobject of $E$ (proper because $I \neq 0$) and, 
		by Lemma~\ref{lem:seesaw} applied to 
		\[0 \to K \to E \to I \to 0, \] 
		$\varphi(E) = \varphi(I)$ forces $\varphi(K) = \varphi(E)$, contradicting stability; so $K = 0$. If $F$ is stable, then $I \subset F$ with $\varphi(I) = \varphi(F)$ forces $I = F$. The last claim combines the two.
	\end{proof}
	
	\begin{lemma}
		\label{lem:direct_sums}
		If $E$ and $F$ are $(\calA, Z)$-semistable of the same phase $\varphi$, then so is $E \oplus F$.
	\end{lemma}
	
	\begin{proof}
		First, 
		\[ Z(E \oplus F) = Z(E) + Z(F) \] 
		is a sum of two vectors on the ray $\RR_{>0} e^{i \pi \varphi}$, so $\varphi(E \oplus F) = \varphi$. Let $G \subset E \oplus F$ be nontrivial, and put $G' := \ker(G \to F) = G \cap E \subset E$ and $G'' := \image(G \to F) \subset F$, so that $0 \to G' \to G \to G'' \to 0$ is exact. If both $G'$ and $G''$ are nonzero, they have phases $\leqslant \varphi$ by semistability of $E$ and $F$, and Lemma~\ref{lem:seesaw} gives $\varphi(G) \leqslant \varphi$; if one of them vanishes, $G$ is isomorphic to the other and the same conclusion holds.
	\end{proof}

	\begin{lemma}
		\label{lem:semistable_abelian}
		Let $\varphi \in (0, 1]$ and let $\calP_Z(\varphi) \subset \calA$ be the full subcategory consisting of the zero object and the $(\calA, Z)$-semistable objects of phase $\varphi$.
		\begin{enumerate}
			\item If $f \colon E \to F$ is a morphism in $\calP_Z(\varphi)$, then its kernel, image and cokernel in $\calA$ lie in $\calP_Z(\varphi)$.
			\item $\calP_Z(\varphi)$ is closed under extensions in $\calA$.
			\item Consequently $\calP_Z(\varphi)$ is an abelian category, and the inclusion $\calP_Z(\varphi) \subset \calA$ is exact.
			\item A nonzero object $E \in \calP_Z(\varphi)$ is a simple object of $\calP_Z(\varphi)$ if and only if it is $(\calA, Z)$-stable.
		\end{enumerate}
	\end{lemma}

	\begin{proof}
		(1) We may assume $f \neq 0$.
		Let $K$, $I$ and $C$ be the kernel, image and cokernel of $f$ in $\calA$.
		As in the proof of Lemma~\ref{lem:schur}, $\varphi(E) \leqslant \varphi(I) \leqslant \varphi(F)$, so $\varphi(I) = \varphi$, and $I$ is semistable because its subobjects are subobjects of $F$ and hence have phase $\leqslant \varphi$.
		If $K \neq 0$, Lemma~\ref{lem:seesaw} applied to
		\[
		0 \to K \to E \to I \to 0
		\]
		and $\varphi(E) = \varphi(I)$ give $\varphi(K) = \varphi$, and $K \subset E$ is semistable for the same reason as $I$.
		If $C \neq 0$, Lemma~\ref{lem:seesaw} applied to
		\[
		0 \to I \to F \to C \to 0
		\]
		gives $\varphi(C) = \varphi$, and $C$ is semistable by Lemma~\ref{lem:quotients}, because its nonzero quotients are quotients of $F$ and hence have phase $\geqslant \varphi$.

		(2) Let $0 \to E' \to E \to E'' \to 0$ be exact in $\calA$ with $E', E'' \in \calP_Z(\varphi)$; we may assume both nonzero.
		Then $\varphi(E) = \varphi$ by Lemma~\ref{lem:seesaw}.
		For a nontrivial subobject $G \subset E$ put $G' := G \cap E' \subset E'$ and let $G'' \subset E''$ be the image of $G$, so that $0 \to G' \to G \to G'' \to 0$ is exact.
		If both $G'$ and $G''$ are nonzero, they have phases $\leqslant \varphi$, and Lemma~\ref{lem:seesaw} gives $\varphi(G) \leqslant \varphi$;
		if one of them vanishes, $G$ is isomorphic to the other and $\varphi(G) \leqslant \varphi$ as well.
		Hence $E$ is semistable.
		(Lemma~\ref{lem:direct_sums} is the special case of a split extension.)

		(3) $\calP_Z(\varphi)$ is a full additive subcategory of $\calA$ containing $0$ and closed under finite direct sums by (2).
		By (1) every morphism of $\calP_Z(\varphi)$ has a kernel and a cokernel in $\calA$ which lie in $\calP_Z(\varphi)$;
		these satisfy the universal properties of a kernel and a cokernel in $\calP_Z(\varphi)$, since $\calP_Z(\varphi)$ is a full subcategory.
		The canonical morphism from the coimage to the image is computed in $\calA$, where it is an isomorphism.
		Hence $\calP_Z(\varphi)$ is abelian, and the inclusion preserves kernels and cokernels, i.e.\ it is exact.

		(4) Let $E \in \calP_Z(\varphi)$ be nonzero.
		If $E$ is stable and $0 \neq F \subsetneq E$ is a subobject lying in $\calP_Z(\varphi)$, then $\varphi(F) = \varphi$ contradicts \condref{cond:AZ-S}; so $E$ is simple in $\calP_Z(\varphi)$.
		Conversely, if $E$ is not stable, there is a nontrivial proper subobject $F \subset E$ with $\varphi(F) = \varphi$.
		Then $F$ is semistable, since its subobjects are subobjects of $E$, and $\varphi(E / F) = \varphi$ by Lemma~\ref{lem:seesaw}, so $E / F$ is semistable by Lemma~\ref{lem:quotients}.
		Thus $F \subset E$ is a nontrivial proper subobject in $\calP_Z(\varphi)$ with quotient in $\calP_Z(\varphi)$, and $E$ is not simple in $\calP_Z(\varphi)$.
	\end{proof}

	If the abelian category $\calP_Z(\varphi)$ has finite length, then every nonzero object $E \in \calP_Z(\varphi)$ has a finite filtration $0 = E_0 \subset E_1 \subset \ldots \subset E_r = E$ by subobjects in $\calP_Z(\varphi)$ whose quotients $E_i / E_{i-1}$ are simple in $\calP_Z(\varphi)$, i.e.\ $(\calA, Z)$-stable of phase $\varphi$.
	Such a filtration is called a \emph{Jordan--H\"older filtration} of $E$, and by the Jordan--H\"older theorem in $\calP_Z(\varphi)$ its factors are unique up to permutation and isomorphism.

	
	\begin{lemma}
		\label{lem:hom_HN}
		Let $E, F$ be nonzero objects admitting HN filtrations, with HN factors $A_1, \ldots, A_n$ and $B_1, \ldots, B_m$ respectively. If $\Hom(E, F) \neq 0$, then $\varphi^-(E) \leqslant \varphi^+(F)$; more precisely, $\varphi(A_i) \leqslant \varphi(B_j)$ for some $i, j$.
	\end{lemma}
	
	\begin{proof}
		Let $f \colon E \to F$ be nonzero, and let $0 = E_0 \subset \ldots \subset E_n = E$, $0 = F_0 \subset \ldots \subset F_m = F$ be the filtrations. Let $i$ be minimal with $f|_{E_i} \neq 0$. Then $f|_{E_{i-1}} = 0$, so $f|_{E_i}$ factors through a nonzero map $g \colon A_i = E_i / E_{i-1} \to F$. Let $j$ be minimal with $\image g \subset F_j$. Then the composite $A_i \to F_j \to B_j = F_j / F_{j-1}$ is nonzero, since otherwise $\image g \subset F_{j-1}$. By Lemma~\ref{lem:schur}(1) applied to the semistable objects $A_i$ and $B_j$, $\varphi(A_i) \leqslant \varphi(B_j)$. Finally $\varphi^-(E) \leqslant \varphi(A_i)$ and $\varphi(B_j) \leqslant \varphi^+(F)$.
	\end{proof}
	
	
	\subsection{Uniqueness of Harder--Narasimhan filtrations}
	
	\begin{proposition}[{\cite[Thm~2]{Rud97}}, {\cite[Exercise~4.7]{MS17}}]
		\label{prop:HN_unique}
		Let $Z$ be a stability function on $\calA$ and let $E$ be a nonzero object. If
		\[
		0 = E_0 \subset E_1 \subset \ldots \subset E_n = E
		\qquad\text{and}\qquad
		0 = E'_0 \subset E'_1 \subset \ldots \subset E'_m = E
		\]
		are two HN filtrations of $E$ with respect to $(\calA, Z)$, then $n = m$ and $E_i = E'_i$ as subobjects of $E$ for all $i$. In particular the HN factors and the phases $\varphi^{\pm}(E)$ depend only on $E$.
	\end{proposition}
	
	\begin{proof}
		Write $A_i := E_i / E_{i-1}$, $\varphi_i := \varphi(A_i)$ and $A'_j := E'_j / E'_{j-1}$, $\varphi'_j := \varphi(A'_j)$. We argue by induction on $n$.
		
		\step{$E_1 = E'_1$} Let $j$ be minimal with $E_1 \subset E'_j$; it exists since $E'_m = E$. The composite $E_1 \to E'_j \to A'_j$ is nonzero, because otherwise $E_1 \subset E'_{j-1}$. Both $E_1 = A_1$ and $A'_j$ are semistable, so Lemma~\ref{lem:schur}(1) gives $\varphi_1 \leqslant \varphi'_j \leqslant \varphi'_1$. By symmetry $\varphi'_1 \leqslant \varphi_1$. Hence $\varphi_1 = \varphi'_1$, and then $\varphi'_j \geqslant \varphi'_1$ forces $j = 1$ because the phases $\varphi'_j$ strictly decrease; thus $E_1 \subset E'_1$. By symmetry $E'_1 \subset E_1$, so $E_1 = E'_1$.
		
		\step{induction} If $n = 1$, then $E = E_1 = E'_1$, so $m = 1$ as well. If $n > 1$, consider $\bar E := E / E_1 = E / E'_1 \neq 0$. The images $\bar E_i := E_i / E_1$, $i \geqslant 1$, form a chain $0 = \bar E_1 \subset \ldots \subset \bar E_n = \bar E$ with factors $\bar E_i / \bar E_{i-1} \cong E_i / E_{i-1} = A_i$ for $i \geqslant 2$, hence an HN filtration of $\bar E$ of length $n - 1$; similarly for $\bar E'_j := E'_j / E'_1$. By induction $n - 1 = m - 1$ and $\bar E_i = \bar E'_i$ for all $i \geqslant 1$. Since subobjects of $\bar E$ correspond bijectively to subobjects of $E$ containing $E_1 = E'_1$, we get $E_i = E'_i$ for all $i$.
	\end{proof}
	
	\begin{corollary}[{\cite[Prop.~1.9]{Rud97}}]
		\label{cor:HN_sub}
		Let $(\calA, Z)$ be a stability condition and $E$ a nonzero object. Then $\varphi^+(E) = \max \{ \varphi(F) : 0 \neq F \subset E \}$, and the first step $E_1$ of the HN filtration is the unique largest subobject of $E$ of phase $\varphi^+(E)$. Dually $\varphi^-(E) = \min \{ \varphi(G) : E \twoheadrightarrow G \neq 0 \}$.
	\end{corollary}
	
	\begin{proof}
		Let $0 \neq F \subset E$, with HN filtration $0 \subset F_1 \subset \ldots \subset F$. The composite $F_1 \hookrightarrow F \hookrightarrow E$ is a nonzero map from the semistable object $F_1$ to $E$, so Lemma~\ref{lem:hom_HN} gives $\varphi^+(F) = \varphi(F_1) \leqslant \varphi^+(E)$, and $\varphi(F) \leqslant \varphi^+(F)$ by Lemma~\ref{lem:seesaw}. The value $\varphi^+(E)$ is attained by $F = E_1$. Now suppose $\varphi(F) = \varphi^+(E)$. Then $\varphi(F) = \varphi^+(F)$, so by the equality case of Lemma~\ref{lem:seesaw} all HN factors of $F$ have phase $\varphi^+(E)$, i.e.\ $F$ is semistable of phase $\varphi^+(E)$. All HN factors of $E / E_1$ have phase $< \varphi^+(E)$, so the composite $F \to E \to E / E_1$ vanishes by Lemma~\ref{lem:hom_HN}, i.e.\ $F \subset E_1$. The dual statement follows by the same argument applied to quotients, using Lemma~\ref{lem:quotients}.
	\end{proof}
	
	
	\subsection{Existence of Harder--Narasimhan filtrations}
	
	Recall that an abelian category $\calA$ is \emph{noetherian} if every ascending chain $F_1 \subset F_2 \subset \ldots$ of subobjects of any object stabilizes.
	Equivalently, every nonempty family of subobjects of an object has a maximal element with respect to inclusion.
	Throughout this subsection $Z$ is a stability function on $\calA$, and we write $R := \im Z$ and $D := -\re Z$, so that $R(F) \geqslant 0$ for all $F$, and $R(F) = 0$ implies $D(F) > 0$ for $F \neq 0$.
	We say that \emph{$R$ has discrete image} if $R(\Kgroup(\calA))$ is a discrete subgroup of $\RR$;
	in that case either $R \equiv 0$, or $R(\Kgroup(\calA)) = c \ZZ$ for a unique $c > 0$, and then $R(E) \in c \ZZ_{\geqslant 0}$ for every object $E$.
	
	Two elementary identities will be used repeatedly.
	If $F, F' \subset E$ are subobjects, the sequence
	\[
	0 \to F \cap F' \to F \oplus F' \to F + F' \to 0
	\]
	is exact, hence
	\begin{equation}
		\label{eq:modular}
		Z(F + F') + Z(F \cap F') = Z(F) + Z(F') ,
	\end{equation}
	and the same holds for $R$ and $D$ separately.
	If $F \subset E$ and $R(F) = R(E)$, then $R(E / F) = 0$, so $D(E / F) \geqslant 0$ and
	\begin{equation}
		\label{eq:same_rank}
		D(F) \leqslant D(E) \qquad \text{whenever } F \subset E \text{ and } R(F) = R(E).
	\end{equation}
	
	\begin{lemma}[{\cite[Lemma~4.9]{MS17}}]
		\label{lem:D_bounded}
		Assume that
		\begin{itemize}
			\item $\calA$ is noetherian and
			\item that $R$ has discrete image in $ \mathbb{R} $.
		\end{itemize}
		Then for every object $E$ there is a constant $D_E \in \RR$ such that $D(F) \leqslant D_E$ for every subobject $F \subset E$.
	\end{lemma}
	
	\begin{proof}
		If $R \equiv 0$, then every $F \subset E$ satisfies $R(F) = R(E)$ and \eqref{eq:same_rank} gives $D(F) \leqslant D(E)$.
		Otherwise write $R(\Kgroup(\calA)) = c \ZZ$ with $c > 0$ and argue by induction on $k(E) := R(E) / c \in \ZZ_{\geqslant 0}$.
		
		If $k(E) = 0$, then $R(F) = 0 = R(E)$ for every $F \subset E$, and \eqref{eq:same_rank} gives $D(F) \leqslant D(E)$.
		
		Let $k(E) > 0$ and assume the lemma for all objects $E'$ with $k(E') < k(E)$.
		Suppose, for a contradiction, that there are subobjects $F_n \subset E$, $n \in \ZZ_{>0}$, with $D(F_n) \to \infty$.
		By \eqref{eq:same_rank} only finitely many of them can satisfy $R(F_n) = R(E)$, so after discarding these we may assume $R(F_n) < R(E)$ for all $n$.
		We construct indices $n_1 < n_2 < \ldots$ such that the subobjects $G_k := F_{n_1} + \ldots + F_{n_k}$ satisfy
		\[
		R(G_k) < R(E)
		\qquad\text{and}\qquad
		D(G_k) > D(G_{k-1}) \quad (k \geqslant 2).
		\]
		Put $n_1 := 1$.
		Given $n_1 < \ldots < n_{k-1}$, the object $G_{k-1}$ satisfies $k(G_{k-1}) < k(E)$, so by the induction hypothesis there is a constant $D'$ with $D(F') \leqslant D'$ for all $F' \subset G_{k-1}$.
		For $n > n_{k-1}$, identity \eqref{eq:modular} and $G_{k-1} \cap F_n \subset G_{k-1}$ give
		\[
		D(G_{k-1} + F_n) = D(G_{k-1}) + D(F_n) - D(G_{k-1} \cap F_n) \geqslant D(G_{k-1}) + D(F_n) - D' ,
		\]
		which tends to $\infty$ as $n \to \infty$.
		Choose $n_k > n_{k-1}$ with 
		\[
		D(G_{k-1} + F_{n_k}) > \max \{ D(E), D(G_{k-1}) \}.
		\]
		Then $R(G_{k-1} + F_{n_k}) < R(E)$ by \eqref{eq:same_rank}, and $D(G_k) > D(G_{k-1})$ as required.
		Since $D(G_k) > D(G_{k-1})$, the inclusion $G_{k-1} \subset G_k$ is strict, so $G_1 \subsetneq G_2 \subsetneq \ldots$ is an infinite strictly ascending chain of subobjects of $E$, contradicting noetherianity.
	\end{proof}
	
	\begin{proposition}[{\cite[Prop.~4.10]{MS17}}; cf.\ {\cite[Thm~2]{Rud97}}, {\cite[Prop.~2.4]{Bri07}}]
		\label{prop:HN_existence}
		Let $Z$ be a stability function on $\calA$.
		Assume that $\calA$ is noetherian and that $R = \im Z$ has discrete image.
		Then every nonzero object of $\calA$ admits a Harder--Narasimhan filtration, i.e.\ $(\calA, Z)$ is a stability condition.
	\end{proposition}
	
	\begin{proof}
		If $R \equiv 0$, then every nonzero object has phase $1$, every nonzero object is semistable, and $0 \subset E$ is an HN filtration.
		So assume $R(\Kgroup(\calA)) = c \ZZ$ with $c > 0$, and let $E \neq 0$.
		
		\step{the part of rank zero}
		The family of subobjects $F \subset E$ with $R(F) = 0$ is nonempty (it contains $0$), so it has a maximal element $T$.
		If $F \subset E$ is any subobject with $R(F) = 0$, then $R(T + F) = R(T) + R(F) - R(T \cap F) = -R(T \cap F) \leqslant 0$ by \eqref{eq:modular}, hence $R(T + F) = 0$ and $T + F = T$ by maximality, i.e.\ $F \subset T$.
		Thus $T$ is the largest subobject of $E$ of rank zero.
		If $T \neq 0$, it is semistable of phase $1$, because all its nonzero subobjects have $R = 0$, hence phase $1$.
		Moreover the quotient $\bar E := E / T$ has no nonzero subobject of rank zero:
		the preimage $F \subset E$ of such a subobject $\bar F$ satisfies $R(F) = R(T) + R(\bar F) = 0$, hence $F \subset T$ and $\bar F = 0$.
		Consequently every nonzero subobject of $\bar E$ has $R \geqslant c$, hence phase $< 1$.
		If $\bar E = 0$, then $E = T$ is semistable and we are done.
		Otherwise, if $0 = \bar E_0 \subset \bar E_1 \subset \ldots \subset \bar E_m = \bar E$ is an HN filtration of $\bar E$, then its factors have phases $< 1$, and the preimages
		\[
		0 \subset T \subset \pi^{-1}(\bar E_1) \subset \ldots \subset \pi^{-1}(\bar E_m) = E
		\]
		form an HN filtration of $E$ (with the first step omitted if $T = 0$), since the factors are $T$ (of phase $1$) followed by the factors of the filtration of $\bar E$.
		Hence we may and do assume from now on that
		\begin{equation}
			\label{eq:no_torsion}
			\text{every nonzero subobject } F \subset E \text{ satisfies } R(F) \geqslant c .
		\end{equation}
		\begin{illustration}
		In the pictures accompanying this proof we draw the plane of $Z$ in the usual way, with $\re Z$ to the right and $\im Z$ upwards;
		a subobject $F \subset E$ is represented by the point $Z(F)$, whose height is the rank $R(F)$ and whose degree $D(F) = -\re Z(F)$ grows to the left, so that the point with coordinates $(R, D) = (r, d)$ in the proof is the point $-d + i r$ of the plane.
		In Step~1, the subobjects of rank zero are the points on the negative real axis, $T$ is the leftmost of them, and passing from $E$ to $\bar E = E / T$ shifts every point to the right by $D(T)$.

			\end{tikzpicture}
		\end{center}
		\end{illustration}
		
		\step{the Harder--Narasimhan polygon}
		For $r \in c \ZZ$ with $0 \leqslant r \leqslant R(E)$ let $\mathcal{S}_r$ be the set of subobjects $F \subset E$ with $R(F) = r$, and call $r$ \emph{occupied} if $\mathcal{S}_r \neq \emptyset$.
		The set $\Lambda_E$ of occupied values is finite and contains $0$ and $R(E)$.
		For $r \in \Lambda_E$ put
		\[
		d(r) := \sup \{ D(F) : F \in \mathcal{S}_r \} ,
		\]
		which is finite by Lemma~\ref{lem:D_bounded}.
		By \eqref{eq:no_torsion} we have $\mathcal{S}_0 = \{ 0 \}$ and $d(0) = 0$, and by \eqref{eq:same_rank} we have $d(R(E)) = D(E)$, attained by $E$ itself.
		Let $p \colon [0, R(E)] \to \RR$ be the upper concave envelope of the finite set of points $(r, d(r))$, $r \in \Lambda_E$, i.e.\ the smallest concave function with $p(r) \geqslant d(r)$ for all $r \in \Lambda_E$.
		It is piecewise linear, and its graph is the upper boundary of the convex hull of these points.
		Let $0 = r_0 < r_1 < \ldots < r_n = R(E)$ be the endpoints together with the points where the slope of $p$ changes; we call them the \emph{vertices}.
		They belong to $\Lambda_E$, satisfy $p(r_i) = d(r_i) =: d_i$, and the slopes
		\[
		s_i := \frac{d_i - d_{i-1}}{r_i - r_{i-1}} , \qquad i = 1, \ldots, n ,
		\]
		are strictly decreasing, $s_1 > s_2 > \ldots > s_n$, because $p$ is concave and changes slope at every $r_i$ with $0 < i < n$.
		On $[r_{i-1}, r_i]$ the function $p$ is affine with slope $s_i$.
		Finally, for every subobject $F \subset E$ we have $R(F) \in \Lambda_E$ and
		\begin{equation}
			\label{eq:below_p}
			D(F) \leqslant d(R(F)) \leqslant p(R(F)) .
		\end{equation}

		\step{the suprema at the vertices are attained}
		We claim that for every vertex $r_i$ there is $F_i \in \mathcal{S}_{r_i}$ with $D(F_i) = d_i$ (this is \cite[Exercise~4.11]{MS17}).
		For $i = 0$ take $F_0 = 0$ and for $i = n$ take $F_n = E$.
		Let $0 < i < n$ and choose a number $s$ with $s_i > s > s_{i+1}$.
		Consider the linear functional $\ell(x, y) := y - s x$ on $\RR^2$ and put $M := \ell(r_i, d_i) = d_i - s r_i$.
		Since $s$ lies between the slopes of $p$ to the left and to the right of $r_i$, concavity gives $p(x) \leqslant d_i + s (x - r_i)$ for all $x \in [0, R(E)]$, with equality only for $x = r_i$.
		Together with \eqref{eq:below_p} this shows
		\begin{equation}
			\label{eq:ell_bound}
			\ell\big( R(F), D(F) \big) \leqslant M \qquad \text{for every subobject } F \subset E ,
		\end{equation}
		and moreover, since $\Lambda_E$ is finite,
		\[
		\gamma := M - \max \big\{ p(r) - s r : r \in \Lambda_E,\ r \neq r_i \big\} > 0 ,

			\ell(F + F') = \ell(F) + \ell(F') - \ell(F \cap F') \geqslant \ell(F) + \ell(F') - M .
		\end{equation}
		Now choose $F^{(1)}, F^{(2)}, \ldots \in \mathcal{S}_{r_i}$ with $D(F^{(j)}) \to d_i$, and put $\varepsilon_j := M - \ell(F^{(j)}) = d_i - D(F^{(j)}) \geqslant 0$.
		Passing to a subsequence we may assume $\sum_j \varepsilon_j < \gamma$.
		Let $G_k := F^{(1)} + \ldots + F^{(k)}$.
		By induction on $k$, \eqref{eq:ell_sum} gives
		\[
		\ell(G_k) \geqslant \sum_{j=1}^{k} \ell(F^{(j)}) - (k-1) M = M - \sum_{j=1}^{k} \varepsilon_j > M - \gamma .
		\]
		The chain $G_1 \subset G_2 \subset \ldots$ stabilizes by noetherianity, say $G_k = G$ for $k \geqslant N$.
		Then $\ell(G) > M - \gamma$, hence $R(G) = r_i$, and $G \supset F^{(j)}$ for every $j$.
		For $j \geqslant N$ the quotient $G / F^{(j)}$ has $R(G / F^{(j)}) = r_i - r_i = 0$, so $D(G / F^{(j)}) \geqslant 0$ and $D(G) \geqslant D(F^{(j)})$.
		Letting $j \to \infty$ we obtain $D(G) \geqslant d_i$, hence $D(G) = d_i$ by definition of $d_i$.
		Thus $F_i := G$ has the required property.
		
		\step{the vertices form a chain}
		Fix $1 \leqslant i \leqslant n$ and consider the functional $\ell_i(x, y) := y - s_i x$ and the number $M_i := \ell_i(r_{i-1}, d_{i-1}) = \ell_i(r_i, d_i)$.
		By concavity, $p(x) \leqslant d_{i-1} + s_i (x - r_{i-1})$ for all $x$, with equality exactly for $x \in [r_{i-1}, r_i]$, because the slope of $p$ is $> s_i$ to the left of $r_{i-1}$ and $< s_i$ to the right of $r_i$.
		Hence, by \eqref{eq:below_p}, $\ell_i(F) \leqslant M_i$ for every $F \subset E$, and $\ell_i(F) = M_i$ implies $r_{i-1} \leqslant R(F) \leqslant r_i$ and $D(F) = p(R(F))$.
		Apply \eqref{eq:modular} to $F_{i-1}$ and $F_i$:
		\[
		\ell_i(F_{i-1} \cap F_i) + \ell_i(F_{i-1} + F_i) = \ell_i(F_{i-1}) + \ell_i(F_i) = 2 M_i .
		\]
		Both summands on the left are $\leqslant M_i$, so both equal $M_i$.
		\begin{illustration}
		In the picture, the points $Z(F_{i-1} \cap F_i)$ and $Z(F_{i-1} + F_i)$ are symmetric with respect to the midpoint of the edge from $Z(F_{i-1})$ to $Z(F_i)$ by \eqref{eq:modular}, and both must lie on the line $\ell_i = M_i$ carrying that edge or to the right of it; the dashed configuration is therefore impossible, and both points are forced onto the line, hence onto the edge.

		Therefore the quotient $F_{i-1} / (F_{i-1} \cap F_i)$ has $R = 0$ and $D = 0$, i.e.\ $Z = 0$, so it is zero by \condref{cond:Z}.
		Hence $F_{i-1} \subset F_i$.
		We have obtained a chain
		\[
		0 = F_0 \subset F_1 \subset \ldots \subset F_n = E .
		\]
		
		\step{the factors are semistable with strictly decreasing phases}
		Put $A_i := F_i / F_{i-1}$.
		Then $Z(A_i) = Z(F_i) - Z(F_{i-1})$, i.e.\ $R(A_i) = r_i - r_{i-1} > 0$ and $D(A_i) = d_i - d_{i-1}$, so $A_i \neq 0$ and its generalized slope is $M(A_i) = s_i$.
		Since $s_1 > \ldots > s_n$, the phases of the $A_i$ strictly decrease.
		It remains to show that $A_i$ is semistable.
		Let $S \subset A_i$ be a nontrivial subobject and $F' \subset F_i$ its preimage, so that $F_{i-1} \subsetneq F' \subset F_i$ and $Z(S) = Z(F') - Z(F_{i-1})$.
		Then $r_{i-1} \leqslant R(F') \leqslant r_i$, because $R(F' / F_{i-1}) \geqslant 0$ and $R(F_i / F') \geqslant 0$, and \eqref{eq:below_p} together with the affine form of $p$ on $[r_{i-1}, r_i]$ gives
		\[
		D(F') \leqslant p(R(F')) = d_{i-1} + s_i \big( R(F') - r_{i-1} \big) .
		\]
		In other words $D(S) \leqslant s_i R(S)$.

		Thus $A_i$ is semistable, and $0 = F_0 \subset \ldots \subset F_n = E$ is an HN filtration of $E$.

	\begin{corollary}[Harder--Narasimhan \cite{HN75}]
		\label{cor:HN_curves}
		Let $C$ be a smooth projective curve over a field $\kk$ and $Z(E) = -\deg E + i \operatorname{rk} E$ on $\mathrm{Coh}(C)$.
		Then $(\mathrm{Coh}(C), Z)$ is a stability condition.
	\end{corollary}
	
	\begin{proof}
		$\mathrm{Coh}(C)$ is noetherian and $R = \operatorname{rk}$ takes values in $\ZZ$, so Proposition~\ref{prop:HN_existence} applies.
	\end{proof}
	
	\begin{remark}
		The proposition does not cover Example~\ref{ex:finite_dim_algebra} in general, since there $\im Z$ need not have discrete image; that example was treated by a direct finite-length argument.
		Bridgeland's original formulation \cite[Prop.~2.4]{Bri07} replaces the two hypotheses by two chain conditions on phases, which cover both situations at once;
		noetherianity excludes infinite descending chains of quotients with decreasing phases, and discreteness of $R$ excludes infinite descending chains of subobjects with increasing phases.
	\end{remark}
	
	
	\subsection{Examples}
	
	\begin{example}
		Let $C$ be a smooth projective curve over a field $\kk$, $\calA = \mathrm{Coh}(C)$, and
		\[
		Z(E) := -\deg E + i \operatorname{rk} E .
		\]
		This is a stability function: for a nonzero sheaf $E$ either $\operatorname{rk} E > 0$, or $E$ is a torsion sheaf, in which case $\deg E$ is the length of $E$, which is positive, so $Z(E) \in \RR_{<0}$. For $\operatorname{rk} E > 0$ the generalized slope $M(E) = \deg E / \operatorname{rk} E$ is the usual slope, and torsion sheaves have $M = +\infty$; thus $(\calA, Z)$-(semi)stability is slope (semi)stability, and semistable sheaves of positive rank are torsion-free. The Harder--Narasimhan property is Corollary~\ref{cor:HN_curves}, originally due to Harder and Narasimhan \cite{HN75}; the convex-polygon proof goes back to Shatz \cite{Sha77}. So $(\mathrm{Coh}(C), Z)$ is a stability condition.
	\end{example}
	
	\begin{example}
		\label{ex:surfaces}
		Let $X$ be a smooth projective variety of dimension $n \geqslant 2$ and $\omega$ an ample divisor class. The homomorphism $Z(E) := -\omega^{n-1} \cdot c_1(E) + i \, \omega^n \cdot \operatorname{rk} E$ on $\Kgroup(\mathrm{Coh}(X))$ is not a stability function: for a nonzero sheaf $T$ supported in codimension $\geqslant 2$ both $\operatorname{rk} T$ and $c_1(T)$ vanish, so $Z(T) = 0 \notin \HH$. One can show that no homomorphism factoring through the Chern character is a stability function on $\mathrm{Coh}(X)$ (see \cite[Example~4.3]{MS17}, where this is deduced from \cite[Lemma~2.7]{Tod09}); this is the reason for passing to other hearts in the derived category.
	\end{example}
	
	\begin{example}[{\cite[Exercise~4.13]{MS17}}]
		\label{ex:finite_dim_algebra}
		Let $A$ be a finite-dimensional algebra over a field $\kk$, $\calA = \operatorname{mod}\text{-}A$ the category of finite-dimensional right $A$-modules, and $S_1, \ldots, S_m$ representatives of the isomorphism classes of simple modules. Every $M \in \calA$ has a composition series, and by the Jordan--H\"older theorem the multiplicity $[M : S_j]$ of $S_j$ among its factors is well defined and additive in short exact sequences. Hence
		\[
		\Kgroup(\calA) \longrightarrow \ZZ^m, \qquad [M] \longmapsto \big( [M : S_1], \ldots, [M : S_m] \big)
		\]
		is a well-defined homomorphism; it is surjective, and it is injective because every class $[M]$ equals $\sum_j [M : S_j] [S_j]$. So $[S_1], \ldots, [S_m]$ form a basis of $\Kgroup(\calA) \cong \ZZ^m$, and for any $z_1, \ldots, z_m \in \CC$ there is a unique homomorphism $Z$ with $Z(S_j) = z_j$, namely $Z(M) = \sum_j [M : S_j] z_j$.
		
		Now let $z_1, \ldots, z_m \in \HH$. Then $Z$ is a stability function, since $Z(M)$ is a sum of vectors in $\HH$ with non-negative coefficients, not all zero. We claim that $(\calA, Z)$ is a stability condition. Let $M \neq 0$; we construct an HN filtration by induction on the length $\ell(M)$.
		
		For a subobject $N \subset M$ one has $0 \leqslant [N : S_j] \leqslant [M : S_j]$, so the classes $[N]$, hence the phases $\varphi(N)$, range over a finite set. Let $\varphi_1$ be the maximal phase of a nontrivial subobject, and among the nontrivial subobjects of phase $\varphi_1$ choose $M_1$ of maximal length. Then $M_1$ is semistable: its subobjects are subobjects of $M$, hence have phase $\leqslant \varphi_1$. Moreover every nontrivial $N \subset M$ with $\varphi(N) = \varphi_1$ satisfies $N \subset M_1$. Indeed, from the exact sequence
		\[
		0 \to M_1 \cap N \to M_1 \oplus N \to M_1 + N \to 0
		\]
		and Lemma~\ref{lem:direct_sums}, $\varphi(M_1 \oplus N) = \varphi_1$, while $\varphi(M_1 \cap N) \leqslant \varphi_1$ if $M_1 \cap N \neq 0$; so Lemma~\ref{lem:seesaw} gives $\varphi(M_1 + N) \geqslant \varphi_1$, hence $= \varphi_1$ by maximality of $\varphi_1$. By maximality of $\ell(M_1)$, $\ell(M_1 + N) \leqslant \ell(M_1)$, so $M_1 + N = M_1$. (If $M_1 \cap N = 0$, then $M_1 + N \cong M_1 \oplus N$ has phase $\varphi_1$ and larger length, which is impossible, so this case does not occur.)
		
		Finally, every nontrivial subobject $N' \subset M / M_1$ has $\varphi(N') < \varphi_1$: if $N \subset M$ is its preimage, then $0 \to M_1 \to N \to N' \to 0$ with $\varphi(N) \leqslant \varphi_1 = \varphi(M_1)$, so Lemma~\ref{lem:seesaw} gives $\varphi(N') \leqslant \varphi(N) \leqslant \varphi_1$, and $\varphi(N') = \varphi_1$ would force $\varphi(N) = \varphi_1$, hence $N \subset M_1$ and $N' = 0$. Since $\ell(M / M_1) < \ell(M)$, by induction $M / M_1$ has an HN filtration; its first factor has phase $< \varphi_1$ by what we just proved, so the preimage of this filtration in $M$, preceded by $M_1$, is an HN filtration of $M$. If $M_1 = M$ there is nothing to do.
	\end{example}
	
	\begin{remark}
		The argument of Example~\ref{ex:finite_dim_algebra} uses only that $\calA$ has finite length and that the classes of subobjects of a given object form a finite set; it applies verbatim to any abelian category of finite length with finitely many simple objects.
	\end{remark}
	
	\begin{example}[a stability function without the Harder--Narasimhan property]
		\label{ex:no_HN}
		Let $\calA$ be the category of finitely generated $\ZZ$-graded modules over the polynomial ring $\kk[x]$ with $\deg x = 1$, and graded homomorphisms of degree $0$. For $M = \bigoplus_n M_n \in \calA$ the pieces $M_n$ vanish for $n \ll 0$ and have bounded dimension, so
		\[
		Z(M) := \sum_{n \in \ZZ} \big( -2^{-n} + i \, 4^{-n} \big) \dim_\kk M_n
		\]
		converges absolutely. Short exact sequences in $\calA$ are exact in each degree, so $Z$ is additive, and $\im Z(M) > 0$ for $M \neq 0$; thus $Z$ is a stability function.
		
		Consider $E := \kk[x]$, generated in degree $0$. Its graded submodules are exactly the ideals $(x^j) \cong \kk[x](-j)$, $j \geqslant 0$, and $0$. For $j \geqslant 0$,
		\[
		Z\big( (x^j) \big) = \sum_{n \geqslant j} \big( -2^{-n} + i \, 4^{-n} \big) = -2^{1-j} + i \, \tfrac{4}{3} \, 4^{-j},
		\]
		so $\tan\big( \pi - \pi \varphi((x^j)) \big) = \tfrac{2}{3} \cdot 2^{-j}$ is strictly decreasing in $j$; hence the phases $\varphi((x^j))$ strictly increase with $j$. Consequently no nonzero subobject of $E$ is semistable: $(x^j)$ has the subobject $(x^{j+1})$ of strictly larger phase. But the first step $E_1$ of an HN filtration of $E$ would be a nonzero semistable subobject of $E$. Hence $E$ has no HN filtration, and $(\calA, Z)$ is not a stability condition.
	\end{example}
	
	\section{t-structures and their hearts}
	
	\subsection{Conventions on triangulated categories}
	
	Throughout this section $\calD$ is an essentially small triangulated category with shift functor $[1]$.
	A \emph{triangle} always means a distinguished triangle $X \xrightarrow{f} Y \xrightarrow{g} Z \xrightarrow{h} X[1]$;
	we call $Z$ a \emph{cone} of $f$ and write $Z = \Cone(f)$, keeping in mind that a cone is unique only up to (non-unique) isomorphism.
	We shall freely use the following standard facts, for which we refer to \cite[Ch.~IV]{GM03}.
	\begin{itemize}
		\item[] \condlabel{TR-rot}{cond:TR-rot} (\emph{rotation}) If $X \xrightarrow{f} Y \xrightarrow{g} Z \xrightarrow{h} X[1]$ is a triangle, then so are $Y \xrightarrow{g} Z \xrightarrow{h} X[1] \xrightarrow{-f[1]} Y[1]$ and $Z[-1] \xrightarrow{-h[-1]} X \xrightarrow{f} Y \xrightarrow{g} Z$.
		\item[] \condlabel{TR-coh}{cond:TR-coh} (\emph{Hom is cohomological}) For every object $T$ and every triangle $X \to Y \to Z \to X[1]$ the sequences
		\begin{align*}
			&\Hom(T, X) \to \Hom(T, Y) \to \Hom(T, Z), \\
			&\Hom(Z, T) \to \Hom(Y, T) \to \Hom(X, T)
		\end{align*}
		are exact; combined with \condref{cond:TR-rot} this gives the long exact sequences $\ldots \to \Hom(T, Z[-1]) \to \Hom(T, X) \to \Hom(T, Y) \to \Hom(T, Z) \to \Hom(T, X[1]) \to \ldots$ and dually.
		In particular the composite of two consecutive maps in a triangle is zero.
		\item[] \condlabel{TR-5}{cond:TR-5} (\emph{five lemma}) If $(\alpha, \beta, \gamma)$ is a morphism of triangles and two of $\alpha, \beta, \gamma$ are isomorphisms, so is the third.
		In particular, in a triangle $0 \to Y \to Z \to 0$ the map $Y \to Z$ is an isomorphism.
		\item[] \condlabel{TR-oct}{cond:TR-oct} (\emph{octahedral axiom}) For composable morphisms $X \xrightarrow{f} Y \xrightarrow{g} Z$ and chosen cones there is a triangle
		\[
		\Cone(f) \to \Cone(g f) \to \Cone(g) \to \Cone(f)[1]
		\]
		whose maps are compatible with the maps from $Y$, $Z$ and to $X[1]$, $Y[1]$.
	\end{itemize}
	
	All subcategories are full and closed under isomorphisms.
	For a class of objects $\mathcal{S}$ we write
	\begin{align*}
		\mathcal{S}^{\perp} &:= \defineset{ Y \in \calD }{ \Hom(S, Y) = 0 \text{ for all } S \in \mathcal{S} }, \\
		{}^{\perp}\mathcal{S} &:= \defineset{ X \in \calD }{ \Hom(X, S) = 0 \text{ for all } S \in \mathcal{S} },
	\end{align*}
	and $\mathcal{S}[n] := \{ S[n] : S \in \mathcal{S} \}$.
	A subcategory $\mathcal{C}$ is \emph{closed under extensions} if for every triangle $X \to Y \to Z \to X[1]$ with $X, Z \in \mathcal{C}$ also $Y \in \mathcal{C}$;
	the \emph{extension closure} $\langle \mathcal{S} \rangle$ of a class $\mathcal{S}$ is the smallest subcategory containing $\mathcal{S}$ and $0$ and closed under extensions (it is described concretely in Proposition~\ref{prop:extension_closure} below).
	Since the split triangle $X \to X \oplus Z \to Z \xrightarrow{0} X[1]$ is distinguished, a subcategory closed under extensions is closed under finite direct sums.
	
	\begin{lemma}
		\label{lem:orthogonals}
		For every class of objects $\mathcal{S}$ the subcategories $\mathcal{S}^{\perp}$ and ${}^{\perp}\mathcal{S}$ contain $0$ and are closed under isomorphisms, extensions, finite direct sums and direct summands.
		If moreover $\mathcal{S}[1] = \mathcal{S}$, then they are closed under $[1]$ and $[-1]$.
	\end{lemma}
	
	\begin{proof}
		We treat ${}^{\perp}\mathcal{S}$; the case of $\mathcal{S}^{\perp}$ is dual.
		Closure under isomorphisms and the membership of $0$ are clear.
		If $X \to Y \to Z \to X[1]$ is a triangle with $\Hom(X, S) = \Hom(Z, S) = 0$ for some $S \in \mathcal{S}$, then $\Hom(Y, S) = 0$ by the exact sequence
		\[
		\Hom(Z, S) \to \Hom(Y, S) \to \Hom(X, S)
		\]
		of \condref{cond:TR-coh}; this gives closure under extensions, hence under finite direct sums.
		Closure under direct summands follows from $\Hom(X \oplus X', S) = \Hom(X, S) \oplus \Hom(X', S)$.
		Finally, if $\mathcal{S}[1] = \mathcal{S}$, then $\Hom(X[\pm 1], S) = \Hom(X, S[\mp 1])$ vanishes for all $S \in \mathcal{S}$ as soon as $\Hom(X, -)$ vanishes on $\mathcal{S}$.
	\end{proof}
	
	\subsection{Definition}
	
	A \emph{t-structure} on $\calD$ (\cite[D\'ef.~1.3.1]{BBD82}) is a pair $(\Dle{0}, \Dge{0})$ of subcategories of $\calD$ such that, with the notation
	\[
	\Dle{n} := \Dle{0}[-n], \qquad \Dge{n} := \Dge{0}[-n] \qquad (n \in \ZZ),
	\]
	the following conditions hold:
	\begin{itemize}
		\item[] \condlabel{T1}{cond:T1} $\Dle{-1} \subset \Dle{0}$ and $\Dge{1} \subset \Dge{0}$, i.e.\ $\Dle{0}[1] \subset \Dle{0}$ and $\Dge{0}[-1] \subset \Dge{0}$;
		\item[] \condlabel{T2}{cond:T2} $\Hom(X, Y) = 0$ for all $X \in \Dle{0}$ and $Y \in \Dge{1}$;
		\item[] \condlabel{T3}{cond:T3} for every object $E \in \calD$ there is a triangle
		\[
		A \to E \to B \to A[1] \qquad \text{with } A \in \Dle{0} \text{ and } B \in \Dge{1}.
		\]
	\end{itemize}
	The subcategory $\Dle{0}$ is called the \emph{aisle} and $\Dge{0}$ the \emph{coaisle} of the t-structure.
	Applying the shift, \condref{cond:T1}--\condref{cond:T3} hold with $0$ replaced by any $n \in \ZZ$:
	$\Dle{n-1} \subset \Dle{n}$, $\Dge{n+1} \subset \Dge{n}$, $\Hom(\Dle{n}, \Dge{n+1}) = 0$, and every object sits in a triangle $A \to E \to B \to A[1]$ with $A \in \Dle{n}$, $B \in \Dge{n+1}$.
	By induction, $\Dle{n} \subset \Dle{m}$ and $\Dge{m} \subset \Dge{n}$ for $n \leqslant m$.
	
	The intuition comes from the standard example.
	For an abelian category $\calA$ and $\calD = \D^{\mathrm{b}}(\calA)$, the complexes with cohomology concentrated in degrees $\leqslant 0$, respectively $\geqslant 0$, form a t-structure, and the triangle in \condref{cond:T3} is the usual truncation 
	\[
	\tau^{\leqslant 0} E \to E \to \tau^{\geqslant 1} E
	\] 
	of complexes; see Example~\ref{ex:standard_t} below.
	
	\subsection{Truncation functors}
	
	\begin{lemma}[{\cite[Prop.~1.3.3]{BBD82}}]
		\label{lem:truncation_unique}
		Let $(\Dle{0}, \Dge{0})$ satisfy \condref{cond:T1} and \condref{cond:T2}, let $E \in \calD$, and let $A \xrightarrow{a} E \xrightarrow{b} B \to A[1]$ be a triangle with $A \in \Dle{0}$ and $B \in \Dge{1}$.
		\begin{enumerate}
			\item For every $T \in \Dle{0}$ the map $a_* \colon \Hom(T, A) \to \Hom(T, E)$ is bijective, and for every $T \in \Dge{1}$ the map $b^* \colon \Hom(B, T) \to \Hom(E, T)$ is bijective.
			\item If $A' \xrightarrow{a'} E \xrightarrow{b'} B' \to A'[1]$ is another triangle with $A' \in \Dle{0}$ and $B' \in \Dge{1}$, then there are unique morphisms $\alpha \colon A \to A'$ with $a' \alpha = a$ and $\beta \colon B \to B'$ with $\beta b = b'$; both are isomorphisms, and $(\alpha, \id_E, \beta)$ is an isomorphism of triangles.
		\end{enumerate}
	\end{lemma}
	
	\begin{proof}
		(1) By \condref{cond:TR-coh} and \condref{cond:TR-rot} the sequence
		\[
		\Hom(T, B[-1]) \to \Hom(T, A) \xrightarrow{a_*} \Hom(T, E) \to \Hom(T, B)
		\]
		is exact.
		For $T \in \Dle{0}$ the last term vanishes by \condref{cond:T2}, and so does the first, because $B[-1] \in \Dge{2} \subset \Dge{1}$ by \condref{cond:T1}.
		Hence $a_*$ is bijective.
		Dually, the sequence 
		\[
		\Hom(A[1], T) \to \Hom(B, T) \xrightarrow{b^*} \Hom(E, T) \to \Hom(A, T)
		\] 
		is exact, and for $T \in \Dge{1}$ both outer terms vanish, since $A[1] \in \Dle{-1} \subset \Dle{0}$.
		
		(2) Applying (1) to the second triangle with $T = A \in \Dle{0}$ gives a unique $\alpha \colon A \to A'$ with $a' \alpha = a$;
		applying it to the first triangle with $T = A'$ gives a unique $\alpha' \colon A' \to A$ with $a \alpha' = a'$.
		Then $a (\alpha' \alpha) = a' \alpha = a = a \, \id_A$, so $\alpha' \alpha = \id_A$ by the uniqueness in (1), and likewise $\alpha \alpha' = \id_{A'}$.
		In the same way there is a unique $\beta \colon B \to B'$ with $\beta b = b'$, and it is an isomorphism.
		By the axioms of a triangulated category the commutative square $a' \alpha = a \, \id_E$ extends to a morphism of triangles $(\alpha, \id_E, \beta'')$ for some $\beta'' \colon B \to B'$;
		then $\beta'' b = b'$, so $\beta'' = \beta$ by uniqueness, and $(\alpha, \id_E, \beta)$ is a morphism of triangles, which is an isomorphism by \condref{cond:TR-5}.
	\end{proof}
	
	Let now $(\Dle{0}, \Dge{0})$ be a t-structure.
	For every object $E$ we fix a triangle as in \condref{cond:T3} and write it
	\begin{equation}
		\label{eq:truncation_triangle}
		\tau^{\leqslant 0} E \to E \to \tau^{\geqslant 1} E \to (\tau^{\leqslant 0} E)[1] .
	\end{equation}
	By Lemma~\ref{lem:truncation_unique}(2) it is unique up to unique isomorphism.
	For a morphism $f \colon E \to E'$ there is a unique morphism $\tau^{\leqslant 0} f \colon \tau^{\leqslant 0} E \to \tau^{\leqslant 0} E'$ compatible with the maps to $E$ and $E'$, namely the preimage of $f \circ (\tau^{\leqslant 0} E \to E)$ under the bijection of Lemma~\ref{lem:truncation_unique}(1) with $T = \tau^{\leqslant 0} E$;
	uniqueness implies $\tau^{\leqslant 0}(g f) = \tau^{\leqslant 0} g \circ \tau^{\leqslant 0} f$ and $\tau^{\leqslant 0} \id = \id$.
	Thus $\tau^{\leqslant 0}$ is a functor $\calD \to \Dle{0}$, and Lemma~\ref{lem:truncation_unique}(1) says precisely that it is right adjoint to the inclusion $\Dle{0} \hookrightarrow \calD$.
	Dually $\tau^{\geqslant 1} \colon \calD \to \Dge{1}$ is a functor, left adjoint to the inclusion $\Dge{1} \hookrightarrow \calD$.
	For $n \in \ZZ$ put
	\[
	\tau^{\leqslant n} := [-n] \circ \tau^{\leqslant 0} \circ [n], \qquad \tau^{\geqslant n} := [-n+1] \circ \tau^{\geqslant 1} \circ [n-1] ;
	\]
	these are right, respectively left, adjoint to the inclusions of $\Dle{n}$ and $\Dge{n}$, and for every $E$ there is a triangle $\tau^{\leqslant n} E \to E \to \tau^{\geqslant n+1} E \to (\tau^{\leqslant n} E)[1]$, unique up to unique isomorphism among triangles with first term in $\Dle{n}$ and third term in $\Dge{n+1}$.
	We call these the \emph{truncation triangles} and $\tau^{\leqslant n}$, $\tau^{\geqslant n}$ the \emph{truncation functors}.
	
	\begin{lemma}[{\cite[1.3.4]{BBD82}}]
		\label{lem:t_orthogonal}
		Let $(\Dle{0}, \Dge{0})$ be a t-structure. Then:
		\begin{enumerate}
			\item $\Dge{1} = (\Dle{0})^{\perp}$ and $\Dle{0} = {}^{\perp}(\Dge{1})$; in particular each of $\Dle{0}$, $\Dge{0}$ determines the other.
			\item An object $E$ lies in $\Dle{0}$ if and only if $\tau^{\geqslant 1} E = 0$, and in $\Dge{1}$ if and only if $\tau^{\leqslant 0} E = 0$.
			\item $\Dle{0}$ and $\Dge{0}$ are closed under extensions, finite direct sums and direct summands.
		\end{enumerate}
		The same statements hold with $0$ replaced by any $n \in \ZZ$.
	\end{lemma}
	
	\begin{proof}
		(1) The inclusion $\Dge{1} \subset (\Dle{0})^{\perp}$ is \condref{cond:T2}.
		Conversely let $E \in (\Dle{0})^{\perp}$.
		The map $\tau^{\leqslant 0} E \to E$ is then zero, since $\tau^{\leqslant 0} E \in \Dle{0}$.
		By Lemma~\ref{lem:truncation_unique}(1) with $T = \tau^{\leqslant 0} E$ the map $\Hom(\tau^{\leqslant 0} E, \tau^{\leqslant 0} E) \to \Hom(\tau^{\leqslant 0} E, E)$ is injective, and it sends the identity to zero;
		hence $\id_{\tau^{\leqslant 0} E} = 0$, i.e.\ $\tau^{\leqslant 0} E = 0$.
		The truncation triangle becomes $0 \to E \to \tau^{\geqslant 1} E \to 0$, so $E \cong \tau^{\geqslant 1} E \in \Dge{1}$ by \condref{cond:TR-5}.
		The second equality is proved dually.
		
		(2) If $E \in \Dle{0}$, then $E \xrightarrow{\id} E \to 0 \to E[1]$ is a triangle with first term in $\Dle{0}$ and third term in $\Dge{1}$, so $\tau^{\geqslant 1} E \cong 0$ by uniqueness.
		Conversely, if $\tau^{\geqslant 1} E = 0$, then $E \cong \tau^{\leqslant 0} E \in \Dle{0}$.
		The second statement is dual.
		
		(3) By (1), $\Dle{0} = {}^{\perp}(\Dge{1})$ is a left orthogonal and $\Dge{0} = (\Dle{0})^{\perp}[1] = (\Dle{-1})^{\perp}$ is a right orthogonal, so Lemma~\ref{lem:orthogonals} applies.
	\end{proof}
	
	\begin{remark}
		Because of Lemma~\ref{lem:t_orthogonal}(1) a t-structure is often specified by its aisle $\Dle{0}$ alone.
		Note also the shape of the truncation triangle in Lemma~\ref{lem:truncation_unique}: it is exactly the triangle obtained by cutting a Harder--Narasimhan filtration between two consecutive factors, with $\Dle{0}$ playing the role of ``large phases'' and $\Dge{1}$ of ``small phases'';
		this analogy will be made precise when we discuss slicings (Proposition~\ref{prop:slicing_t}); see also the notion of $t$-stability in \cite{GKR04}.
	\end{remark}
	
	\subsection{The heart}
	
	Let $(\Dle{0}, \Dge{0})$ be a t-structure on $\calD$.
	Its \emph{heart} is the subcategory
	\[
	\calA := \Dle{0} \cap \Dge{0} .
	\]
	By Lemma~\ref{lem:t_orthogonal}(3) it is closed under extensions, finite direct sums and direct summands, and it contains $0$;
	so it is an additive subcategory of $\calD$.
	For $n \in \ZZ$ we have $\calA[n] = \Dle{-n} \cap \Dge{-n}$ (the objects ``concentrated in degree $-n$''), and
	\begin{equation}
		\label{eq:heart_negative_ext}
		\Hom(X, Y[k]) = 0 \qquad \text{for all } X, Y \in \calA \text{ and } k < 0,
	\end{equation}
	because $X \in \Dle{0}$ and $Y[k] \in \Dge{-k} \subset \Dge{1}$.
	
	Throughout the rest of this subsection we fix the t-structure and its heart $\calA$, and we repeatedly use the following observation.
	
	\begin{lemma}
		\label{lem:cone_in_heart}
		Let $f \colon X \to Y$ be a morphism in $\calA$ and $X \xrightarrow{f} Y \to C \to X[1]$ a triangle.
		Then $C \in \Dle{0} \cap \Dge{-1}$, and in the truncation triangle $\tau^{\leqslant -1} C \to C \to \tau^{\geqslant 0} C \to (\tau^{\leqslant -1} C)[1]$ the objects
		\[
		K := (\tau^{\leqslant -1} C)[-1] \qquad\text{and}\qquad Q := \tau^{\geqslant 0} C
		\]
		both lie in $\calA$.
	\end{lemma}
	
	\begin{proof}
		By \condref{cond:TR-rot}, $C$ is an extension of $X[1]$ by $Y$, namely there is a triangle 
		\[
		Y \to C \to X[1] \to Y[1].
		\]
		Since $Y \in \Dle{0}$ and $X[1] \in \Dle{-1} \subset \Dle{0}$, Lemma~\ref{lem:t_orthogonal}(3) gives $C \in \Dle{0}$;
		since $Y \in \Dge{0}$ and $X[1] \in \Dge{-1}$, and $\Dge{0} \subset \Dge{-1}$, it gives $C \in \Dge{-1}$.
		
		By definition $\tau^{\leqslant -1} C \in \Dle{-1}$ and $Q = \tau^{\geqslant 0} C \in \Dge{0}$.
		The rotated truncation triangle 
		\[
		Q[-1] \to \tau^{\leqslant -1} C \to C \to Q
		\] 
		exhibits $\tau^{\leqslant -1} C$ as an extension of $C \in \Dge{-1}$ by $Q[-1] \in \Dge{1} \subset \Dge{-1}$,
		so $\tau^{\leqslant -1} C \in \Dge{-1}$ and hence 
		$K = (\tau^{\leqslant -1} C)[-1] \in \Dle{0} \cap \Dge{0} = \calA$.
		The triangle 
		\[
		C \to Q \to (\tau^{\leqslant -1} C)[1] \to C[1]
		\] 
		exhibits $Q$ as an extension of $(\tau^{\leqslant -1} C)[1] \in \Dle{-2} \subset \Dle{0}$ by $C \in \Dle{0}$, so $Q \in \Dle{0}$ and hence $Q \in \calA$.
	\end{proof}
	
	In the situation of Lemma~\ref{lem:cone_in_heart} we define morphisms $\kappa \colon K \to X$ and $\pi \colon Y \to Q$ as the composites
	\[
	\kappa \colon K = (\tau^{\leqslant -1} C)[-1] \longrightarrow C[-1] \longrightarrow X ,
	\qquad
	\pi \colon Y \longrightarrow C \longrightarrow Q ,
	\]
	where $C[-1] \to X$ is the map of the rotated triangle $C[-1] \to X \xrightarrow{f} Y \to C$.
	
	\begin{proposition}
		\label{prop:kernel_cokernel}
		In the situation of Lemma~\ref{lem:cone_in_heart}, $\kappa \colon K \to X$ is a kernel of $f$ in $\calA$ and $\pi \colon Y \to Q$ is a cokernel of $f$ in $\calA$.
	\end{proposition}
	
	\begin{proof}
		We show that for every $T \in \calA$ the sequence
		\[
		0 \to \Hom(T, K) \xrightarrow{\kappa_*} \Hom(T, X) \xrightarrow{f_*} \Hom(T, Y)
		\]
		is exact; by the Yoneda lemma this means that $\kappa$ is a kernel of $f$ in $\calA$.
		
		First, $\Hom(T, K) \to \Hom(T, C[-1])$ is bijective.
		Indeed, the triangle 
		\[
		K \to C[-1] \to Q[-1] \to K[1]
		\] 
		(the truncation triangle of $C$ shifted by $[-1]$) gives an exact sequence
		\[
		\Hom(T, Q[-2]) \to \Hom(T, K) \to \Hom(T, C[-1]) \to \Hom(T, Q[-1]) ,
		\]
		and both outer terms vanish by \condref{cond:T2}, because $T \in \Dle{0}$ while $Q[-1] \in \Dge{1}$ and $Q[-2] \in \Dge{2} \subset \Dge{1}$.
		
		Next, the rotated triangle $Y[-1] \to C[-1] \to X \xrightarrow{f} Y$ gives an exact sequence
		\[
		\Hom(T, Y[-1]) \to \Hom(T, C[-1]) \to \Hom(T, X) \xrightarrow{f_*} \Hom(T, Y) ,
		\]
		in which $\Hom(T, Y[-1]) = 0$ by \eqref{eq:heart_negative_ext}.
		Hence $\Hom(T, C[-1]) \to \Hom(T, X)$ is injective with image $\ker f_*$.
		Composing with the bijection of the previous paragraph gives the claim for $\kappa_* = (\text{this map}) \circ (\text{bijection})$.
		
		The statement about $\pi$ is proved dually, using the exact sequences
		\begin{align*}
			&\Hom(K[2], T) \to \Hom(Q, T) \to \Hom(C, T) \to \Hom(K[1], T), \\
			&\Hom(X[1], T) \to \Hom(C, T) \to \Hom(Y, T) \xrightarrow{f^*} \Hom(X, T),
		\end{align*}
		in which $\Hom(K[1], T) = \Hom(K[2], T) = 0$ by \condref{cond:T2} (as $K[1] \in \Dle{-1}$, $K[2] \in \Dle{-2}$ and $T \in \Dge{0}$, so $\Hom(K[j], T) = \Hom(K[j-1], T[-1])$ with $T[-1] \in \Dge{1}$) and $\Hom(X[1], T) = 0$ by \eqref{eq:heart_negative_ext}.
	\end{proof}
	
	\begin{lemma}
		\label{lem:triangles_vs_ses}
		Let $X', X, X'' \in \calA$.
		\begin{enumerate}
			\item If $X' \xrightarrow{u} X \xrightarrow{v} X'' \to X'[1]$ is a triangle, then $0 \to X' \xrightarrow{u} X \xrightarrow{v} X'' \to 0$ is a short exact sequence in $\calA$, i.e.\ $u$ is a kernel of $v$ and $v$ is a cokernel of $u$.
			\item Conversely, if $0 \to X' \xrightarrow{u} X \xrightarrow{v} X'' \to 0$ is a short exact sequence in $\calA$ (meaning that $u$ is a kernel of $v$ and $v$ is a cokernel of $u$), then there is a unique morphism $w \colon X'' \to X'[1]$ such that $X' \xrightarrow{u} X \xrightarrow{v} X'' \xrightarrow{w} X'[1]$ is a triangle.
		\end{enumerate}
	\end{lemma}
	
	\begin{proof}
		(1) Apply Lemma~\ref{lem:cone_in_heart} and Proposition~\ref{prop:kernel_cokernel} to $f = u$ with cone $C = X''$.
		Since $X'' \in \Dge{0}$, Lemma~\ref{lem:t_orthogonal}(2) (shifted) gives $\tau^{\leqslant -1} X'' = 0$, so $K = 0$ and $Q = X''$ with $\pi = v$;
		thus $u$ is a monomorphism (its kernel is zero) and $v$ is a cokernel of $u$.
		Apply them next to $f = v$, whose cone is $X'[1]$ by \condref{cond:TR-rot}:
		since $X'[1] \in \Dle{-1}$ we get $\tau^{\leqslant -1}(X'[1]) = X'[1]$ and $\tau^{\geqslant 0}(X'[1]) = 0$, so $K = X'$ and $Q = 0$;
		thus $v$ is an epimorphism and its kernel is $\kappa \colon X' \to X$, which is $u$ up to sign (the sign comes from the rotation in \condref{cond:TR-rot} and does not affect the kernel).
		
		(2) Complete $u$ to a triangle 
		\[
		X' \xrightarrow{u} X \xrightarrow{v'} C \to X'[1]
		\] 
		and form $K$, $Q$, $\kappa$, $\pi$ as in Lemma~\ref{lem:cone_in_heart}.
		By Proposition~\ref{prop:kernel_cokernel}, $\kappa \colon K \to X'$ is a kernel of $u$; as $u$ is a monomorphism, $K = 0$, i.e.\ $\tau^{\leqslant -1} C = 0$, and the map $C \to Q = \tau^{\geqslant 0} C$ is an isomorphism by \condref{cond:TR-5}.
		In particular $C \in \calA$, and $\pi \colon X \to Q$ identifies with $v' \colon X \to C$, which is therefore a cokernel of $u$.
		Two cokernels of $u$ are related by a unique isomorphism $\theta \colon C \to X''$ with $\theta v' = v$.
		Transporting the triangle along $\theta$ we obtain a triangle 
		\[
		X' \xrightarrow{u} X \xrightarrow{v} X'' \xrightarrow{w} X'[1].
		\]
		As for the uniqueness of $w$, if $w'$ also completes $u, v$ to a triangle, then by the axioms there is a morphism of triangles $(\id, \id, \gamma)$ between the two, with $\gamma \in \End(X'')$ satisfying $\gamma v = v$, hence $\gamma = \id$ since $v$ is an epimorphism; the compatibility with the third maps then reads $w' = w$.
	\end{proof}
	
	\begin{theorem}[{\cite[Th\'eor\`eme~1.3.6]{BBD82}}]
		\label{thm:heart_abelian}
		The heart $\calA$ of a t-structure is an abelian category.
		Its short exact sequences are exactly the triangles of $\calD$ with all three terms in $\calA$, in the sense of Lemma~\ref{lem:triangles_vs_ses}.
	\end{theorem}
	
	\begin{proof}
		$\calA$ is additive, and every morphism has a kernel and a cokernel by Proposition~\ref{prop:kernel_cokernel}.
		It remains to show that for every $f \colon X \to Y$ in $\calA$ the canonical morphism from the coimage $\operatorname{coker}(\ker f \to X)$ to the image $\ker(Y \to \operatorname{coker} f)$ is an isomorphism.
		
		Let $X \xrightarrow{f} Y \xrightarrow{g} C \to X[1]$ be a triangle, and let $K$, $Q$, $\kappa$, $\pi$ be as above, with $q \colon C \to Q$ the truncation map, so that $\pi = q g$.
		Choose a triangle 
		\[
		I \xrightarrow{j} Y \xrightarrow{\pi} Q \to I[1].
		\]
		We apply \condref{cond:TR-oct} to the pair $Y \xrightarrow{g} C \xrightarrow{q} Q$.
		The cones are $\Cone(g) = X[1]$, $\Cone(q g) = \Cone(\pi) = I[1]$ and $\Cone(q) = (\tau^{\leqslant -1} C)[1] = K[2]$, so there is a triangle
		\[
		X[1] \to I[1] \to K[2] \to X[2] ,
		\]
		which after rotating and shifting becomes a triangle 
		\[
		K \to X \xrightarrow{e} I \to K[1];
		\]
		the compatibilities in the octahedral axiom say that $j e = f$ and that $K \to X$ is the map $\kappa$ (up to sign).
		
		We claim that $I \in \calA$.
		From the triangle 
		\[
		Q[-1] \to I \xrightarrow{j} Y \to Q
		\] 
		and $Q[-1] \in \Dge{1} \subset \Dge{0}$, $Y \in \Dge{0}$ we get $I \in \Dge{0}$;
		from the triangle $X \xrightarrow{e} I \to K[1] \to X[1]$ and $X \in \Dle{0}$, $K[1] \in \Dle{-1} \subset \Dle{0}$ we get $I \in \Dle{0}$.
		
		Now Lemma~\ref{lem:triangles_vs_ses}(1) applies to both triangles 
		\[
		K \to X \xrightarrow{e} I \to K[1]
		\] and $I \xrightarrow{j} Y \xrightarrow{\pi} Q \to I[1]$, all of whose terms lie in $\calA$:
		\[
		0 \to K \xrightarrow{\pm\kappa} X \xrightarrow{e} I \to 0
		\qquad\text{and}\qquad
		0 \to I \xrightarrow{j} Y \xrightarrow{\pi} Q \to 0
		\]
		are short exact sequences in $\calA$.
		The first says that $e$ is a cokernel of $\kappa = \ker f$, i.e.\ $e \colon X \to I$ is the coimage of $f$;
		the second says that $j$ is a kernel of $\pi = \operatorname{coker} f$, i.e.\ $j \colon I \to Y$ is the image of $f$.
		Since $f = j e$, the canonical map $\operatorname{coim} f \to \operatorname{im} f$ is the identity of $I$, an isomorphism.
		Hence $\calA$ is abelian.
		The last assertion is Lemma~\ref{lem:triangles_vs_ses}.
	\end{proof}
	
	\begin{remark}
		The proof shows more precisely how kernels, cokernels and images of a morphism $f \colon X \to Y$ in $\calA$ are read off from its cone $C$.
		The cone lives in two consecutive ``degrees'' $-1$ and $0$, and
		\[
		\ker f = (\tau^{\leqslant -1} C)[-1], \qquad \operatorname{coker} f = \tau^{\geqslant 0} C .
		\]
		In particular $f$ is a monomorphism in $\calA$ if and only if $C \in \calA$, and an epimorphism if and only if $C \in \calA[1]$.
		Note also that a monomorphism in $\calA$ need not be a monomorphism in $\calD$, and the heart is in general not closed under cones, since $C \notin \calA$ as soon as $f$ has nonzero kernel.
	\end{remark}


	\subsection{Cohomology functors}
	
	We continue with a fixed t-structure $(\Dle{0}, \Dge{0})$ on $\calD$ with heart $\calA$.
	
	\begin{lemma}[{\cite[1.3.4, 1.3.5]{BBD82}}]
		\label{lem:truncations_compose}
		Let $a \leqslant b$ be integers and $E \in \calD$.
		\begin{enumerate}
			\item The natural morphisms induce isomorphisms $\tau^{\leqslant a} \tau^{\leqslant b} E \cong \tau^{\leqslant a} E$ and $\tau^{\geqslant b} \tau^{\geqslant a} E \cong \tau^{\geqslant b} E$.
			\item There is a unique morphism $\tau^{\geqslant a} \tau^{\leqslant b} E \to \tau^{\leqslant b} \tau^{\geqslant a} E$ compatible with the natural maps from $\tau^{\leqslant b} E$ and to $\tau^{\geqslant a} E$, and it is an isomorphism.
		\end{enumerate}
	\end{lemma}
	
	\begin{proof}
		(1) Since $\Dle{a} \subset \Dle{b}$, for every $T \in \Dle{a}$ the adjunction bijections give
		\[
		\Hom(T, \tau^{\leqslant a} \tau^{\leqslant b} E) \cong \Hom(T, \tau^{\leqslant b} E) \cong \Hom(T, E) \cong \Hom(T, \tau^{\leqslant a} E) ,
		\]
		all induced by the natural maps to $E$;
		by the Yoneda lemma the natural map 
		\[
		\tau^{\leqslant a} \tau^{\leqslant b} E \to \tau^{\leqslant a} E
		\] 
		is an isomorphism.
		The second isomorphism is dual.
		
		(2) Put $X := \tau^{\leqslant b} E$.
		The composite $\tau^{\leqslant a-1} E \to E$ factors uniquely through $X$ 
		(adjunction, as $\tau^{\leqslant a-1} E \in \Dle{a-1} \subset \Dle{b}$), 
		and by (1) the resulting map $\tau^{\leqslant a-1} E \to X$ is the truncation 
		$\tau^{\leqslant a-1} X \to X$.
		Apply \condref{cond:TR-oct} to the pair $\tau^{\leqslant a-1} E \to X \to E$.
		The cones are $\tau^{\geqslant a} X$, $\tau^{\geqslant a} E$ and $\tau^{\geqslant b+1} E$, so there is a triangle
		\[
		\tau^{\geqslant a} \tau^{\leqslant b} E \longrightarrow \tau^{\geqslant a} E \longrightarrow \tau^{\geqslant b+1} E \longrightarrow (\tau^{\geqslant a} \tau^{\leqslant b} E)[1] ,
		\]
		compatible with the natural maps.
		Its first term lies in $\Dle{b}$ (it is an extension of $(\tau^{\leqslant a-1} E)[1] \in \Dle{b}$ by $X \in \Dle{b}$, cf.\ the rotated triangle $X \to \tau^{\geqslant a} X \to (\tau^{\leqslant a-1} E)[1]$) and its third term lies in $\Dge{b+1}$.
		By Lemma~\ref{lem:truncation_unique}(2) this triangle is the truncation triangle of $\tau^{\geqslant a} E$ at level $b$, i.e.\ $\tau^{\geqslant a} \tau^{\leqslant b} E \cong \tau^{\leqslant b} \tau^{\geqslant a} E$ compatibly with the maps to $\tau^{\geqslant a} E$;
		compatibility with the maps from $\tau^{\leqslant b} E$ follows from the adjunction bijection 
		\[
		\Hom(\tau^{\leqslant b} E, \tau^{\leqslant b} \tau^{\geqslant a} E) \cong \Hom(\tau^{\leqslant b} E, \tau^{\geqslant a} E),
		\] 
		since both composites $\tau^{\leqslant b} E \to \tau^{\leqslant b} \tau^{\geqslant a} E \to \tau^{\geqslant a} E$ equal the composite $\tau^{\leqslant b} E \to E \to \tau^{\geqslant a} E$.
		Finally, a morphism $\tau^{\geqslant a} \tau^{\leqslant b} E \to \tau^{\leqslant b} \tau^{\geqslant a} E$ compatible with the maps to $\tau^{\geqslant a} E$ is unique, because $\tau^{\geqslant a} \tau^{\leqslant b} E \in \Dle{b}$ and $\tau^{\leqslant b}$ is right adjoint to the inclusion of $\Dle{b}$, so that 
		\[
		\Hom(\tau^{\geqslant a} \tau^{\leqslant b} E, \tau^{\leqslant b} \tau^{\geqslant a} E) \cong \Hom(\tau^{\geqslant a} \tau^{\leqslant b} E, \tau^{\geqslant a} E).
		\]
	\end{proof}
	
	For $a \leqslant b$ we write $\Dint{a}{b} := \Dle{b} \cap \Dge{a}$ and $\tau^{[a, b]} := \tau^{\geqslant a} \tau^{\leqslant b} \cong \tau^{\leqslant b} \tau^{\geqslant a}$; by the proof of Lemma~\ref{lem:truncations_compose}(2) this functor takes values in $\Dint{a}{b}$.
	In particular we define the \emph{cohomology functors}
	\[
	H^0 := \tau^{[0, 0]} = \tau^{\geqslant 0} \tau^{\leqslant 0} \colon \calD \longrightarrow \calA ,
	\qquad
	H^i(E) := H^0(E[i]) \qquad (i \in \ZZ) .
	\]
	Unwinding the definitions, $H^i(E)[-i] = \tau^{[i, i]} E = \tau^{\geqslant i} \tau^{\leqslant i} E$.
	
	\begin{lemma}
		\label{lem:H0_basic}
		\begin{enumerate}
			\item For $E \in \Dle{0}$ one has $H^0(E) = \tau^{\geqslant 0} E$, and for $E \in \Dge{0}$ one has $H^0(E) = \tau^{\leqslant 0} E$.
			\item For $A \in \calA$ one has $H^0(A) = A$ and $H^i(A) = 0$ for $i \neq 0$.
			\item For every $E$ there is a triangle $\tau^{\leqslant i-1} E \to \tau^{\leqslant i} E \to H^i(E)[-i] \to (\tau^{\leqslant i-1} E)[1]$, natural in $E$.
			\item For $E \in \Dle{0}$ and $T \in \calA$ the natural map $\Hom(H^0 E, T) \to \Hom(E, T)$ is bijective; for $E \in \Dge{0}$ and $T \in \calA$ the natural map $\Hom(T, H^0 E) \to \Hom(T, E)$ is bijective.
		\end{enumerate}
	\end{lemma}
	
	\begin{proof}
		(1) If $E \in \Dle{0}$ then $\tau^{\leqslant 0} E = E$ by Lemma~\ref{lem:t_orthogonal}(2), and dually.
		(2) follows from (1).
		For $A \in \Dle{0} \cap \Dge{0}$ we get $H^0 A = \tau^{\geqslant 0} A = A$;
		for $i > 0$ we have $A[i] \in \Dle{-i} \subset \Dle{-1}$, so $\tau^{\geqslant 0}(A[i]) = 0$ 
		and 
		\[
		H^0(A[i]) = \tau^{\geqslant 0} \tau^{\leqslant 0} (A[i]) = \tau^{\geqslant 0}(A[i]) = 0;
		\]
		for $i < 0$ we have $A[i] \in \Dge{-i} \subset \Dge{1}$, so $\tau^{\leqslant 0}(A[i]) = 0$.
		(3) By Lemma~\ref{lem:truncations_compose}(1), 
		$\tau^{\leqslant i-1}(\tau^{\leqslant i} E) = \tau^{\leqslant i-1} E$, 
		so the truncation triangle of $\tau^{\leqslant i} E$ at level $i-1$ reads 
		\[
		\tau^{\leqslant i-1} E \to \tau^{\leqslant i} E \to \tau^{\geqslant i} \tau^{\leqslant i} E = H^i(E)[-i].
		\]
		(4) For $E \in \Dle{0}$, $H^0 E = \tau^{\geqslant 0} E$ by (1), and $\tau^{\geqslant 0}$ is left adjoint to the inclusion of $\Dge{0} \ni T$; dually for $E \in \Dge{0}$.
	\end{proof}
	
	\begin{theorem}[{\cite[Th\'eor\`eme~1.3.6]{BBD82}}]
		\label{thm:H0_cohomological}
		The functor $H^0 \colon \calD \to \calA$ is cohomological, i.e.\ for every triangle $X \to Y \to Z \to X[1]$ the sequence
		\[
		H^0 X \to H^0 Y \to H^0 Z
		\]
		is exact in $\calA$.
		Consequently there is a long exact sequence in $\calA$
		\[
		\cdots \to H^{i-1} Z \to H^i X \to H^i Y \to H^i Z \to H^{i+1} X \to \cdots
		\]
	\end{theorem}
	
	\begin{proof}
		The long exact sequence follows from the first statement applied to the rotated triangles \condref{cond:TR-rot}, since $H^i = H^0 \circ [i]$.
		We prove the first statement in three steps, following \cite{BBD82}.
		
		\step{$X, Y, Z \in \Dle{0}$}
		We show that $H^0 X \to H^0 Y \to H^0 Z \to 0$ is exact.
		In an abelian category a sequence $A \to B \to C \to 0$ is exact if and only if
		\[
		0 \to \Hom(C, T) \to \Hom(B, T) \to \Hom(A, T)
		\]
		is exact for every object $T$ (by the Yoneda lemma).
		For $T \in \calA$ the long exact sequence \condref{cond:TR-coh} gives
		\[
		\Hom(X[1], T) \to \Hom(Z, T) \to \Hom(Y, T) \to \Hom(X, T) ,
		\]
		and $\Hom(X[1], T) = 0$ by \condref{cond:T2}, as $X[1] \in \Dle{-1}$ and $T \in \Dge{0}$.
		By Lemma~\ref{lem:H0_basic}(4) this sequence identifies with
		\[
		0 \to \Hom(H^0 Z, T) \to \Hom(H^0 Y, T) \to \Hom(H^0 X, T) ,
		\]
		which is therefore exact.
		
		\step{$X \in \Dle{0}$, or dually $Z \in \Dge{0}$}
		We show that $H^0 X \to H^0 Y \to H^0 Z \to 0$ is exact when $X \in \Dle{0}$.
		By adjunction the map $X \to Y$ factors as $X \to \tau^{\leqslant 0} Y \to Y$.
		Apply \condref{cond:TR-oct} to this pair.
		With $W := \Cone(X \to \tau^{\leqslant 0} Y)$ there is a triangle
		\[
		W \longrightarrow Z \longrightarrow \tau^{\geqslant 1} Y \longrightarrow W[1] ,
		\]
		compatible with the maps $\tau^{\leqslant 0} Y \to W$, $\tau^{\leqslant 0} Y \to Y \to Z$ and $Y \to \tau^{\geqslant 1} Y$.
		Here $W \in \Dle{0}$ (an extension of $X[1]$ by $\tau^{\leqslant 0} Y$) and $\tau^{\geqslant 1} Y \in \Dge{1}$, so by Lemma~\ref{lem:truncation_unique}(2) this is the truncation triangle of $Z$.
		Hence $W \cong \tau^{\leqslant 0} Z$, and under this identification $\tau^{\leqslant 0} Y \to W$ is $\tau^{\leqslant 0}$ of the map $Y \to Z$ (both are compatible with the maps to $Z$, and such a map is unique by adjunction).
		Thus 
		\[
		X \to \tau^{\leqslant 0} Y \to \tau^{\leqslant 0} Z \to X[1]
		\] 
		is a triangle with all terms in $\Dle{0}$, and Step~1 gives an exact sequence 
		\[
		H^0 X \to H^0 \tau^{\leqslant 0} Y \to H^0 \tau^{\leqslant 0} Z \to 0.
		\]
		Finally $H^0 \tau^{\leqslant 0} = H^0$ by Lemma~\ref{lem:truncations_compose}(1), compatibly with the maps, which gives the claim.
		The dual statement, exactness of 
		\[
		0 \to H^0 X \to H^0 Y \to H^0 Z
		\] 
		when $Z \in \Dge{0}$, is proved in the same way (or by passing to the opposite category, which carries the t-structure $(\Dge{0}, \Dle{0})$).
		
		\step{the general case}
		Apply \condref{cond:TR-oct} to the pair $\tau^{\leqslant 0} X \to X \to Y$.
		With $U := \Cone(\tau^{\leqslant 0} X \to Y)$ there is a triangle
		\[
		\tau^{\geqslant 1} X \longrightarrow U \longrightarrow Z \longrightarrow (\tau^{\geqslant 1} X)[1] ,
		\]
		compatible with the maps $Y \to U$ and $Y \to Z$.
		Step~2 applied to the triangle 
		\[
		\tau^{\leqslant 0} X \to Y \to U \to (\tau^{\leqslant 0} X)[1]
		\] 
		gives an exact sequence 
		\[
		H^0 \tau^{\leqslant 0} X \to H^0 Y \to H^0 U \to 0,
		\] 
		and $H^0 \tau^{\leqslant 0} X = H^0 X$.
		The dual half of Step~2 applied to the rotated triangle 
		\[
		U \to Z \to (\tau^{\geqslant 1} X)[1] \to U[1],
		\] 
		whose third term lies in $\Dge{0}$, shows that $H^0 U \to H^0 Z$ is a monomorphism.
		Since $H^0 Y \to H^0 Z$ is the composite $H^0 Y \to H^0 U \to H^0 Z$, its kernel equals the kernel of $H^0 Y \to H^0 U$, which is the image of $H^0 X \to H^0 Y$.
	\end{proof}
	
	\subsection{Extension closures}
	
	Recall from the conventions at the beginning of this section that a subcategory $\mathcal{C} \subset \calD$ is closed under extensions if $X, Z \in \mathcal{C}$ implies $Y \in \mathcal{C}$ for every triangle $X \to Y \to Z \to X[1]$, and that the extension closure $\langle \mathcal{S} \rangle$ of a class of objects $\mathcal{S}$ is the smallest subcategory of $\calD$ which contains $\mathcal{S}$ and $0$ and is closed under extensions.
	In this subsection we describe $\langle \mathcal{S} \rangle$ concretely.
	
	A \emph{filtration} of an object $E$ \emph{with factors in} $\mathcal{S}$ is a finite sequence of morphisms
	\[
	0 = E_0 \to E_1 \to \ldots \to E_{n-1} \to E_n = E
	\]
	such that every cone $A_i := \Cone(E_{i-1} \to E_i)$, $i = 1, \ldots, n$, lies in $\mathcal{S}$.
	The objects $A_i$ are called the \emph{factors} and $n$ the \emph{length} of the filtration;
	the zero object has the filtration of length $0$.
	We often write $E_i / E_{i-1}$ for $A_i$, although in a triangulated category this is only a notation for a cone.
	
	\begin{lemma}[refining a filtration]
		\label{lem:refinement}
		Let $X' \to X \to C \to X'[1]$ be a triangle and let $0 = C_0 \to C_1 \to \ldots \to C_m = C$ be a filtration of $C$ with factors $D_l = \Cone(C_{l-1} \to C_l)$.
		Then there is a sequence of morphisms
		\[
		X' = Y_0 \to Y_1 \to \ldots \to Y_m = X
		\]
		with $\Cone(Y_{l-1} \to Y_l) \cong D_l$ for $l = 1, \ldots, m$.
	\end{lemma}
	
	\begin{proof}
		For $0 \leqslant l \leqslant m$ choose a cone $C / C_l$ of $C_l \to C$, with $C / C_0 = C$ and $C / C_m = 0$, and let $Y_l$ be a fibre of the composite $X \to C \to C / C_l$, so that
		\[
		Y_l \longrightarrow X \longrightarrow C / C_l \longrightarrow Y_l[1]
		\]
		is a triangle.
		Then $Y_0 = X'$ and $Y_m = X$.
		Applying \condref{cond:TR-oct} to the pair $C_{l-1} \to C_l \to C$, whose cones are $D_l$, $C / C_{l-1}$ and $C / C_l$, we obtain a triangle
		\[
		D_l \longrightarrow C / C_{l-1} \longrightarrow C / C_l \longrightarrow D_l[1]
		\]
		in which the second map is compatible with the maps from $C$.
		Applying \condref{cond:TR-oct} to the pair $X \to C / C_{l-1} \to C / C_l$, whose cones are $Y_{l-1}[1]$, $Y_l[1]$ and $D_l[1]$, we obtain a triangle
		\[
		Y_{l-1}[1] \longrightarrow Y_l[1] \longrightarrow D_l[1] \longrightarrow Y_{l-1}[2] ,
		\]
		and shifting by $[-1]$ gives the required triangle $Y_{l-1} \to Y_l \to D_l \to Y_{l-1}[1]$.
	\end{proof}
	
	\begin{lemma}
		\label{lem:devissage}
		Let $\mathcal{S}, \mathcal{T}$ be classes of objects of $\calD$ with $\Hom(S, T) = 0$ for all $S \in \mathcal{S}$, $T \in \mathcal{T}$.
		Then $\Hom(X, Y) = 0$ for all $X \in \langle \mathcal{S} \rangle$ and $Y \in \langle \mathcal{T} \rangle$.
	\end{lemma}
	
	\begin{proof}
		For fixed $T \in \mathcal{T}$ the subcategory ${}^{\perp}\{T\}$ contains $\mathcal{S}$, and by Lemma~\ref{lem:orthogonals} it contains $0$ and is closed under extensions, hence it contains $\langle \mathcal{S} \rangle$.
		Then for fixed $X \in \langle \mathcal{S} \rangle$ the subcategory $\{X\}^{\perp}$ contains $\mathcal{T}$ and, again by Lemma~\ref{lem:orthogonals}, contains $\langle \mathcal{T} \rangle$.
	\end{proof}
	
	\begin{lemma}
		\label{lem:cone_of_filtration}
		Let $0 = E_0 \to E_1 \to \ldots \to E_n = E$ be a filtration with factors $A_i = \Cone(E_{i-1} \to E_i)$, and for $0 \leqslant m \leqslant n$ let $Q_m$ be a cone of the composite $E_m \to E$.
		Then $Q_n = 0$, $Q_0 = E$, and for $0 \leqslant m < n$ there is a triangle
		\[
		A_{m+1} \longrightarrow Q_m \longrightarrow Q_{m+1} \longrightarrow A_{m+1}[1] .
		\]
		In particular $E_m \in \langle A_1, \ldots, A_m \rangle$ and $Q_m \in \langle A_{m+1}, \ldots, A_n \rangle$.
	\end{lemma}
	
	\begin{proof}
		$Q_n$ is a cone of the identity and $Q_0$ a cone of $0 \to E$, whence the first two claims.
		For $m < n$ apply \condref{cond:TR-oct} to the pair $E_m \to E_{m+1} \to E$.
		Its cones are $A_{m+1}$, $Q_m$ and $Q_{m+1}$, which gives the triangle.
		The membership of $E_m$ follows by induction on $m$ from the triangles $E_{m-1} \to E_m \to A_m$, that of $Q_m$ by descending induction on $m$ from the triangles just constructed.
	\end{proof}
	
	\begin{proposition}[structure of the extension closure]
		\label{prop:extension_closure}
		Let $\mathcal{S}$ be a class of objects of $\calD$.
		Then $\langle \mathcal{S} \rangle$ is the class of all objects admitting a filtration with factors in $\mathcal{S}$.
		In particular $\langle \mathcal{S} \rangle$ is closed under finite direct sums, and if $\mathcal{S}[1] = \mathcal{S}$ then $\langle \mathcal{S} \rangle[1] = \langle \mathcal{S} \rangle$.
	\end{proposition}
	
	\begin{proof}
		Let $\mathcal{F}$ be the class of objects admitting a filtration with factors in $\mathcal{S}$.
		
		$\mathcal{F} \subset \langle \mathcal{S} \rangle$.
		If $0 = E_0 \to \ldots \to E_n = E$ is a filtration with factors in $\mathcal{S}$, then $E_0 = 0 \in \langle \mathcal{S} \rangle$, and $E_i$ is an extension of $A_i \in \mathcal{S}$ by $E_{i-1}$, so $E_i \in \langle \mathcal{S} \rangle$ by induction on $i$.
		
		$\langle \mathcal{S} \rangle \subset \mathcal{F}$.
		It suffices to show that $\mathcal{F}$ contains $\mathcal{S}$ and $0$ and is closed under isomorphisms and extensions.
		The first two are clear (filtrations of length $1$ and $0$), and if $E \in \mathcal{F}$ and $E \cong E'$, composing the last map of a filtration of $E$ with the isomorphism gives a filtration of $E'$ with the same factors.
		Let $X \to Y \to Z \to X[1]$ be a triangle with $X, Z \in \mathcal{F}$, and choose filtrations $0 = X_0 \to \ldots \to X_p = X$ and $0 = Z_0 \to \ldots \to Z_q = Z$ with factors in $\mathcal{S}$.
		By Lemma~\ref{lem:refinement} applied to the triangle $X \to Y \to Z \to X[1]$ and the filtration of $Z$ there is a sequence $X = Y_0 \to \ldots \to Y_q = Y$ whose cones are the factors of the filtration of $Z$.
		Concatenating,
		\[
		0 = X_0 \to \ldots \to X_p = X = Y_0 \to Y_1 \to \ldots \to Y_q = Y
		\]
		is a filtration of $Y$ with factors in $\mathcal{S}$, so $Y \in \mathcal{F}$.
		
		The last assertions follow from the split triangle $X \to X \oplus Z \to Z \to X[1]$ and from the fact that the shift of a filtration is a filtration with shifted factors.
	\end{proof}
	
	\begin{remark}
		Lemma~\ref{lem:cone_of_filtration} and Proposition~\ref{prop:extension_closure} together say that the ``quotient'' $Q_m = E / E_m$ of a filtered object by a step of its filtration is filtered by the remaining factors, exactly as for filtrations of modules.
		We shall use this constantly, for Harder--Narasimhan filtrations in triangulated categories as well as for the cohomology filtration of the next subsection.
	\end{remark}
	
	\subsection{Bounded t-structures}
	
	A t-structure is called \emph{bounded} if
	\[
	\calD = \bigcup_{a \leqslant b} \Dint{a}{b} ,
	\]
	i.e.\ every object lies in $\Dle{b} \cap \Dge{a}$ for some integers $a \leqslant b$.
	It is called \emph{non-degenerate} if $\bigcap_{n} \Dle{n} = 0 = \bigcap_{n} \Dge{n}$.
	A bounded t-structure is non-degenerate.
	Indeed, if $E \in \Dle{n}$ for all $n$ and $E \in \Dge{a}$, then $E \in \Dle{a-1} \cap \Dge{a}$, so $E \in \Dle{a-1} = {}^{\perp}\Dge{a}$ gives $\Hom(E, E) = 0$ and $E = 0$.
	Dually, if $E \in \Dge{n}$ for all $n$ and $E \in \Dle{b}$, then $E \in \Dle{b} \cap \Dge{b+1}$ and again $\Hom(E, E) = 0$.
	
	\begin{lemma}[cohomology filtration]
		\label{lem:cohomology_filtration}
		Let the t-structure be bounded, and let $E \in \Dint{a}{b}$.
		\begin{enumerate}
			\item $H^i(E) = 0$ for $i \notin [a, b]$, $\tau^{\leqslant a-1} E = 0$ and $\tau^{\leqslant b} E = E$, and the triangles of Lemma~\ref{lem:H0_basic}(3) form a filtration
			\[
			0 = \tau^{\leqslant a-1} E \to \tau^{\leqslant a} E \to \tau^{\leqslant a+1} E \to \ldots \to \tau^{\leqslant b} E = E
			\]
			with $\Cone(\tau^{\leqslant i-1} E \to \tau^{\leqslant i} E) = H^i(E)[-i]$, $a \leqslant i \leqslant b$.
			\item $E \in \Dle{n}$ if and only if $H^i(E) = 0$ for all $i > n$; $E \in \Dge{n}$ if and only if $H^i(E) = 0$ for all $i < n$; $E = 0$ if and only if $H^i(E) = 0$ for all $i$.
			\item $\Dle{0} = \langle \calA[k] : k \geqslant 0 \rangle$ and $\Dge{0} = \langle \calA[k] : k \leqslant 0 \rangle$.
			In particular a bounded t-structure is determined by its heart.
		\end{enumerate}
	\end{lemma}
	
	\begin{proof}
		(1) By Lemma~\ref{lem:t_orthogonal}(2) (shifted), $E \in \Dge{a}$ gives $\tau^{\leqslant a-1} E = 0$ and $E \in \Dle{b}$ gives $\tau^{\leqslant b} E = E$.
		For $i < a$ we get $\tau^{\leqslant i} E = \tau^{\leqslant i} \tau^{\leqslant a-1} E = 0$ by Lemma~\ref{lem:truncations_compose}(1), hence $H^i(E)[-i] = \tau^{\geqslant i} \tau^{\leqslant i} E = 0$;
		for $i > b$ the map $\tau^{\leqslant i-1} E \to \tau^{\leqslant i} E$ is the identity of $E$ (both sides equal $E$ since $\Dle{b} \subset \Dle{i-1}$), so its cone $H^i(E)[-i]$ vanishes.
		The filtration statement is Lemma~\ref{lem:H0_basic}(3).
		
		(2) If $E \in \Dle{n}$, then $E \in \Dint{a}{n}$ for some $a$, and (1) gives $H^i(E) = 0$ for $i > n$.
		Conversely, if $H^i(E) = 0$ for $i > n$, then in the filtration of (1) all maps $\tau^{\leqslant i-1} E \to \tau^{\leqslant i} E$ with $i > n$ have zero cone, hence are isomorphisms by \condref{cond:TR-5}, so $E = \tau^{\leqslant b} E \cong \tau^{\leqslant n} E \in \Dle{n}$.
		The second statement is dual (using the filtration by the $\tau^{\geqslant i} E$), and the third follows from the first two and non-degeneracy.
		Directly, if all $H^i(E)$ vanish, the filtration in (1) shows $E \cong \tau^{\leqslant a-1} E = 0$.
		
		(3) $\Dle{0}$ is closed under extensions and contains $\calA[k] = \Dint{-k}{-k}$ for $k \geqslant 0$, so it contains $\langle \calA[k] : k \geqslant 0 \rangle$.
		Conversely, for $E \in \Dle{0}$ the filtration of (1) has factors $H^i(E)[-i] \in \calA[-i]$ with $i \leqslant 0$, so $E \in \langle \calA[k] : k \geqslant 0 \rangle$ by Proposition~\ref{prop:extension_closure}.
		Dually for $\Dge{0}$.
		The last assertion follows since $\Dle{0}$ and $\Dge{0}$ are expressed in terms of $\calA$.
	\end{proof}
	
	\subsection{Hearts of bounded t-structures}
	
	The following characterization, due to Bridgeland, describes hearts of bounded t-structures without mentioning the t-structure.
	It is the model for the notion of a slicing.
	
	\begin{proposition}[{\cite[Lemma~3.2]{Bri07}}, {\cite[Def.~5.1]{MS17}}]
		\label{prop:bounded_heart}
		Let $\calA \subset \calD$ be a full additive subcategory.
		Then $\calA$ is the heart of a bounded t-structure on $\calD$ if and only if
		\begin{itemize}
			\item[] \condlabel{BH1}{cond:BH1} $\Hom(A, B[k]) = 0$ for all $A, B \in \calA$ and all integers $k < 0$;
			\item[] \condlabel{BH2}{cond:BH2} every nonzero object $E \in \calD$ admits a finite sequence of morphisms
			\[
			0 = E_0 \to E_1 \to \ldots \to E_{n-1} \to E_n = E
			\]
			whose cones $A_i := \Cone(E_{i-1} \to E_i)$ are of the form $A_i = B_i[k_i]$ with $0 \neq B_i \in \calA$ and $k_1 > k_2 > \ldots > k_n$.
		\end{itemize}
		In this case the t-structure is $(\langle \calA[k] : k \geqslant 0 \rangle, \langle \calA[k] : k \leqslant 0 \rangle)$, and the sequence in \condref{cond:BH2} is the cohomology filtration of Lemma~\ref{lem:cohomology_filtration}, with $B_i = H^{-k_i}(E)$.
	\end{proposition}
	
	\begin{proof}
		Suppose $\calA$ is the heart of a bounded t-structure.
		Then \condref{cond:BH1} is \eqref{eq:heart_negative_ext}.
		For \condref{cond:BH2}, let $E \neq 0$, say $E \in \Dint{a}{b}$, and take the filtration of Lemma~\ref{lem:cohomology_filtration}(1), discarding the steps with $H^i(E) = 0$ (their maps are isomorphisms);
		its cones are $H^i(E)[-i]$ with $i$ increasing, i.e.\ $B_i[k_i]$ with $k_i = -i$ strictly decreasing.
		The description of the t-structure is Lemma~\ref{lem:cohomology_filtration}(3).
		
		Conversely, let $\calA$ satisfy \condref{cond:BH1} and \condref{cond:BH2}, and put
		\[
		\Dle{0} := \langle \calA[k] : k \geqslant 0 \rangle , \qquad \Dge{0} := \langle \calA[k] : k \leqslant 0 \rangle .
		\]
		Then $\Dle{n} = \langle \calA[k] : k \geqslant -n \rangle$ and $\Dge{n} = \langle \calA[k] : k \leqslant -n \rangle$, so \condref{cond:T1} is clear.
		
		\step{\condref{cond:T2}}
		For $A, B \in \calA$, $k \geqslant 0$ and $l \leqslant -1$ we have $\Hom(A[k], B[l]) = \Hom(A, B[l - k]) = 0$ by \condref{cond:BH1}, since $l - k < 0$.
		Lemma~\ref{lem:devissage} gives $\Hom(\Dle{0}, \Dge{1}) = 0$.
		
		\step{\condref{cond:T3}}
		Let $E \neq 0$ (for $E = 0$ take the zero triangle) and take a sequence as in \condref{cond:BH2}.
		Let $m$ be the number of indices $i$ with $k_i \geqslant 0$, so that $k_1 > \ldots > k_m \geqslant 0 > k_{m+1} > \ldots > k_n$.
		By Lemma~\ref{lem:cone_of_filtration}, $E_m \in \langle A_1, \ldots, A_m \rangle \subset \Dle{0}$ and a cone $Q_m$ of $E_m \to E$ lies in $\langle A_{m+1}, \ldots, A_n \rangle \subset \langle \calA[k] : k \leqslant -1 \rangle = \Dge{1}$.
		Thus 
		\[
		E_m \to E \to Q_m \to E_m[1]
		\] 
		is the required triangle.
		So $(\Dle{0}, \Dge{0})$ is a t-structure.
		
		\step{boundedness}
		With notation as in Step~2, Lemma~\ref{lem:cone_of_filtration} gives 
		\[
		E \in \langle A_1, \ldots, A_n \rangle \subset \langle \calA[k] : k_n \leqslant k \leqslant k_1 \rangle \subset \Dle{-k_n} \cap \Dge{-k_1}.
		\]
		
		\step{the heart is $\calA$}
		Clearly $\calA \subset \Dle{0} \cap \Dge{0}$.
		Conversely let $0 \neq E \in \Dle{0} \cap \Dge{0}$, with a sequence as in \condref{cond:BH2}.
		We claim $k_1 \leqslant 0$.
		Otherwise $k_1 \geqslant 1$, and $\Hom(A_1, E) = 0$ by Lemma~\ref{lem:devissage}, since $\Hom(B_1[k_1], B[k]) = \Hom(B_1, B[k - k_1]) = 0$ for $B \in \calA$, $k \leqslant 0$.
		So the map $E_1 = A_1 \to E$ is zero, and the triangle 
		\[
		E_1 \xrightarrow{0} E \to Q_1 \to E_1[1]
		\] 
		splits.
		Indeed, by \condref{cond:TR-coh} applied to $\Hom(E_1[1], -)$ the map $Q_1 \to E_1[1]$ has a section, so $E_1[1]$ is a direct summand of $Q_1$.
		But $Q_1 \in \langle A_2, \ldots, A_n \rangle$ by Lemma~\ref{lem:cone_of_filtration}, and 
		\[
		\Hom(B_1[k_1 + 1], B_i[k_i]) = \Hom(B_1, B_i[k_i - k_1 - 1]) = 0
		\] 
		for $i \geqslant 2$ since $k_i < k_1$;
		hence $\Hom(E_1[1], Q_1) = 0$ by Lemma~\ref{lem:devissage}, so the identity of $E_1[1]$ factors through $0$ and $B_1 = 0$, a contradiction.
		Dually, $k_n \geqslant 0$.
		Otherwise $k_n \leqslant -1$ and $\Hom(E, A_n) = 0$ by Lemma~\ref{lem:devissage}, since $E \in \langle \calA[k] : k \geqslant 0 \rangle$ and $\Hom(B[k], B_n[k_n]) = \Hom(B, B_n[k_n - k]) = 0$ for $k \geqslant 0$.
		So in the triangle 
		\[
		E_{n-1} \to E \to A_n \to E_{n-1}[1]
		\] 
		the middle map is zero, and by \condref{cond:TR-coh} applied to $\Hom(-, A_n)$ the map $A_n \to E_{n-1}[1]$ has a retraction, so $A_n$ is a direct summand of $E_{n-1}[1] \in \langle A_1[1], \ldots, A_{n-1}[1] \rangle$.
		But $\Hom(B_i[k_i + 1], B_n[k_n]) = \Hom(B_i, B_n[k_n - k_i - 1]) = 0$ for $i < n$ because $k_n < k_i$, so $\Hom(E_{n-1}[1], A_n) = 0$ by Lemma~\ref{lem:devissage}, and again $B_n = 0$, a contradiction.
		Hence $0 \geqslant k_1 > \ldots > k_n \geqslant 0$, which forces $n = 1$ and $k_1 = 0$, i.e.\ $E = E_1 = A_1 = B_1 \in \calA$.
		
		\step{the filtration is the cohomology filtration}
		For the t-structure just constructed, Step~2 shows that for every $j$ the object $E_{m_j}$, where $m_j := \#\{ i : k_i \geqslant -j \}$, is $\tau^{\leqslant j} E$ (its cone in $E$ lies in $\Dge{j+1}$, and it lies in $\Dle{j}$).
		Hence the sequence in \condref{cond:BH2} is obtained from the cohomology filtration by discarding the trivial steps, and $B_i[k_i] = \Cone(\tau^{\leqslant -k_i - 1} E \to \tau^{\leqslant -k_i} E) = H^{-k_i}(E)[k_i]$.
	\end{proof}
	
	\begin{corollary}[{\cite[Exercise~5.6]{MS17}}]
		\label{cor:hearts_inclusion}
		Let $\calA \subset \calB$ be hearts of bounded t-structures on $\calD$. Then $\calA = \calB$.
	\end{corollary}
	
	\begin{proof}
		By Lemma~\ref{lem:cohomology_filtration}(3), $\Dle{0}_{\calA} = \langle \calA[k] : k \geqslant 0 \rangle \subset \langle \calB[k] : k \geqslant 0 \rangle = \Dle{0}_{\calB}$ and likewise $\Dge{0}_{\calA} \subset \Dge{0}_{\calB}$.
		Taking right orthogonals in the first inclusion and using Lemma~\ref{lem:t_orthogonal}(1) gives $\Dge{1}_{\calA} \supset \Dge{1}_{\calB}$, i.e.\ $\Dge{0}_{\calA} \supset \Dge{0}_{\calB}$.
		Hence $\Dge{0}_{\calA} = \Dge{0}_{\calB}$, so the two t-structures coincide by Lemma~\ref{lem:t_orthogonal}(1), and so do their hearts.
	\end{proof}
	
	\subsection{The Grothendieck group}
	
	The \emph{Grothendieck group} $\Kgroup(\calD)$ of the (essentially small) triangulated category $\calD$ is the abelian group generated by the isomorphism classes $[E]$ of objects of $\calD$ subject to the relations $[Y] = [X] + [Z]$ for every triangle $X \to Y \to Z \to X[1]$.
	From the triangles $X \to 0 \to X[1] \to X[1]$ and $X \to X \oplus Z \to Z \to X[1]$ one gets
	\[
	[X[1]] = -[X], \qquad [X \oplus Z] = [X] + [Z], \qquad [0] = 0 ,
	\]
	and for a sequence $0 = E_0 \to E_1 \to \ldots \to E_n = E$ with cones $A_i$ one gets $[E] = \sum_i [A_i]$ by induction.
	
	\begin{proposition}[{\cite[Exercise~5.4]{MS17}}]
		\label{prop:K0_heart}
		Let $\calA$ be the heart of a bounded t-structure on $\calD$.
		Then the homomorphism
		\[
		\iota \colon \Kgroup(\calA) \longrightarrow \Kgroup(\calD), \qquad [A] \longmapsto [A],
		\]
		is an isomorphism, with inverse $\chi \colon [E] \mapsto \sum_{i \in \ZZ} (-1)^i [H^i(E)]$.
	\end{proposition}
	
	\begin{proof}
		The map $\iota$ is well defined.
		A short exact sequence 
		\[
		0 \to A' \to A \to A'' \to 0
		\] 
		in $\calA$ is a triangle in $\calD$ by Lemma~\ref{lem:triangles_vs_ses}(2), so $[A] = [A'] + [A'']$ holds in $\Kgroup(\calD)$.
		
		The map $\chi$ is well defined.
		The sum is finite by Lemma~\ref{lem:cohomology_filtration}(1).
		First, for a finite exact sequence 
		\[0 \to M_1 \to M_2 \to \ldots \to M_r \to 0\] 
		in $\calA$ one has $\sum_j (-1)^j [M_j] = 0$ in $\Kgroup(\calA)$.
		Splitting it into the short exact sequences
		\[
		0 \to I_{j-1} \to M_j \to I_j \to 0 ,
		\]
		where $I_j$ is the image of $M_j \to M_{j+1}$ and $I_0 = 0 = I_r$, gives $[M_j] = [I_{j-1}] + [I_j]$, and the alternating sum telescopes to $0$.
		Now let $X \to Y \to Z \to X[1]$ be a triangle.
		By Theorem~\ref{thm:H0_cohomological} and Lemma~\ref{lem:cohomology_filtration}(1) the long exact sequence
		\[
		0 \to H^{a} X \to H^{a} Y \to H^{a} Z \to H^{a+1} X \to \ldots \to H^{b} X \to H^{b} Y \to H^{b} Z \to 0
		\]
		is a finite exact sequence in $\calA$, where $a \leqslant b$ are chosen so that all cohomology objects of $X, Y, Z$ outside $[a, b]$ vanish.
		Its terms in positions $3(i - a) + 1$, $3(i - a) + 2$, $3(i - a) + 3$ are $H^i X$, $H^i Y$, $H^i Z$, so the alternating sum is $\pm \sum_i (-1)^{i} \big( [H^i X] - [H^i Y] + [H^i Z] \big)$, and its vanishing says $\chi(Y) = \chi(X) + \chi(Z)$.
		
		Next, $\chi \circ \iota = \id$, because for $A \in \calA$ Lemma~\ref{lem:H0_basic}(2) gives $H^0 A = A$ and $H^i A = 0$ for $i \neq 0$.
		
		Finally, $\iota \circ \chi = \id$.
		For $E \in \calD$ the cohomology filtration of Lemma~\ref{lem:cohomology_filtration}(1) has factors $H^i(E)[-i]$, so
		\[
		[E] = \sum_i [H^i(E)[-i]] = \sum_i (-1)^i [H^i(E)]
		\]
		in $\Kgroup(\calD)$.
	\end{proof}
	
	\begin{remark}
		Different bounded t-structures on the same $\calD$ thus have hearts with canonically isomorphic Grothendieck groups, all identified with $\Kgroup(\calD)$.
		This is what allows one to fix a single lattice $\Lambda$ and a map $\Kgroup(\calD) \to \Lambda$ once and for all when studying stability conditions whose hearts vary.
		Note also that the construction only makes sense for essentially small categories, and that $\Kgroup$ of a triangulated category with arbitrary direct sums vanishes (Eilenberg swindle: $X \oplus \bigoplus_{\NN} X \cong \bigoplus_{\NN} X$ forces $[X] = 0$).
	\end{remark}
	
	\subsection{Examples}
	
	\begin{example}[the standard t-structure, {\cite[Exemple~1.3.2]{BBD82}}]
		\label{ex:standard_t}
		Let $\calA$ be an abelian category and let $\calD := \D^{\mathrm{b}}(\calA)$ be its bounded derived category.
		We use cohomological grading, so a complex is $X = (\ldots \to X^{i-1} \xrightarrow{d^{i-1}} X^i \xrightarrow{d^i} X^{i+1} \to \ldots)$, and $H^i(X) = \ker d^i / \image d^{i-1}$.
		We recall the facts about $\D^{\mathrm{b}}(\calA)$ that we need, with references to \cite{GM03} for proofs (\cite[Ch.~III]{GM03} works with $\D(\calA)$, but all arguments apply verbatim to $\D^{\mathrm{b}}(\calA)$, see \cite[Prop.~III.4.2 and Thm~III.4.4]{GM03}).
		\begin{itemize}
			\item[] \condlabel{D1}{cond:D1} (\cite[Lemma~III.2.8, Prop.~III.4.2, Thm~III.4.4]{GM03}) The objects of $\calD$ are the bounded complexes in $\calA$, and every morphism $X \to Y$ in $\calD$ can be written as a \emph{roof} $X \xleftarrow{s} X' \xrightarrow{f} Y$, where $f$ and $s$ are morphisms of complexes and $s$ is a quasi-isomorphism; such a roof represents the zero morphism if and only if there is a quasi-isomorphism $t \colon X'' \to X'$ such that $f t$ is homotopic to zero.
			A morphism of complexes defines an isomorphism in $\calD$ if and only if it is a quasi-isomorphism.
			\item[] \condlabel{D2}{cond:D2} (\cite[Cor.~IV.2.7 and Prop.~IV.2.8]{GM03}) $\calD$ is triangulated, the shift being $X[1]^i = X^{i+1}$ with differential $-d$, and every short exact sequence of complexes $0 \to X' \to X \to X'' \to 0$ gives rise to a triangle $X' \to X \to X'' \to X'[1]$ in $\calD$; every triangle in $\calD$ is isomorphic to one of this form.
			\item[] \condlabel{D3}{cond:D3} (\cite[Thm~III.3.6]{GM03}) The cohomology objects $H^i(X)$ depend only on the isomorphism class of $X$ in $\calD$ and define functors $H^i \colon \calD \to \calA$; a triangle $X \to Y \to Z \to X[1]$ induces a long exact sequence $\ldots \to H^i X \to H^i Y \to H^i Z \to H^{i+1} X \to \ldots$ in $\calA$.
		\end{itemize}
		An object $A \in \calA$ is regarded as the complex $A[0]$ concentrated in degree $0$.
		
		\step{the standard t-structure}
		Define
		\begin{align*}
			&\Dle{0} := \defineset{ X \in \calD }{ H^i(X) = 0 \text{ for all } i > 0 }, \\
			&\Dge{0} := \defineset{ X \in \calD }{ H^i(X) = 0 \text{ for all } i < 0 } .
		\end{align*}
		These are subcategories closed under isomorphisms by \condref{cond:D3}, and $\Dle{n}$, $\Dge{n}$ consist of the complexes with cohomology vanishing in degrees $> n$, respectively $< n$.
		We claim that $(\Dle{0}, \Dge{0})$ is a bounded t-structure on $\calD$, the \emph{standard t-structure}.
		Condition \condref{cond:T1} is clear from the description of $\Dle{n}$ and $\Dge{n}$.
		
		For \condref{cond:T3}, let $X \in \calD$ and consider the subcomplex and the quotient complex
		\begin{align*}
			X_{\leqslant 0} := &\big( \ldots \to X^{-2} \to X^{-1} \to \ker d^0 \to 0 \to \ldots \big), \\
			X_{\geqslant 1} := X / X_{\leqslant 0} = &\big( \ldots \to 0 \to X^0 / \ker d^0 \to X^1 \to X^2 \to 
			\ldots \big),
		\end{align*}
		with $\ker d^0$ and $X^0 / \ker d^0$ placed in degree $0$.
		A direct computation gives $H^i(X_{\leqslant 0}) = H^i(X)$ for $i \leqslant 0$ and $H^i(X_{\leqslant 0}) = 0$ for $i > 0$;
		similarly $H^0(X_{\geqslant 1}) = 0$, since $X^0 / \ker d^0 \cong \image d^0 \to X^1$ is a monomorphism, and $H^i(X_{\geqslant 1}) = \ker d^i / \image d^{i-1} = H^i(X)$ for $i \geqslant 1$.
		So $X_{\leqslant 0} \in \Dle{0}$, $X_{\geqslant 1} \in \Dge{1}$, and by \condref{cond:D2} the short exact sequence $0 \to X_{\leqslant 0} \to X \to X_{\geqslant 1} \to 0$ gives a triangle
		\[
		X_{\leqslant 0} \longrightarrow X \longrightarrow X_{\geqslant 1} \longrightarrow X_{\leqslant 0}[1]
		\]
		of the required form.
		
		For \condref{cond:T2}, let $X \in \Dle{0}$, $Y \in \Dge{1}$ and let $X \xleftarrow{s} X' \xrightarrow{f} Y$ be a roof as in \condref{cond:D1}.
		Replacing $Y$ by the quasi-isomorphic quotient $Y_{\geqslant 1}$ (which changes the roof by an isomorphism of $\calD$), we may assume $Y^i = 0$ for $i < 0$ and that $d^0 \colon Y^0 \to Y^1$ is a monomorphism.
		Since $s$ is a quasi-isomorphism, $H^i(X') = H^i(X) = 0$ for $i > 0$, so the inclusion $t \colon X'_{\leqslant 0} \to X'$ is a quasi-isomorphism.
		The composite $g := f t$ is a morphism of complexes from a complex concentrated in degrees $\leqslant 0$ to a complex concentrated in degrees $\geqslant 0$, so $g^i = 0$ for $i \neq 0$.
		Moreover $d^0_Y g^0 = g^1 d^0_{X'} = 0$, and $d^0_Y$ is a monomorphism, so $g^0 = 0$ as well and $f t = 0$.
		By \condref{cond:D1} the roof represents the zero morphism, and $\Hom(X, Y) = 0$.
		
		Boundedness is clear, since a bounded complex has cohomology only in finitely many degrees.
		By the uniqueness in Lemma~\ref{lem:truncation_unique}(2), the truncation functors of this t-structure are $\tau^{\leqslant 0} X = X_{\leqslant 0}$ and $\tau^{\geqslant 1} X = X_{\geqslant 1}$, and more generally $\tau^{\leqslant n}$ and $\tau^{\geqslant n}$ are the usual truncations of complexes.
		
		\step{the heart}
		The heart $\calH := \Dle{0} \cap \Dge{0}$ consists of the complexes whose cohomology is concentrated in degree $0$.
		We show that
		\[
		H^0 \colon \calH \longrightarrow \calA
		\qquad\text{and}\qquad
		\calA \longrightarrow \calH, \quad A \longmapsto A[0],
		\]
		are mutually quasi-inverse equivalences of categories.
		
		First, every $X \in \calH$ is isomorphic in $\calD$ to $H^0(X)[0]$.
		Indeed, the inclusion $X_{\leqslant 0} \to X$ is a quasi-isomorphism because $H^i(X) = 0$ for $i > 0$, and the morphism of complexes $X_{\leqslant 0} \to H^0(X)[0]$ which is the projection $\ker d^0 \to \ker d^0 / \image d^{-1} = H^0(X)$ in degree $0$ and zero elsewhere is a quasi-isomorphism because $H^i(X_{\leqslant 0}) = H^i(X) = 0$ for $i < 0$.
		By \condref{cond:D1} both maps are isomorphisms in $\calD$, and the resulting isomorphism $X \cong H^0(X)[0]$ induces the identity on $H^0$.
		
		Second, the functor $A \mapsto A[0]$ is fully faithful, i.e.\ for $A, B \in \calA$ the map
		\[
		\Hom_{\calA}(A, B) \longrightarrow \Hom_{\calD}(A[0], B[0])
		\]
		is bijective.
		For surjectivity, let $A[0] \xleftarrow{s} X' \xrightarrow{f} B[0]$ be a roof.
		As above we may replace $X'$ by $X'_{\leqslant 0}$ and assume $X'^i = 0$ for $i > 0$; then $H^0(X') = X'^0 / \image d^{-1}$.
		A morphism of complexes from $X'$ to a complex concentrated in degree $0$ is the same as a morphism $X'^0 \to B$ vanishing on $\image d^{-1}$, i.e.\ a morphism $H^0(X') \to B$;
		thus $f$ and $s$ are induced by morphisms $\bar f \colon H^0(X') \to B$ and $\bar s \colon H^0(X') \to A$, and $\bar s$ is an isomorphism because $s$ is a quasi-isomorphism.
		Let $p \colon X' \to H^0(X')[0]$ be the projection, a quasi-isomorphism.
		Then $s = \bar s p$ and $f = \bar f p$, so the roof $(s, f)$ represents the same morphism as the roof $A[0] \xleftarrow{\bar s} H^0(X')[0] \xrightarrow{\bar f} B[0]$, which is the morphism $\bar f \bar s^{-1} \colon A \to B$ of $\calA$.
		For injectivity, let $g \colon A \to B$ be a morphism in $\calA$ which becomes zero in $\calD$.
		By \condref{cond:D1} there is a quasi-isomorphism $t \colon X'' \to A[0]$ with $g t$ homotopic to zero, and after replacing $X''$ by $X''_{\leqslant 0}$ we may assume $X''^i = 0$ for $i > 0$.
		A homotopy between $g t$ and $0$ consists of maps $X''^{i} \to B[0]^{i-1}$, which all vanish since $B[0]$ is concentrated in degree $0$ and $X''^{1} = 0$;
		so $g t = 0$ as a morphism of complexes, i.e.\ $g$ vanishes on the image of $t^0 \colon X''^0 \to A$.
		This image is all of $A$, because $t$ induces an isomorphism $H^0(X'') = X''^0 / \image d^{-1} \to A$.
		Hence $g = 0$.
		
		Combining the two facts, $A \mapsto A[0]$ is fully faithful with image $\calH$ (up to isomorphism), so it is an equivalence $\calA \to \calH$, and $H^0$ is a quasi-inverse since $H^0(A[0]) = A$.
		An equivalence of categories preserves kernels and cokernels, so the abelian structure which Theorem~\ref{thm:heart_abelian} puts on $\calH$ is the one transported from $\calA$;
		in particular, by Lemma~\ref{lem:triangles_vs_ses}, the short exact sequences of $\calA$ are exactly the triangles of $\calD$ with all three vertices in $\calA$, up to isomorphism.
		Likewise the cohomology functors of the t-structure are the usual ones, since $\tau^{\geqslant 0} \tau^{\leqslant 0} X = (X_{\leqslant 0})_{\geqslant 0}$ is the complex $X^{-1} / \ker d^{-1} \to \ker d^0$ in degrees $-1, 0$, which is isomorphic in $\calD$ to $H^0(X)[0]$ by the first part of this step.
		
		\step{consequences}
		Proposition~\ref{prop:K0_heart} gives $\Kgroup(\D^{\mathrm{b}}(\calA)) \cong \Kgroup(\calA)$, the isomorphism being $[X] \mapsto \sum_i (-1)^i [H^i(X)]$.
		Finally, the bounded derived category of the heart is $\D^{\mathrm{b}}(\calH) \simeq \D^{\mathrm{b}}(\calA) = \calD$, tautologically.
		For an arbitrary bounded t-structure on a triangulated category $\calD$ with heart $\calH$ this last statement has no analogue in general.
		There is a natural way to compare $\D^{\mathrm{b}}(\calH)$ with $\calD$ when $\calD$ comes with an enhancement (the \emph{realization functor} of \cite[\S 3.1]{BBD82}), but it need not be an equivalence, and for triangulated categories which are not derived categories of anything, such as the one in Example~\ref{ex:SH}, there is nothing to compare at all.
	\end{example}
	
	\begin{example}[the stable homotopy category]
		\label{ex:SH}
		Let $\SH$ be the stable homotopy category.
		Its objects are spectra, i.e.\ sequences $X = (X_n)_{n \geqslant 0}$ of pointed spaces with structure maps $\Sigma X_n \to X_{n+1}$, and its morphisms are obtained from maps of spectra by inverting the stable weak equivalences, i.e.\ the maps inducing isomorphisms on all \emph{stable homotopy groups}
		\[
		\pi_k(X) := \operatorname*{colim}_{n} \pi_{n+k}(X_n), \qquad k \in \ZZ .
		\]
		We shall use the following facts, for which we refer to \cite[Part~III]{Ada74} and \cite{Mar83} for the classical construction and to \cite[\S 1.4.3]{Lur17} for the $\infty$-categorical one;
		in particular \condref{cond:S4} and the identification of the heart below are contained in \cite[Prop.~1.4.3.6]{Lur17}, and \condref{cond:S2} is \cite[Remark~1.4.3.8]{Lur17}.
		In \cite{Mar83} results are numbered within chapters, so ``Prop.~3.6'' means Proposition~6 of Chapter~3.
		Note that homotopy groups are graded homologically, so some care is needed when comparing with the cohomological conventions of this section.
		\begin{itemize}
			\item[] \condlabel{S1}{cond:S1} (\cite[Ch.~2, Axioms~1 and 2]{Mar83}) $\SH$ is a triangulated category with arbitrary coproducts, whose shift is the suspension $X[1] = \Sigma X$, and $\pi_k(\Sigma X) = \pi_{k-1}(X)$.
			The functors $\pi_k \colon \SH \to \Ab$ are homological and commute with coproducts (\cite[Part~III, Props.~3.9 and 3.10]{Ada74}, \cite[Ch.~2, Axiom~4]{Mar83}): a triangle $X \to Y \to Z \to \Sigma X$ induces a long exact sequence $\ldots \to \pi_k X \to \pi_k Y \to \pi_k Z \to \pi_{k-1} X \to \ldots$
			\item[] \condlabel{S2}{cond:S2} (\emph{Whitehead's theorem}, \cite[Part~III, Cor.~3.5]{Ada74}, \cite[Ch.~2, Axiom~4]{Mar83}) A morphism $f$ in $\SH$ is an isomorphism if and only if $\pi_k(f)$ is an isomorphism for all $k$; equivalently, $X = 0$ if and only if $\pi_k(X) = 0$ for all $k$.
			\item[] \condlabel{S3}{cond:S3} (\emph{Postnikov sections}, \cite[Prop.~3.8]{Mar83}) For every $X \in \SH$ and $n \in \ZZ$ there is a morphism $X \to X_{\leqslant n}$ which induces isomorphisms on $\pi_k$ for $k \leqslant n$, where $\pi_k(X_{\leqslant n}) = 0$ for $k > n$.
			\item[] \condlabel{S4}{cond:S4} (\cite[Prop.~3.6]{Mar83}, \cite[Prop.~1.4.3.6]{Lur17}) If $\pi_k(X) = 0$ for all $k < 0$ and $\pi_k(Y) = 0$ for all $k \geqslant 0$, then $\Hom_{\SH}(X, Y) = 0$.
			\item[] \condlabel{S5}{cond:S5} (\emph{Eilenberg--MacLane spectra}, \cite[Thm~6.1 and Prop.~3.6]{Mar83}) For every abelian group $A$ there is a spectrum $HA$ with $\pi_0(HA) = A$ and $\pi_k(HA) = 0$ for $k \neq 0$.
			If $\pi_k(X) = 0$ for all $k < 0$, then $\pi_0$ induces a bijection $\Hom_{\SH}(X, HA) \to \Hom_{\Ab}(\pi_0 X, A)$.
		\end{itemize}
		A spectrum $X$ with $\pi_k(X) = 0$ for $k < 0$ is called \emph{connective}, and \emph{coconnective} if $\pi_k(X) = 0$ for $k > 0$.
		
		\step{the Postnikov t-structure}
		Define
		\begin{align*}
			\SH^{\leqslant 0} &:= \defineset{ X \in \SH }{ \pi_k(X) = 0 \text{ for all } k < 0 }, \\
			\SH^{\geqslant 0} &:= \defineset{ X \in \SH }{ \pi_k(X) = 0 \text{ for all } k > 0 },
		\end{align*}
		the connective and the coconnective spectra.
		Since $\pi_k(X[n]) = \pi_{k-n}(X)$, we get
		\[
		\SH^{\leqslant n} = \defineset{ X }{ \pi_k(X) = 0 \text{ for } k < -n },
		\qquad
		\SH^{\geqslant n} = \defineset{ X }{ \pi_k(X) = 0 \text{ for } k > -n } ;
		\]
		the direction of the inequalities is reversed with respect to Example~\ref{ex:standard_t} because homotopy groups are graded homologically.
		We claim that $(\SH^{\leqslant 0}, \SH^{\geqslant 0})$ is a t-structure, the \emph{Postnikov t-structure} (\cite[Prop.~1.4.3.6]{Lur17}).
		Condition \condref{cond:T1} is clear from the displayed description, and \condref{cond:T2} is exactly \condref{cond:S4}, since $\SH^{\geqslant 1} = \{ X : \pi_k(X) = 0 \text{ for } k \geqslant 0 \}$.
		For \condref{cond:T3}, let $X \in \SH$, let $p \colon X \to X_{\leqslant -1}$ be a Postnikov section as in \condref{cond:S3}, and complete it to a triangle
		\[
		A \longrightarrow X \xrightarrow{\ p\ } X_{\leqslant -1} \longrightarrow \Sigma A .
		\]
		Then $X_{\leqslant -1} \in \SH^{\geqslant 1}$ by construction, and we show $A \in \SH^{\leqslant 0}$ using the long exact sequence of \condref{cond:S1}
		\[
		\pi_{k+1} X \xrightarrow{\ \pi_{k+1}(p)\ } \pi_{k+1} X_{\leqslant -1} \longrightarrow \pi_k A \longrightarrow \pi_k X \xrightarrow{\ \pi_k(p)\ } \pi_k X_{\leqslant -1} .
		\]
		Let $k \leqslant -1$.
		Then $\pi_k(p)$ is an isomorphism, so $\pi_k A$ is the cokernel of $\pi_{k+1}(p)$, and $\pi_{k+1}(p)$ is surjective, being an isomorphism if $k + 1 \leqslant -1$ and a map to $\pi_0 X_{\leqslant -1} = 0$ if $k + 1 = 0$.
		Hence $\pi_k A = 0$ for $k \leqslant -1$, i.e.\ $A$ is connective.
		This proves \condref{cond:T3}, and by Lemma~\ref{lem:truncation_unique} the truncation functors are $\tau^{\geqslant 1} X = X_{\leqslant -1}$ and $\tau^{\leqslant 0} X = A$, the \emph{connective cover} of $X$.
		
		\step{the heart}
		The heart $\SH^{\heartsuit} := \SH^{\leqslant 0} \cap \SH^{\geqslant 0}$ consists of the spectra with homotopy concentrated in degree $0$, and by \condref{cond:S5} it contains the Eilenberg--MacLane spectra.
		We show that
		\[
		\pi_0 \colon \SH^{\heartsuit} \longrightarrow \Ab
		\qquad\text{and}\qquad
		H \colon \Ab \longrightarrow \SH^{\heartsuit}, \quad A \longmapsto HA,
		\]
		are mutually quasi-inverse equivalences of categories.
		Let $X, Y \in \SH^{\heartsuit}$ and put $A := \pi_0 X$, $B := \pi_0 Y$.
		By \condref{cond:S5} there is a unique morphism $u_Y \colon Y \to HB$ inducing the identity on $\pi_0$, and by \condref{cond:S2} it is an isomorphism, since both sides have no other homotopy groups.
		Thus every object of the heart is an Eilenberg--MacLane spectrum, and the composite
		\[
		\Hom_{\SH}(X, Y) \xrightarrow{\ (u_Y)_* \ } \Hom_{\SH}(X, HB) \xrightarrow{\ \pi_0 \ } \Hom_{\Ab}(A, B)
		\]
		is bijective by \condref{cond:S5}; it equals the map $f \mapsto \pi_0(f)$, because $\pi_0(u_Y) = \id_B$.
		So $\pi_0$ is fully faithful on $\SH^{\heartsuit}$, essentially surjective by \condref{cond:S5}, hence an equivalence, and $H$ is a quasi-inverse because $\pi_0(HA) = A$.
		In particular the abelian structure of Theorem~\ref{thm:heart_abelian} on $\SH^{\heartsuit}$ is that of $\Ab$, and, as in Example~\ref{ex:standard_t}, the cohomology functor $H^0$ of the t-structure is identified with $\pi_0$;
		more precisely $H^i(X) = \tau^{\geqslant 0} \tau^{\leqslant 0}(X[i])$ corresponds to $\pi_0(X[i]) = \pi_{-i}(X)$ under $\pi_0$.
		
		\step{the t-structure is non-degenerate but not bounded}
		By \condref{cond:S2}, $\bigcap_n \SH^{\leqslant n} = \{ X : \pi_k(X) = 0 \text{ for all } k \} = 0$, and likewise $\bigcap_n \SH^{\geqslant n} = 0$, so the Postnikov t-structure is non-degenerate.
		It is not bounded.
		Indeed, by \condref{cond:S1} the spectrum $X := \bigoplus_{k \geqslant 0} \Sigma^k H\ZZ$ has $\pi_k(X) = \ZZ$ for all $k \geqslant 0$, so $X \notin \SH^{\geqslant n}$ for every $n$.
		(The sphere spectrum is another example, since infinitely many of its homotopy groups are nonzero, but this is a deep theorem, whereas the previous example is formal.)
		Consequently Lemma~\ref{lem:cohomology_filtration} and Proposition~\ref{prop:K0_heart} do not apply.
		
		\step{$\SH$ is not a derived category}
		Although the heart of the Postnikov t-structure is $\Ab$, the triangulated category $\SH$ is not equivalent to $\D(\Ab)$.
		We give the classical argument.
		For an object $X$ of a triangulated category write $X / 2$ for a cone of $2 \cdot \id_X \colon X \to X$.
		
		In $\D(\Ab)$ one has $2 \cdot \id_{X/2} = 0$ for every object $X$ (this is a special case of \cite[Prop.~1]{Sch10}).
		Indeed, represent $X$ by a complex; then $X / 2$ is represented by the cone $C$ with $C^i = X^{i+1} \oplus X^i$ and $d(a, b) = (-da, 2a + db)$.
		The maps $h^i \colon C^i \to C^{i-1}$, $h^i(a, b) = (b, 0)$, satisfy
		\[
		(dh + hd)(a, b) = (-db, 2b) + (2a + db, 0) = (2a, 2b),
		\]
		so $2 \cdot \id_C = dh + hd$ is null-homotopic and hence zero in $\D(\Ab)$.
		
		In $\SH$, on the other hand, the \emph{mod $2$ Moore spectrum} $S / 2$, the cone of $2 \cdot \id$ on the sphere spectrum $S$, satisfies $2 \cdot \id_{S/2} \neq 0$ \cite[Prop.~4]{Sch10}; in fact $\End(S / 2) \cong \ZZ / 4$, generated by the identity (loc.\ cit.).
		
		Now suppose $F \colon \SH \to \D(\Ab)$ were an exact equivalence.
		Since $F$ is additive, $F(2 \cdot \id_S) = 2 \cdot \id_{F(S)}$, and since $F$ is exact it sends the triangle $S \xrightarrow{2} S \to S / 2 \to \Sigma S$ to a triangle, so $F(S / 2) \cong F(S) / 2$.
		Then $F(2 \cdot \id_{S/2}) = 2 \cdot \id_{F(S/2)} = 0$ by the first part, whereas $2 \cdot \id_{S/2} \neq 0$ and $F$ is faithful, a contradiction.
		The same argument applies verbatim to $\D(\mathrm{Mod}\text{-}R)$ for any ring $R$, and more generally \cite[Props.~1 and 4]{Sch10} show that the full subcategory of finite spectra, hence $\SH$ itself, is not equivalent to any ``algebraic'' triangulated category, in particular not to the derived category of any abelian category.
	\end{example}


	\section{Slicings}

	\subsection{Definition}

	Throughout this section $\calD$ is an essentially small triangulated category, and we keep the conventions of Section~2.1;
	in particular all subcategories are full and closed under isomorphisms, and $\langle \mathcal{S} \rangle$ denotes the extension closure of a class of objects $\mathcal{S}$, described in Proposition~\ref{prop:extension_closure}.

	Proposition~\ref{prop:bounded_heart} describes the heart of a bounded t-structure by means of filtrations whose factors are shifts $B_i[k_i]$ of objects of the heart, with strictly decreasing integers $k_i$.
	A slicing is obtained by replacing the integers $k_i$ by real numbers.

	A \emph{slicing} $\calP$ of $\calD$ (\cite[Def.~3.3]{Bri07}, \cite[Def.~5.5]{MS17}) is a collection of subcategories $\calP(\varphi) \subset \calD$, one for each $\varphi \in \RR$, each containing the zero object, such that
	\begin{itemize}
		\item[] \condlabel{SL1}{cond:SL1} $\calP(\varphi + 1) = \calP(\varphi)[1]$ for all $\varphi \in \RR$;
		\item[] \condlabel{SL2}{cond:SL2} $\Hom(A, B) = 0$ whenever $A \in \calP(\varphi_1)$, $B \in \calP(\varphi_2)$ and $\varphi_1 > \varphi_2$;
		\item[] \condlabel{SL3}{cond:SL3} every nonzero object $E \in \calD$ admits a finite sequence of morphisms
		\[
		0 = E_0 \to E_1 \to \ldots \to E_{n-1} \to E_n = E
		\]
		whose cones $A_i := \Cone(E_{i-1} \to E_i)$, $i = 1, \ldots, n$, are nonzero objects of $\calP(\varphi_i)$ for some real numbers $\varphi_1 > \varphi_2 > \ldots > \varphi_n$.
	\end{itemize}
	The nonzero objects of $\calP(\varphi)$ are called \emph{$\calP$-semistable of phase $\varphi$}, and we write $\varphi(E) := \varphi$ for such an object.
	The phase of a semistable object is well defined, since a nonzero object $E$ satisfies $\Hom(E, E) \neq 0$ and therefore lies in at most one of the subcategories $\calP(\varphi)$ by \condref{cond:SL2}.
	A sequence as in \condref{cond:SL3} is called a \emph{Harder--Narasimhan filtration} (\emph{HN filtration}) of $E$ with respect to $\calP$, the objects $A_i$ are its \emph{HN factors}, and we put $\varphi^+(E) := \varphi_1$ and $\varphi^-(E) := \varphi_n$.
	\begin{center}
		\begin{tikzcd}[column sep=large, row sep=large]
			0 = E_0 \arrow[r]
			& E_1 \arrow[r] \arrow[d]
			& E_2 \arrow[r] \arrow[d]
			& \cdots \arrow[r]
			& E_{n-1} \arrow[r] \arrow[d]
			& E_n = E \arrow[d] \\
			& A_1 & A_2 & & A_{n-1} & A_n
		\end{tikzcd}
	\end{center}
	Each column $E_{i-1} \to E_i \to A_i$ of the diagram is part of a triangle $E_{i-1} \to E_i \to A_i \to E_{i-1}[1]$.
	Until uniqueness of HN filtrations is proved in Proposition~\ref{prop:HN_unique_slicing}, the numbers $\varphi^{\pm}(E)$ refer to a chosen filtration.
	A semistable object $E$ has the HN filtration $0 \to E$ of length $1$, and conversely an object with an HN filtration of length $1$ is isomorphic to $A_1$, hence is semistable.

	For a subset $I \subset \RR$ we define
	\[
	\calP(I) := \big\langle \calP(\varphi) : \varphi \in I \big\rangle ,
	\]
	the extension closure of the union of the $\calP(\varphi)$ with $\varphi \in I$.
	We write $\calP(\geqslant \varphi)$, $\calP(> \varphi)$, $\calP([a, b))$ and so on for the subsets $I = [\varphi, \infty)$, $(\varphi, \infty)$, $[a, b)$, and $\calP(\emptyset) = \{ 0 \}$.
	By Proposition~\ref{prop:extension_closure}, $\calP(I)$ consists of the objects admitting a filtration whose factors lie in $\calP(\varphi)$ for various $\varphi \in I$, in no particular order of the phases.
	Clearly $\calP(I) \subset \calP(J)$ for $I \subset J$, and \condref{cond:SL1} together with Proposition~\ref{prop:extension_closure} gives
	\[
	\calP(I)[n] = \calP(I + n) \qquad (n \in \ZZ) .
	\]

	\begin{lemma}
		\label{lem:slicing_orthogonal}
		Let $I, J \subset \RR$ be subsets such that $\varphi > \psi$ for all $\varphi \in I$ and $\psi \in J$.
		Then $\Hom(X, Y) = 0$ for all $X \in \calP(I)$ and $Y \in \calP(J)$.
	\end{lemma}

	\begin{proof}
		By \condref{cond:SL2}, $\Hom(A, B) = 0$ for all $A \in \bigcup_{\varphi \in I} \calP(\varphi)$ and $B \in \bigcup_{\psi \in J} \calP(\psi)$, and Lemma~\ref{lem:devissage} extends this to the extension closures.
	\end{proof}

	\begin{lemma}
		\label{lem:HN_steps}
		Let $0 = E_0 \to E_1 \to \ldots \to E_n = E$ be an HN filtration of $E \neq 0$ with factors $A_i \in \calP(\varphi_i)$, $\varphi_1 > \ldots > \varphi_n$, and for $0 \leqslant m \leqslant n$ let $Q_m$ be a cone of $E_m \to E$.
		\begin{enumerate}
			\item $E_m \in \calP([\varphi_m, \varphi_1])$ for $1 \leqslant m \leqslant n$, and $Q_m \in \calP([\varphi_n, \varphi_{m+1}])$ for $0 \leqslant m < n$; in particular $E \in \calP([\varphi_n, \varphi_1])$.
			\item $E_m \neq 0$ for $1 \leqslant m \leqslant n$, and $Q_m \neq 0$ for $0 \leqslant m < n$.
			\item The morphisms $E_1 = A_1 \to E$ and $E \to Q_{n-1} \cong A_n$ are nonzero.
		\end{enumerate}
	\end{lemma}

	\begin{proof}
		(1) By Lemma~\ref{lem:cone_of_filtration}, $E_m \in \langle A_1, \ldots, A_m \rangle$ and $Q_m \in \langle A_{m+1}, \ldots, A_n \rangle$, and these extension closures are contained in $\calP([\varphi_m, \varphi_1])$ and $\calP([\varphi_n, \varphi_{m+1}])$ respectively.

		(2) Suppose $E_m = 0$ for some $m \geqslant 1$.
		The triangle 
		\[
		E_{m-1} \to E_m \to A_m \to E_{m-1}[1]
		\] 
		then gives $E_{m-1} \cong A_m[-1]$ by \condref{cond:TR-5}.
		If $m = 1$ this says $A_1 = 0$, a contradiction.
		If $m \geqslant 2$, then $E_{m-1} \in \calP([\varphi_{m-1}, \varphi_1])$ by (1) and $E_{m-1} \cong A_m[-1] \in \calP(\varphi_m - 1)$, where $\varphi_m - 1 < \varphi_{m-1}$;
		so Lemma~\ref{lem:slicing_orthogonal} gives $\Hom(E_{m-1}, E_{m-1}) = 0$, hence $E_{m-1} = 0$ and $A_m = 0$, a contradiction.
		Dually, suppose $Q_m = 0$ for some $m < n$.
		Lemma~\ref{lem:cone_of_filtration} provides a triangle
		\[
		A_{m+1} \to Q_m \to Q_{m+1} \to A_{m+1}[1] ,
		\]
		so $Q_{m+1} \cong A_{m+1}[1] \in \calP(\varphi_{m+1} + 1)$.
		If $m + 1 = n$, then $Q_n = 0$ gives $A_n = 0$, a contradiction.
		If $m + 1 < n$, then $Q_{m+1} \in \calP([\varphi_n, \varphi_{m+2}])$ by (1), and $\varphi_{m+2} < \varphi_{m+1} + 1$, so $\Hom(Q_{m+1}, Q_{m+1}) = 0$ by Lemma~\ref{lem:slicing_orthogonal}, hence $Q_{m+1} = 0$ and $A_{m+1} = 0$, a contradiction.

		(3) Suppose that $E_1 \to E$ is zero.
		Applying \condref{cond:TR-coh} with $\Hom(E_1[1], -)$ to the triangle
		\[
		E_1 \xrightarrow{0} E \to Q_1 \to E_1[1]
		\] 
		shows that $\Hom(E_1[1], Q_1) \to \Hom(E_1[1], E_1[1])$ is surjective, so the map $Q_1 \to E_1[1]$ has a section.
		But $E_1[1] = A_1[1] \in \calP(\varphi_1 + 1)$ and $Q_1 \in \calP([\varphi_n, \varphi_2])$ with $\varphi_2 < \varphi_1 + 1$ (if $n = 1$, then $Q_1 = 0$), so $\Hom(E_1[1], Q_1) = 0$ by Lemma~\ref{lem:slicing_orthogonal}.
		Hence the identity of $E_1[1]$ is zero and $A_1 = 0$, a contradiction.
		Dually, the triangle 
		\[
		A_n \to Q_{n-1} \to Q_n = 0
		\] 
		gives $Q_{n-1} \cong A_n$.
		If $E \to Q_{n-1}$ were zero, then \condref{cond:TR-coh} with $\Hom(-, Q_{n-1})$ applied to the triangle 
		\[
		E_{n-1} \to E \xrightarrow{0} Q_{n-1} \to E_{n-1}[1]
		\] 
		would show that $Q_{n-1} \to E_{n-1}[1]$ has a retraction, while $\Hom(E_{n-1}[1], Q_{n-1}) = 0$ by (1) and Lemma~\ref{lem:slicing_orthogonal}, because $E_{n-1}[1] \in \calP([\varphi_{n-1} + 1, \varphi_1 + 1])$ and $\varphi_n < \varphi_{n-1} + 1$;
		so $Q_{n-1} = 0$, contradicting (2).
	\end{proof}

	\subsection{The t-structures associated with a slicing}

	\begin{proposition}
		\label{prop:slicing_t}
		Let $\calP$ be a slicing of $\calD$ and $\psi \in \RR$.
		\begin{enumerate}
			\item The pair $(\calP(\geqslant \psi), \calP(< \psi + 1))$ is a bounded t-structure on $\calD$ with heart $\calP([\psi, \psi + 1))$.
			\item The pair $(\calP(> \psi), \calP(\leqslant \psi + 1))$ is a bounded t-structure on $\calD$ with heart $\calP((\psi, \psi + 1])$.
			\item Let $E \neq 0$ have an HN filtration $0 = E_0 \to \ldots \to E_n = E$ with phases $\varphi_1 > \ldots > \varphi_n$, let $Q_m$ be a cone of $E_m \to E$, and put $m := \# \{ i : \varphi_i \geqslant \psi \}$.
			Then
			\[
			E_m \to E \to Q_m \to E_m[1]
			\]
			is a truncation triangle of $E$ with respect to the t-structure in (1), i.e.\ $\tau^{\leqslant 0} E \cong E_m$ and $\tau^{\geqslant 1} E \cong Q_m$ compatibly with the morphisms to and from $E$.
			The same holds for the t-structure in (2) with $m := \# \{ i : \varphi_i > \psi \}$.
		\end{enumerate}
	\end{proposition}

	We write $E_{\geqslant \psi} \to E \to E_{< \psi}$ for the truncation triangle of the t-structure in (1), so that $E_{\geqslant \psi} \in \calP(\geqslant \psi)$ and $E_{< \psi} \in \calP(< \psi)$, and likewise $E_{> \psi} \to E \to E_{\leqslant \psi}$ for the t-structure in (2);
	for the slicing attached to a bounded heart in Proposition~\ref{prop:discrete_slicing}(1) below one has $E_{\geqslant -n} = \tau^{\leqslant n} E$ and $E_{< -n} = \tau^{\geqslant n + 1} E$, which for the standard t-structure are the truncations $X_{\leqslant n}$ and $X_{\geqslant n + 1}$ of complexes used in Example~\ref{ex:standard_t}.
	Beware that the aisle of these t-structures consists of the objects of \emph{large} phase, whereas in the notation $\Dle{0}$ for a t-structure the aisle is suggested to consist of objects of small degree;
	this is the unavoidable clash of conventions noted in \cite[footnote in \S 3]{Bri07}, and it is the reason for the reversal of inequalities already observed in Example~\ref{ex:SH}.

	\begin{proof}
		We prove (1) and the first half of (3); the proofs of (2) and of the second half of (3) are obtained by replacing $\geqslant$ with $>$ and $<$ with $\leqslant$ throughout.
		Put $\Dle{0} := \calP(\geqslant \psi)$ and $\Dge{0} := \calP(< \psi + 1)$, so that
		\[
		\Dle{n} = \calP(\geqslant \psi - n), \qquad \Dge{n} = \calP(< \psi + 1 - n)
		\]
		by $\calP(I)[n] = \calP(I + n)$.

		\step{the axioms}
		Condition \condref{cond:T1} holds because $\calP(\geqslant \psi + 1) \subset \calP(\geqslant \psi)$ and $\calP(< \psi) \subset \calP(< \psi + 1)$.
		Condition \condref{cond:T2}, i.e.\ $\Hom(\calP(\geqslant \psi), \calP(< \psi)) = 0$, is Lemma~\ref{lem:slicing_orthogonal}.
		For \condref{cond:T3} let $E \neq 0$ (for $E = 0$ take the zero triangle), choose an HN filtration of $E$ and let $m$ be as in (3), so that 
		\[
		\varphi_1 > \ldots > \varphi_m \geqslant \psi > \varphi_{m+1} > \ldots > \varphi_n.
		\]
		By Lemma~\ref{lem:HN_steps}(1), $E_m \in \calP([\varphi_m, \varphi_1]) \subset \calP(\geqslant \psi)$ if $m \geqslant 1$, and $E_0 = 0$;
		similarly $Q_m \in \calP([\varphi_n, \varphi_{m+1}]) \subset \calP(< \psi)$ if $m < n$, and $Q_n = 0$.
		Hence $E_m \to E \to Q_m \to E_m[1]$ is a triangle as required in \condref{cond:T3}, and $(\Dle{0}, \Dge{0})$ is a t-structure.
		By Lemma~\ref{lem:truncation_unique}(2) this triangle is isomorphic to the truncation triangle, which proves (3).

		\step{boundedness}
		With $E$ as above, $E \in \calP([\varphi_n, \varphi_1])$ by Lemma~\ref{lem:HN_steps}(1).
		Choose integers $a, b \geqslant 0$ with $\psi - a \leqslant \varphi_n$ and $\varphi_1 < \psi + 1 + b$;
		then $E \in \calP(\geqslant \psi - a) \cap \calP(< \psi + 1 + b) = \Dle{a} \cap \Dge{-b}$.

		\step{the heart}
		The inclusion $\calP([\psi, \psi + 1)) \subset \calP(\geqslant \psi) \cap \calP(< \psi + 1)$ is clear.
		Conversely let $0 \neq E \in \Dle{0} \cap \Dge{0}$, with HN filtration and notation as above.
		Since $E \in \Dle{0}$, Lemma~\ref{lem:t_orthogonal}(2) gives $\tau^{\geqslant 1} E = 0$, i.e.\ $Q_m = 0$ by (3), so $m = n$ by Lemma~\ref{lem:HN_steps}(2), i.e.\ $\varphi_n \geqslant \psi$.
		Since $E \in \Dge{0}$, Lemma~\ref{lem:t_orthogonal}(2) gives $\tau^{\leqslant -1} E = 0$;
		now $\Dle{-1} = \calP(\geqslant \psi + 1)$, so applying (3) to the t-structure of (1) for $\psi + 1$ in place of $\psi$ (and the uniqueness of truncation triangles) we get $E_{m'} = 0$ for $m' := \# \{ i : \varphi_i \geqslant \psi + 1 \}$, hence $m' = 0$ by Lemma~\ref{lem:HN_steps}(2), i.e.\ $\varphi_1 < \psi + 1$.
		Thus all HN factors of $E$ lie in $\calP([\psi, \psi + 1))$, and so does $E$.
	\end{proof}

	\subsection{Uniqueness of Harder--Narasimhan filtrations}

	\begin{proposition}
		\label{prop:HN_unique_slicing}
		Let $\calP$ be a slicing of $\calD$ and $E \neq 0$.
		If
		\[
		0 = E_0 \to E_1 \to \ldots \to E_n = E
		\qquad\text{and}\qquad
		0 = E'_0 \to E'_1 \to \ldots \to E'_m = E
		\]
		are HN filtrations of $E$ with phases $\varphi_1 > \ldots > \varphi_n$ and $\varphi'_1 > \ldots > \varphi'_m$, then $n = m$, $\varphi_i = \varphi'_i$ for all $i$, and for every $i$ there is a unique isomorphism $\varepsilon_i \colon E_i \to E'_i$ compatible with the morphisms to $E$.
		The $\varepsilon_i$ are compatible with the morphisms $E_{i-1} \to E_i$ and $E'_{i-1} \to E'_i$, and they induce isomorphisms $A_i \cong A'_i$ of the HN factors.
		In particular the phases $\varphi^{\pm}(E)$, the set of phases of the HN factors and the isomorphism classes of the HN factors depend only on $E$.
	\end{proposition}

	\begin{proof}
		Put $\Phi := \{ \varphi_1, \ldots, \varphi_n \}$ and $\Phi' := \{ \varphi'_1, \ldots, \varphi'_m \}$.

		\step{the set of phases is determined by $E$}
		Fix $\psi \in \RR$.
		Since $\calP(> \psi) \subset \calP(\geqslant \psi)$, Lemma~\ref{lem:truncation_unique}(1) applied to the t-structure of Proposition~\ref{prop:slicing_t}(1) gives a unique morphism
		\[
		c_\psi \colon E_{> \psi} \longrightarrow E_{\geqslant \psi}
		\]
		compatible with the morphisms to $E$.
		We claim that $\psi \in \Phi$ if and only if $c_\psi$ is not an isomorphism.
		Put $k := \# \{ i : \varphi_i \geqslant \psi \}$ and $k' := \# \{ i : \varphi_i > \psi \}$.
		By Proposition~\ref{prop:slicing_t}(3) and Lemma~\ref{lem:truncation_unique}(2) there are isomorphisms $E_{\geqslant \psi} \cong E_k$ and $E_{> \psi} \cong E_{k'}$ compatible with the morphisms to $E$, and under them $c_\psi$ corresponds to a morphism $E_{k'} \to E_k$ compatible with the morphisms to $E$.
		Such a morphism is unique, by Lemma~\ref{lem:truncation_unique}(1) applied to $E_{k'} \in \calP(\geqslant \psi)$ and the truncation triangle 
		\[
		E_k \to E \to Q_k \to E_k[1],
		\]
		so it is the composite $E_{k'} \to E_{k'+1} \to \ldots \to E_k$ of the morphisms of the filtration.
		If $\psi \notin \Phi$, then $k = k'$ and this composite is the identity.
		If $\psi \in \Phi$, say $\psi = \varphi_k$, then $k = k' + 1$ and the composite is the morphism $E_{k-1} \to E_k$, whose cone $A_k$ is nonzero;
		it is not an isomorphism, since the cone of an isomorphism is zero by \condref{cond:TR-5}.
		This proves the claim.
		The morphism $c_\psi$ does not depend on the filtration, so the same claim for the second filtration gives $\Phi = \Phi'$.
		Hence $n = m$ and $\varphi_i = \varphi'_i$ for all $i$.

		\step{the isomorphisms $\varepsilon_i$}
		Fix $i$ and put $\psi := \varphi_i = \varphi'_i$.
		Then $i = \# \{ j : \varphi_j \geqslant \psi \}$, so by Proposition~\ref{prop:slicing_t}(3) both $E_i \to E \to Q_i$ and $E'_i \to E \to Q'_i$ are truncation triangles of $E$ for the t-structure $(\calP(\geqslant \psi), \calP(< \psi + 1))$.
		Lemma~\ref{lem:truncation_unique}(2) gives a unique isomorphism $\varepsilon_i \colon E_i \to E'_i$ compatible with the morphisms to $E$.
		For the compatibility with the filtrations, note that both composites
		\[
		E_{i-1} \to E_i \xrightarrow{\varepsilon_i} E'_i
		\qquad\text{and}\qquad
		E_{i-1} \xrightarrow{\varepsilon_{i-1}} E'_{i-1} \to E'_i
		\]
		are morphisms from $E_{i-1} \in \calP(\geqslant \varphi_{i-1}) \subset \calP(\geqslant \psi)$ to $E'_i = E_{\geqslant \psi}$ compatible with the morphisms to $E$, hence they coincide by Lemma~\ref{lem:truncation_unique}(1).
		Finally, the commutative square formed by $E_{i-1} \to E_i$, $E'_{i-1} \to E'_i$, $\varepsilon_{i-1}$ and $\varepsilon_i$ extends to a morphism of triangles $(\varepsilon_{i-1}, \varepsilon_i, \alpha_i)$ by the axioms of a triangulated category, and $\alpha_i \colon A_i \to A'_i$ is an isomorphism by \condref{cond:TR-5}.
	\end{proof}

	From now on $\varphi^{\pm}(E)$ and the HN factors of a nonzero object $E$ are defined using any HN filtration.

	\begin{corollary}
		\label{cor:slicing_phases}
		Let $\calP$ be a slicing of $\calD$, let $E, F$ be nonzero objects and $\psi \in \RR$.
		\begin{enumerate}
			\item $E \in \calP(\geqslant \psi)$ if and only if $\varphi^-(E) \geqslant \psi$, and $E \in \calP(> \psi)$ if and only if $\varphi^-(E) > \psi$;
			dually $E \in \calP(\leqslant \psi)$ if and only if $\varphi^+(E) \leqslant \psi$, and $E \in \calP(< \psi)$ if and only if $\varphi^+(E) < \psi$.
			\item For every interval $I \subset \RR$ one has $E \in \calP(I)$ if and only if $\varphi^-(E) \in I$ and $\varphi^+(E) \in I$.
			In particular $E$ is semistable if and only if $\varphi^+(E) = \varphi^-(E)$, and $\calP(\varphi) = \calP(\{ \varphi \})$ for every $\varphi \in \RR$.
			\item If $\Hom(E, F) \neq 0$, then $\varphi^-(E) \leqslant \varphi^+(F)$.
			\item For every interval $I \subset \RR$ the subcategory $\calP(I)$ is closed under extensions, finite direct sums and direct summands.
			In particular this applies to $\calP(\varphi) = \calP(\{ \varphi \})$.
		\end{enumerate}
	\end{corollary}

	\begin{proof}
		(1) Let $0 = E_0 \to \ldots \to E_n = E$ be an HN filtration with phases $\varphi_1 > \ldots > \varphi_n$, let $Q_k$ be a cone of $E_k \to E$ and $k := \# \{ i : \varphi_i \geqslant \psi \}$.
		For the t-structure $(\Dle{0}, \Dge{0}) = (\calP(\geqslant \psi), \calP(< \psi + 1))$ of Proposition~\ref{prop:slicing_t}, Lemma~\ref{lem:t_orthogonal}(2) says that $E \in \Dle{0}$ if and only if $\tau^{\geqslant 1} E = 0$, i.e.\ $Q_k = 0$ by Proposition~\ref{prop:slicing_t}(3), which by Lemma~\ref{lem:HN_steps}(2) means $k = n$, i.e.\ $\varphi_n \geqslant \psi$.
		Likewise $E \in \Dge{1} = \calP(< \psi)$ if and only if $\tau^{\leqslant 0} E = 0$, i.e.\ $E_k = 0$, i.e.\ $k = 0$, i.e.\ $\varphi_1 < \psi$.
		The statements for $\calP(> \psi)$ and $\calP(\leqslant \psi)$ follow in the same way from the t-structure of Proposition~\ref{prop:slicing_t}(2).

		(2) If $\varphi^{\pm}(E) \in I$, then all HN factors of $E$ lie in $\calP(\varphi_i)$ with $\varphi_i \in I$, so $E \in \calP(I)$ by Lemma~\ref{lem:HN_steps}(1).
		Conversely let $E \in \calP(I)$.
		If $I$ is bounded below with infimum $a$, then $I \subset [a, \infty)$ or $I \subset (a, \infty)$ according to whether $a \in I$ or not, so $E \in \calP(\geqslant a)$ or $E \in \calP(> a)$, and (1) gives $\varphi^-(E) \geqslant a$ or $\varphi^-(E) > a$ accordingly.
		In the same way, if $I$ is bounded above with supremum $b$, then $\varphi^+(E) \leqslant b$ or $\varphi^+(E) < b$ according to whether $b \in I$ or not.
		Since $\varphi^-(E) \leqslant \varphi^+(E)$, both numbers satisfy the bounds defining $I$, so they lie in $I$.
		For the second assertion, if $E \in \calP(\varphi)$ then $0 \to E$ is an HN filtration and $\varphi^{\pm}(E) = \varphi$;
		conversely if $\varphi^+(E) = \varphi^-(E) = \varphi$, then $n = 1$ and $E \cong A_1 \in \calP(\varphi)$.
		Finally $\calP(\varphi) \subset \calP(\{ \varphi \})$ by definition, and a nonzero $E \in \calP(\{ \varphi \})$ has $\varphi^{\pm}(E) = \varphi$ by what was just proved, so $E \in \calP(\varphi)$.

		(3) By Lemma~\ref{lem:HN_steps}(1), $E \in \calP([\varphi^-(E), \varphi^+(E)])$ and $F \in \calP([\varphi^-(F), \varphi^+(F)])$.
		If $\varphi^-(E) > \varphi^+(F)$, Lemma~\ref{lem:slicing_orthogonal} gives $\Hom(E, F) = 0$.

		(4) Closure under extensions and finite direct sums holds for every extension closure.
		By (1) and (2), $\calP(I)$ is the intersection of at most two of the subcategories $\calP(\geqslant a)$, $\calP(> a)$, $\calP(\leqslant b)$, $\calP(< b)$, where $a$ and $b$ are the finite endpoints of $I$ (or $\calP(I) = \calD$ if $I = \RR$, or $\calP(I) = 0$ if $I = \emptyset$).
		Each of these is the aisle or the coaisle of one of the t-structures of Proposition~\ref{prop:slicing_t}, namely $\calP(\geqslant a) = \Dle{0}$ and $\calP(< b) = \Dge{0}$ for the t-structures of type (1) with $\psi = a$ and $\psi = b - 1$, and $\calP(> a) = \Dle{0}$ and $\calP(\leqslant b) = \Dge{0}$ for those of type (2);
		so each is closed under direct summands by Lemma~\ref{lem:t_orthogonal}(3), and so is their intersection.
	\end{proof}

	\begin{lemma}[{\cite[Lemma~3.4]{Bri07}}]
		\label{lem:slicing_seesaw}
		Let $\calP$ be a slicing of $\calD$, let $I \subset \RR$ be an interval of length at most $1$, and let
		\[
		A \to E \to B \to A[1]
		\]
		be a triangle whose vertices $A, E, B$ are nonzero objects of $\calP(I)$.
		Then $\varphi^+(A) \leqslant \varphi^+(E)$ and $\varphi^-(E) \leqslant \varphi^-(B)$.
	\end{lemma}

	\begin{proof}
		Choose $t \in \RR$ with $I \subset [t, t + 1]$;
		then $A, E, B \in \calP([t, t + 1])$, so $\varphi^{\pm}$ of each of them lies in $[t, t + 1]$ by Corollary~\ref{cor:slicing_phases}(2).

		For the first inequality put $\varphi := \varphi^+(A)$ and let $A_1 \to A$ be the first step of an HN filtration of $A$, so that $A_1 \in \calP(\varphi)$ and $A_1 \to A$ is nonzero by Lemma~\ref{lem:HN_steps}(3).
		Suppose $\varphi > \varphi^+(E)$.
		Then $\Hom(A_1, E) = 0$ by Corollary~\ref{cor:slicing_phases}(3), so the composite $A_1 \to A \to E$ vanishes, and the exact sequence
		\[
		\Hom(A_1, B[-1]) \to \Hom(A_1, A) \to \Hom(A_1, E)
		\]
		of \condref{cond:TR-coh}, obtained from the rotated triangle $B[-1] \to A \to E \to B$, shows that the nonzero morphism $A_1 \to A$ factors through $B[-1]$.
		Hence $\Hom(A_1, B[-1]) \neq 0$.
		Since $B[-1] \in \calP([t - 1, t])$, Lemma~\ref{lem:slicing_orthogonal} forces $\varphi \leqslant t$.
		But $\varphi^+(E) \geqslant t$, so $\varphi \leqslant t \leqslant \varphi^+(E) < \varphi$, a contradiction.

		For the second inequality put $\psi := \varphi^-(B)$ and let $B \to B_n$ be the nonzero morphism onto the last HN factor $B_n \in \calP(\psi)$ of $B$ (Lemma~\ref{lem:HN_steps}(3)).
		Suppose $\psi < \varphi^-(E)$.
		Then $\Hom(E, B_n) = 0$ by Corollary~\ref{cor:slicing_phases}(3), so the composite $E \to B \to B_n$ vanishes, and the exact sequence
		\[
		\Hom(A[1], B_n) \to \Hom(B, B_n) \to \Hom(E, B_n)
		\]
		shows that $B \to B_n$ factors through $B \to A[1]$, whence $\Hom(A[1], B_n) \neq 0$.
		Since $A[1] \in \calP([t + 1, t + 2])$, Lemma~\ref{lem:slicing_orthogonal} forces $\psi \geqslant t + 1$.
		But $\varphi^-(E) \leqslant t + 1$, so $\psi \geqslant t + 1 \geqslant \varphi^-(E) > \psi$, a contradiction.
	\end{proof}

	\subsection{Slicings and bounded hearts}

	\begin{proposition}
		\label{prop:discrete_slicing}
		\begin{enumerate}
			\item Let $\calA$ be the heart of a bounded t-structure on $\calD$.
			Then
			\[
			\calP_{\calA}(n) := \calA[n] \quad (n \in \ZZ), \qquad \calP_{\calA}(\varphi) := 0 \quad (\varphi \notin \ZZ)
			\]
			is a slicing of $\calD$.
			Its HN filtrations are the cohomology filtrations of Lemma~\ref{lem:cohomology_filtration} with the trivial steps discarded, and for $E \neq 0$
			\[
			\varphi^+(E) = - \min \{ i : H^i(E) \neq 0 \}, \qquad \varphi^-(E) = - \max \{ i : H^i(E) \neq 0 \} .
			\]
			\item Conversely, let $\calP$ be a slicing of $\calD$ with $\calP(\varphi) = 0$ for all $\varphi \notin \ZZ$.
			Then $\calP(0)$ is the heart of a bounded t-structure and $\calP = \calP_{\calP(0)}$.
			\item For every slicing $\calP$ and every $\psi \in \RR$ the subcategories $\calP((\psi, \psi + 1])$ and $\calP([\psi, \psi + 1))$ are hearts of bounded t-structures on $\calD$;
			in particular they are abelian.
		\end{enumerate}
	\end{proposition}

	The abelian category $\calP((0, 1])$ is called the \emph{heart} of the slicing $\calP$.
	By Proposition~\ref{prop:K0_heart} we have $\Kgroup(\calD) \cong \Kgroup(\calP((0, 1]))$;
	this is how a central charge on $\Kgroup(\calD)$ will be compared with a stability function on the heart in the next section.

	\begin{proof}
		(1) Condition \condref{cond:SL1} holds by construction.
		For \condref{cond:SL2} let $A \in \calP_{\calA}(\varphi_1)$ and $B \in \calP_{\calA}(\varphi_2)$ be nonzero with $\varphi_1 > \varphi_2$;
		then $\varphi_1 = n$ and $\varphi_2 = m$ are integers with $n > m$, $A = A_0[n]$ and $B = B_0[m]$ with $A_0, B_0 \in \calA$, and $\Hom(A, B) = \Hom(A_0, B_0[m - n]) = 0$ by \eqref{eq:heart_negative_ext}.
		Condition \condref{cond:SL3} is \condref{cond:BH2} of Proposition~\ref{prop:bounded_heart}, whose factors $B_i[k_i]$ are nonzero objects of $\calP_{\calA}(k_i)$ with $k_1 > \ldots > k_n$;
		by the last assertion of Proposition~\ref{prop:bounded_heart} this sequence is the cohomology filtration with the trivial steps discarded, and $B_i = H^{-k_i}(E)$.
		Hence the HN factors of $E$ are the nonzero objects $H^i(E)[-i]$, which have phase $-i$, and the formulas for $\varphi^{\pm}(E)$ follow.

		(2) The subcategory $\calA := \calP(0)$ is additive by Corollary~\ref{cor:slicing_phases}(4).
		It satisfies \condref{cond:BH1}, because for $A, B \in \calA$ and $k < 0$ we have $B[k] \in \calP(k)$ and $\Hom(A, B[k]) = 0$ by \condref{cond:SL2}.
		It satisfies \condref{cond:BH2}, because in an HN filtration of a nonzero object the factors $A_i$ are nonzero objects of $\calP(\varphi_i)$, which forces $\varphi_i \in \ZZ$, so that $A_i = B_i[\varphi_i]$ with $0 \neq B_i := A_i[-\varphi_i] \in \calP(0)$ and strictly decreasing integers $\varphi_i$.
		By Proposition~\ref{prop:bounded_heart}, $\calA$ is the heart of a bounded t-structure.
		Finally $\calP(n) = \calP(0)[n] = \calP_{\calA}(n)$ for $n \in \ZZ$ by \condref{cond:SL1}, and $\calP(\varphi) = 0 = \calP_{\calA}(\varphi)$ for $\varphi \notin \ZZ$.

		(3) This is Proposition~\ref{prop:slicing_t} together with Theorem~\ref{thm:heart_abelian}.
	\end{proof}

	\begin{proposition}[{\cite[Exercise~5.6]{MS17}}]
		\label{prop:slicing_inclusion}
		Let $\calP$ and $\calP'$ be slicings of $\calD$ with $\calP'(\varphi) \subset \calP(\varphi)$ for all $\varphi \in \RR$.
		Then $\calP' = \calP$.
	\end{proposition}

	\begin{proof}
		Let $E$ be a nonzero object of $\calP(\varphi)$ and let $0 = E_0 \to \ldots \to E_n = E$ be an HN filtration of $E$ with respect to $\calP'$, with factors $A_i \in \calP'(\varphi_i) \subset \calP(\varphi_i)$.
		Then it is also an HN filtration of $E$ with respect to $\calP$.
		Since $0 \to E$ is an HN filtration of $E$ with respect to $\calP$ as well, Proposition~\ref{prop:HN_unique_slicing} gives $n = 1$ and $\varphi_1 = \varphi$, so $E \cong A_1 \in \calP'(\varphi)$.
	\end{proof}

	Proposition~\ref{prop:discrete_slicing}(3) says that $\calP(I)$ is abelian for every half-open interval $I$ of length $1$.
	For intervals of length $< 1$ the subcategory $\calP(I)$ is in general not abelian, not even for $I = \{ \varphi \}$.
	In the next section we shall see that $\calP(\varphi)$ is abelian as soon as the slicing underlies a stability condition (\cite[Lemma~5.2]{Bri07});
	the following example shows that this is a genuine restriction.

	\begin{example}[a slicing whose slices are not abelian]
		\label{ex:P1_slicing}
		Let $\kk$ be a field, $X := \projective^1_{\kk}$ and $\calD := \D^{\mathrm{b}}(\mathrm{Coh}(X))$, with $\mathrm{Coh}(X)$ regarded as the heart of the standard t-structure (Example~\ref{ex:standard_t}).
		We use the following facts.
		\begin{itemize}
			\item[] \condlabel{P1}{cond:P1} Every coherent sheaf $F$ on $X$ is isomorphic to $T \oplus \bigoplus_{j=1}^{r} \oo(n_j)$, where $T$ is a sheaf of finite length (a \emph{torsion sheaf}) and $n_1, \ldots, n_r \in \ZZ$.
			The decomposition of the locally free part is Grothendieck's theorem \cite{HM82};
			the splitting $F \cong T \oplus F / T$ of the maximal torsion subsheaf holds because $F / T$ is torsion-free, hence locally free on the smooth curve $X$, and $\Ext^1(F / T, T) \cong H^1(X, (F / T)^{\vee} \otimes T)$ (\cite[Props.~III.6.3 and III.6.7]{Har77}) vanishes because $(F / T)^{\vee} \otimes T$ has zero-dimensional support \cite[Thm~III.2.7]{Har77}.
			\item[] \condlabel{P2}{cond:P2} $\Hom(\oo(n), \oo(m)) = H^0(X, \oo(m - n))$ is zero for $n > m$ and nonzero for $n \leqslant m$ \cite[Thm~III.5.1]{Har77}; in particular $\Hom(\oo(n), \oo(n)) = \kk$.
			\item[] \condlabel{P3}{cond:P3} $\Hom(T, L) = 0$ for a torsion sheaf $T$ and a locally free sheaf $L$, since the local sections of $L$ have no torsion.
		\end{itemize}
		Choose real numbers $\psi_n \in (0, 1)$, $n \in \ZZ$, with $\psi_0 = \psi_1 = \tfrac12$, $\psi_n < \psi_{n+1}$ for $n \neq 0$, for instance
		\[
		\psi_n := 1 - \frac{1}{n + 1} \quad (n \geqslant 2), \qquad \psi_0 = \psi_1 := \frac12, \qquad \psi_n := \frac{1}{2 (1 - n)} \quad (n \leqslant -1) ,
		\]
		so that $\psi_n \leqslant \psi_m$ if and only if $n \leqslant m$ or $\{ n, m \} = \{ 0, 1 \}$.
		Define subcategories of $\mathrm{Coh}(X) \subset \calD$ by
		\begin{align*}
			\calP(1) &:= \{ \text{torsion sheaves} \}, \\
			\calP(\psi) &:= \operatorname{add} \{ \oo(n) : \psi_n = \psi \} \quad \text{for } \psi \in \{ \psi_n : n \in \ZZ \}, \\
			\calP(\varphi) &:= 0 \quad \text{for all other } \varphi \in (0, 1] ,
		\end{align*}
		where $\operatorname{add}$ denotes the objects isomorphic to finite direct sums of the listed ones, and extend by $\calP(\varphi + k) := \calP(\varphi)[k]$ for $\varphi \in (0, 1]$ and $k \in \ZZ$.
		Thus $\calP(\psi_n) = \operatorname{add} \{ \oo(n) \}$ for $n \neq 0, 1$ and $\calP(\tfrac12) = \operatorname{add} \{ \oo, \oo(1) \}$.

		\step{the axioms \condref{cond:SL1} and \condref{cond:SL2}}
		Condition \condref{cond:SL1} holds by construction.
		For \condref{cond:SL2} first let $A, B \in \mathrm{Coh}(X)$ be nonzero with $A \in \calP(\varphi_1)$, $B \in \calP(\varphi_2)$ and $1 \geqslant \varphi_1 > \varphi_2 > 0$.
		Then $B$ is a direct sum of line bundles $\oo(m)$ with $\psi_m = \varphi_2$, and $A$ is either a torsion sheaf, in which case $\Hom(A, B) = 0$ by \condref{cond:P3}, or a direct sum of line bundles $\oo(n)$ with $\psi_n = \varphi_1 > \psi_m$, in which case $n > m$ and $\Hom(A, B) = 0$ by \condref{cond:P2}.
		In general, $A \in \calP(\varphi_1)$ and $B \in \calP(\varphi_2)$ with $\varphi_1 > \varphi_2$ can be written as $A = A_0[a]$ and $B = B_0[b]$ with $a, b \in \ZZ$, $A_0 \in \calP(\varphi_1 - a)$, $B_0 \in \calP(\varphi_2 - b)$ and $\varphi_1 - a, \varphi_2 - b \in (0, 1]$.
		If $b > a$, then $\varphi_1 - a > \varphi_2 - a \geqslant \varphi_2 - b + 1 > 1$, which is impossible since $\varphi_1 - a \leqslant 1$;
		if $b < a$, then $\Hom(A, B) = \Hom(A_0, B_0[b - a]) = 0$ by \eqref{eq:heart_negative_ext};
		if $b = a$, then $\Hom(A, B) = \Hom(A_0, B_0)$ vanishes by the first case.

		\step{HN filtrations of sheaves}
		Let $0 \neq F \in \mathrm{Coh}(X)$ and decompose it as in \condref{cond:P1}.
		Grouping the line bundle summands according to their phases we obtain
		\[
		F \cong T \oplus F_1 \oplus \ldots \oplus F_r
		\]
		with $0 \neq F_l \in \calP(\psi^{(l)})$ and $1 > \psi^{(1)} > \ldots > \psi^{(r)}$.
		The sequence of partial sums
		\[
		0 \to T \to T \oplus F_1 \to T \oplus F_1 \oplus F_2 \to \ldots \to F
		\]
		(with the first step omitted if $T = 0$) is an HN filtration of $F$ with respect to $\calP$, since the cone of the inclusion of a direct summand $Y \to Y \oplus Z$ is $Z$.

		\step{HN filtrations of complexes}
		Let $0 \neq E \in \calD$.
		By Lemma~\ref{lem:cohomology_filtration}(1) applied to the standard t-structure, $E$ has a filtration by the truncations $\tau^{\leqslant i} E$ with cones $H^i(E)[-i]$, and we discard the steps with $H^i(E) = 0$.
		For each remaining $i$, Step~2 gives an HN filtration of the sheaf $H^i(E)$, and shifting it by $[-i]$ gives an HN filtration of $H^i(E)[-i]$ with phases in $(-i, -i + 1]$, by \condref{cond:SL1}.
		Lemma~\ref{lem:refinement}, applied to the triangle 
		\[
		\tau^{\leqslant i - 1} E \to \tau^{\leqslant i} E \to H^i(E)[-i] \to (\tau^{\leqslant i - 1} E)[1]
		\] 
		and this filtration, replaces the step $\tau^{\leqslant i - 1} E \to \tau^{\leqslant i} E$ by a sequence of morphisms whose cones are the HN factors of $H^i(E)[-i]$.
		Doing this for every $i$ we obtain a filtration of $E$ whose factors are semistable, with phases in $(-i, -i + 1]$ strictly decreasing within each block and the blocks ordered by increasing $i$;
		hence the phases are strictly decreasing overall, and this is an HN filtration of $E$.
		So $\calP$ is a slicing of $\calD$.

		\step{the slice $\calP(\frac12)$ is not abelian}
		Put $\calC := \calP(\tfrac12) = \operatorname{add} \{ \oo, \oo(1) \}$, an additive category whose objects are the sheaves $\oo^{\oplus a} \oplus \oo(1)^{\oplus b}$, and let $s \colon \oo \to \oo(1)$ be a nonzero morphism, which exists by \condref{cond:P2}.
		By \condref{cond:P2} a morphism $\oo^{\oplus a} \oplus \oo(1)^{\oplus b} \to \oo$ is given by $a$ scalars (its components $\oo(1) \to \oo$ vanish), and $s$ composed with a nonzero scalar multiple of $\id_{\oo}$ is nonzero;
		hence every morphism $g \colon Y \to \oo$ in $\calC$ with $s g = 0$ is zero, which says that $0 \to \oo$ is a kernel of $s$ in $\calC$.
		Dually, a morphism $\oo(1) \to \oo^{\oplus a} \oplus \oo(1)^{\oplus b}$ is given by $b$ scalars, and every morphism $h \colon \oo(1) \to Y$ in $\calC$ with $h s = 0$ is zero, so $\oo(1) \to 0$ is a cokernel of $s$ in $\calC$.
		Thus $s$ is a monomorphism and an epimorphism in $\calC$, but it is not an isomorphism, since $\Hom(\oo(1), \oo) = 0$.
		In an abelian category a morphism which is both a monomorphism and an epimorphism is an isomorphism, so $\calC$ is not abelian.
	\end{example}

	\section{Bridgeland stability conditions}

	\subsection{Definition}

	Throughout this section $\calD$ is an essentially small triangulated category with Grothendieck group $\Kgroup(\calD)$ as in Section~2.9.
	Following \cite[\S 5.2]{MS17} we fix once and for all a free abelian group $\Lambda$ of finite rank, a surjective group homomorphism
	\[
	v \colon \Kgroup(\calD) \twoheadrightarrow \Lambda ,
	\]
	and a norm $\| \cdot \|$ on the real vector space $\Lambda_{\RR} := \Lambda \otimes_{\ZZ} \RR$.
	All norms on $\Lambda_{\RR}$ are equivalent, and nothing below depends on the choice of $\| \cdot \|$.
	For an object $E$ we write $v(E) := v([E])$, and for a homomorphism $Z \colon \Lambda \to \CC$ we write $Z(E) := Z(v(E))$;
	thus $E \mapsto Z(E)$ is additive on triangles and satisfies $Z(E[1]) = -Z(E)$.

	A \emph{Bridgeland stability condition} on $\calD$ (with respect to $v$) is a pair $\sigma = (Z, \calP)$, where $Z \colon \Lambda \to \CC$ is a group homomorphism, the \emph{central charge}, and $\calP$ is a slicing of $\calD$, such that
	\begin{itemize}
		\item[] \condlabel{ST}{cond:ST} for every $\varphi \in \RR$ and every nonzero $E \in \calP(\varphi)$ there is a real number $m(E) > 0$ with
		\[
		Z(E) = m(E) \, e^{i \pi \varphi} ;
		\]
		\item[] \condlabel{SP}{cond:SP} (\emph{support property}) there is a constant $C > 0$ such that
		\[
		\| v(E) \| \leqslant C \, |Z(E)| \qquad \text{for every } \varphi \in \RR \text{ and every nonzero } E \in \calP(\varphi) .
		\]
	\end{itemize}
	This is \cite[Def.~5.8]{MS17}; the support property goes back to \cite{KS08}.
	We denote by $\operatorname{Stab}_v(\calD)$ the set of Bridgeland stability conditions on $\calD$ with respect to $v$.

	\begin{remark}
		Bridgeland's original definition \cite[Def.~1.1 and Def.~5.1]{Bri07} has $\Lambda = \Kgroup(\calD)$, $v = \id$ and no support property.
		Everything in this section except Proposition~\ref{prop:slice_abelian}(2) holds verbatim for pairs $(Z, \calP)$ satisfying only \condref{cond:ST}, for an arbitrary abelian group $\Lambda$;
		the support property is what makes the set $\operatorname{Stab}_v(\calD)$ a well-behaved space, and it is the subject of the next section, where we also prove the equivalent formulation of \condref{cond:SP} by means of a quadratic form on $\Lambda_{\RR}$ (\cite[Exercise~5.10]{MS17}).
		Note that \condref{cond:SP} is a condition on the set of semistable objects only, and that it implies in particular $v(E) \neq 0$, hence $Z(E) \neq 0$, for every semistable $E$ (which also follows from \condref{cond:ST}).
	\end{remark}

	We transfer the terminology of Section~3 from $\calP$ to $\sigma$:
	the nonzero objects of $\calP(\varphi)$ are called \emph{$\sigma$-semistable of phase $\varphi$}, the HN filtration of an object with respect to $\calP$ is its HN filtration with respect to $\sigma$, and $\varphi^{\pm}_{\sigma}(E) := \varphi^{\pm}(E)$.
	The number $m_{\sigma}(E) := m(E) = |Z(E)|$ is the \emph{mass} of a semistable object $E$; more generally, for an arbitrary nonzero object $E$ with HN factors $A_1, \ldots, A_n$ one puts $m_{\sigma}(E) := \sum_i |Z(A_i)|$, which is well defined by Proposition~\ref{prop:HN_unique_slicing}.

	Condition \condref{cond:ST} determines the phase of a semistable object $E$ in terms of $Z(E)$ up to an integer:
	$Z(E)$ is nonzero, and $\varphi(E)$ is one of the arguments of $Z(E)$ divided by $\pi$.
	Conversely the central charge is determined by the slicing and the masses of the semistable objects, by additivity.
	The analogy with Section~1 is as follows: if $(\calA, Z)$ is a stability condition on an abelian category in the sense of Section~1.3, then the semistable objects of phase $\varphi \in (0, 1]$ are the nonzero $E$ with $Z(E) \in \RR_{>0} e^{i \pi \varphi}$, and condition \condref{cond:Z} says that all phases lie in $(0, 1]$;
	a Bridgeland stability condition allows all real phases, the shift $[1]$ increasing the phase by $1$ and multiplying $Z$ by $-1$.
	The precise relation between the two notions is the content of the next subsection.

	\subsection{The bijection theorem}

	For a heart $\calA$ of a bounded t-structure on $\calD$ we denote by
	\[
	\iota_{\calA} \colon \Kgroup(\calA) \xrightarrow{\ \sim\ } \Kgroup(\calD)
	\]
	the isomorphism of Proposition~\ref{prop:K0_heart}, which sends $[E]$ to $[E]$.
	A homomorphism $Z \colon \Lambda \to \CC$ defines the homomorphism
	\[
	Z_{\calA} := Z \circ v \circ \iota_{\calA} \colon \Kgroup(\calA) \longrightarrow \CC ,
	\]
	so that $Z_{\calA}(E) = Z(E)$ for every $E \in \calA$.
	We say that a stability condition $(\calA, Z_{\calA})$ on the abelian category $\calA$ in the sense of Section~1.3 \emph{satisfies the support property} if
	\begin{itemize}
		\item[] \condlabel{SP-ab}{cond:SP-ab} there is a constant $C > 0$ with $\| v(E) \| \leqslant C \, |Z(E)|$ for every $(\calA, Z_{\calA})$-semistable object $E$.
	\end{itemize}
	Besides the set $\operatorname{Stab}_v(\calD)$ of Bridgeland stability conditions we consider the set $\operatorname{Stab}^{\mathrm{ab}}_v(\calD)$ of all pairs $(\calA, Z)$, where $\calA \subset \calD$ is the heart of a bounded t-structure and $Z \colon \Lambda \to \CC$ is a homomorphism such that $(\calA, Z_{\calA})$ is a stability condition on the abelian category $\calA$ satisfying \condref{cond:SP-ab}.
	Between these two sets we define two maps.
	The \emph{heart map}
	\[
	\Phi \colon \operatorname{Stab}_v(\calD) \longrightarrow \operatorname{Stab}^{\mathrm{ab}}_v(\calD) , \qquad \Phi(Z, \calP) := \big( \calP((0, 1]), Z \big) ,
	\]
	sends a stability condition to its heart together with the central charge.
	The \emph{slicing map}
	\[
	\Psi \colon \operatorname{Stab}^{\mathrm{ab}}_v(\calD) \longrightarrow \operatorname{Stab}_v(\calD) , \qquad \Psi(\calA, Z) := (Z, \calP_{\calA, Z}) ,
	\]
	is defined as follows: for $\varphi \in (0, 1]$ the subcategory $\calP_{\calA, Z}(\varphi) \subset \calD$ consists of the zero object and the $(\calA, Z_{\calA})$-semistable objects of phase $\varphi$, and $\calP_{\calA, Z}(\varphi + n) := \calP_{\calA, Z}(\varphi)[n]$ for $n \in \ZZ$.
	That $\Phi$ and $\Psi$ indeed take values in the indicated sets is part of the following theorem.

	\begin{theorem}[{\cite[Prop.~5.3]{Bri07}}, {\cite[Lemma~5.11]{MS17}}]
		\label{thm:bijection}
		The maps $\Phi$ and $\Psi$ are well defined and are mutually inverse bijections
		\[
		\Phi \colon \operatorname{Stab}_v(\calD) \xrightarrow{\ \sim\ } \operatorname{Stab}^{\mathrm{ab}}_v(\calD) , \qquad
		\Psi \colon \operatorname{Stab}^{\mathrm{ab}}_v(\calD) \xrightarrow{\ \sim\ } \operatorname{Stab}_v(\calD) .
		\]
		Moreover, if $\sigma = (Z, \calP) \in \operatorname{Stab}_v(\calD)$ and $(\calA, Z) = \Phi(\sigma)$, then for every $E \in \calA$ and $\varphi \in (0, 1]$:
		\begin{enumerate}
			\item $E$ is $\sigma$-semistable of phase $\varphi$ if and only if $E$ is $(\calA, Z_{\calA})$-semistable of phase $\varphi$;
			\item the HN filtration of $E$ with respect to $\sigma$ is a chain of subobjects of $E$ in $\calA$, and it is the HN filtration of $E$ with respect to $(\calA, Z_{\calA})$; in particular $\varphi^{\pm}_{\sigma}(E) = \varphi^{\pm}_{Z_{\calA}}(E)$;
			\item the constants $C$ in \condref{cond:SP} for $\sigma$ and in \condref{cond:SP-ab} for $(\calA, Z_{\calA})$ can be taken to be equal.
		\end{enumerate}
	\end{theorem}

	\begin{proof}
		\step{$\Phi$ is well defined: the stability function}
		Let $\sigma = (Z, \calP) \in \operatorname{Stab}_v(\calD)$ and $\calA := \calP((0, 1])$.
		By Proposition~\ref{prop:discrete_slicing}(3), $\calA$ is the heart of a bounded t-structure, so $Z_{\calA}$ is defined.
		Let $0 \neq E \in \calA$, with HN filtration $0 = E_0 \to E_1 \to \ldots \to E_n = E$ with respect to $\calP$ and factors $A_i \in \calP(\varphi_i)$.
		By Corollary~\ref{cor:slicing_phases}(2), all $\varphi_i$ lie in $(0, 1]$.
		In $\Kgroup(\calD)$ we have $[E] = \sum_i [A_i]$ (Section~2.9), hence
		\[
		Z_{\calA}(E) = Z(E) = \sum_{i=1}^{n} Z(A_i) = \sum_{i=1}^{n} m(A_i) \, e^{i \pi \varphi_i} \in \HH ,
		\]
		because each summand lies in $\HH$ by \condref{cond:ST} and $\HH$ is closed under addition.
		So $Z_{\calA}$ satisfies \condref{cond:Z}, i.e.\ it is a stability function on $\calA$.

		\step{the HN filtration of an object of the heart consists of subobjects}
		Keep $E$ and its HN filtration as in Step~1.
		By Lemma~\ref{lem:HN_steps}(1), $E_i \in \calP([\varphi_i, \varphi_1]) \subset \calP((0, 1]) = \calA$ for $i \geqslant 1$, and $A_i \in \calA$.
		Hence each triangle $E_{i-1} \to E_i \to A_i \to E_{i-1}[1]$ has all its vertices in $\calA$, and Lemma~\ref{lem:triangles_vs_ses}(1) shows that
		\[
		0 \to E_{i-1} \to E_i \to A_i \to 0
		\]
		is a short exact sequence in $\calA$.
		Thus $0 = E_0 \subset E_1 \subset \ldots \subset E_n = E$ is a chain of subobjects of $E$ in $\calA$ with factors $A_i$.
		Applying Lemma~\ref{lem:seesaw} to this chain, and noting that the $Z_{\calA}$-phase of $A_i$ is $\varphi_i$ by \condref{cond:ST} and $\varphi_i \in (0, 1]$, we obtain
		\begin{equation}
			\label{eq:phase_between}
			\varphi_n = \varphi^-_{\sigma}(E) \leqslant \varphi_{Z_{\calA}}(E) \leqslant \varphi^+_{\sigma}(E) = \varphi_1 .
		\end{equation}

		\step{semistable objects, i.e.\ assertion (1)}
		Let $E \in \calA$ be nonzero and $\varphi \in (0, 1]$.

		Suppose first that $E \in \calP(\varphi)$.
		Then $\varphi_{Z_{\calA}}(E) = \varphi$ by \condref{cond:ST}.
		Let $F \subset E$ be a nontrivial subobject in $\calA$ and $G := E / F$.
		If $G = 0$ there is nothing to prove, so let $G \neq 0$.
		By Lemma~\ref{lem:triangles_vs_ses}(2) the short exact sequence $0 \to F \to E \to G \to 0$ is a triangle
		\[
		F \to E \to G \to F[1]
		\]
		with nonzero vertices in $\calA = \calP((0, 1])$, an interval of length $1$, so Lemma~\ref{lem:slicing_seesaw} gives $\varphi^+_{\sigma}(F) \leqslant \varphi^+_{\sigma}(E) = \varphi$.
		By \eqref{eq:phase_between} applied to $F$, $\varphi_{Z_{\calA}}(F) \leqslant \varphi^+_{\sigma}(F) \leqslant \varphi$.
		Hence $E$ is $(\calA, Z_{\calA})$-semistable of phase $\varphi$.

		Conversely, suppose that $E$ is $(\calA, Z_{\calA})$-semistable of phase $\varphi$, and let $0 = E_0 \subset E_1 \subset \ldots \subset E_n = E$ be its HN filtration with respect to $\calP$, a chain of subobjects with factors $A_i \in \calP(\varphi_i)$ by Step~2.
		The subobject $E_1 = A_1$ of $E$ is nontrivial and has $Z_{\calA}$-phase $\varphi_1$, so \condref{cond:AZ-SS} gives $\varphi_1 \leqslant \varphi$.
		The quotient $A_n = E / E_{n-1}$ of $E$ is nonzero and has $Z_{\calA}$-phase $\varphi_n$, so Lemma~\ref{lem:quotients} gives $\varphi \leqslant \varphi_n$.
		Hence $\varphi_1 \leqslant \varphi \leqslant \varphi_n \leqslant \varphi_1$, so $n = 1$ and $E \cong A_1 \in \calP(\varphi)$.

		\step{$\Phi$ is well defined: the Harder--Narasimhan and support properties, and assertions (2), (3)}
		Let $0 \neq E \in \calA$.
		By Steps~2 and~3 its HN filtration with respect to $\calP$ is a chain of subobjects $0 = E_0 \subset \ldots \subset E_n = E$ in $\calA$ whose factors $A_i$ are $(\calA, Z_{\calA})$-semistable of phases $\varphi_1 > \ldots > \varphi_n$, i.e.\ an HN filtration of $E$ with respect to $(\calA, Z_{\calA})$.
		Thus $(\calA, Z_{\calA})$ satisfies \condref{cond:AZ-SC2} and is a stability condition on $\calA$, and by the uniqueness of HN filtrations in $\calA$ (Proposition~\ref{prop:HN_unique}) the two HN filtrations of $E$ coincide, which is (2).
		For the support property, let $E$ be $(\calA, Z_{\calA})$-semistable;
		by Step~3 it is $\sigma$-semistable, so $\| v(E) \| \leqslant C \, |Z(E)|$ with the constant $C$ of \condref{cond:SP}.
		Hence $(\calA, Z_{\calA})$ satisfies \condref{cond:SP-ab} with the same constant, $(\calA, Z) \in \operatorname{Stab}^{\mathrm{ab}}_v(\calD)$, and $\Phi$ is well defined.
		Conversely, every $\sigma$-semistable object is of the form $E[n]$ with $E \in \calA$ semistable (Corollary~\ref{cor:slicing_phases}(2) and \condref{cond:SL1}), and $\| v(E[n]) \| = \| v(E) \|$, $|Z(E[n])| = |Z(E)|$, so a constant for \condref{cond:SP-ab} also works for \condref{cond:SP}; this is (3).

		\step{$\Psi$ is well defined: the slicing}
		Let $(\calA, Z) \in \operatorname{Stab}^{\mathrm{ab}}_v(\calD)$ and write $\calP := \calP_{\calA, Z}$.
		Every real number $\psi$ can be written uniquely as $\psi = \varphi + n$ with $\varphi \in (0, 1]$ and $n \in \ZZ$, so the subcategories $\calP(\psi)$ are well defined, closed under isomorphisms, and contain $0$.
		By Lemma~\ref{lem:triangles_vs_ses}(2), a chain of subobjects $0 = F_0 \subset F_1 \subset \ldots \subset F_r = F$ in $\calA$ is the same as a filtration of $F$ in the sense of Section~2.6 with all $F_j$ in $\calA$, the factors $F_j / F_{j-1}$ being the cones;
		we use this identification in the rest of the proof.

		Condition \condref{cond:SL1} holds by construction.
		For \condref{cond:SL2} let $A \in \calP(\psi_1)$ and $B \in \calP(\psi_2)$ be nonzero with $\psi_1 > \psi_2$, and write $A = A_0[a]$, $B = B_0[b]$ with $a, b \in \ZZ$, $A_0 \in \calP(\psi_1 - a)$, $B_0 \in \calP(\psi_2 - b)$ and $\psi_1 - a, \psi_2 - b \in (0, 1]$;
		thus $A_0, B_0 \in \calA$ are $(\calA, Z_{\calA})$-semistable of phases $\psi_1 - a$ and $\psi_2 - b$.
		If $b > a$, then $\psi_1 - a > \psi_2 - a \geqslant \psi_2 - b + 1 > 1$, which is impossible since $\psi_1 - a \leqslant 1$.
		If $b < a$, then $\Hom(A, B) = \Hom(A_0, B_0[b - a]) = 0$ by \eqref{eq:heart_negative_ext}.
		If $b = a$, then $\psi_1 - a > \psi_2 - a$, and $\Hom(A, B) = \Hom(A_0, B_0) = 0$ by Schur's Lemma~\ref{lem:schur}(1).

		For \condref{cond:SL3} let $0 \neq E \in \calD$.
		By Lemma~\ref{lem:cohomology_filtration}(1) for the bounded t-structure with heart $\calA$, $E$ has a filtration by the truncations $\tau^{\leqslant i} E$ with cones $H^i(E)[-i]$, and we discard the steps with $H^i(E) = 0$.
		For each remaining $i$ let $0 = F_0 \subset F_1 \subset \ldots \subset F_r = H^i(E)$ be the HN filtration of $H^i(E) \in \calA$ with respect to $(\calA, Z_{\calA})$, with factors semistable of phases $\psi_1 > \ldots > \psi_r$ in $(0, 1]$, i.e.\ lying in $\calP(\psi_1), \ldots, \calP(\psi_r)$;
		shifting by $[-i]$ gives a filtration of $H^i(E)[-i]$ with factors in $\calP(\psi_j - i)$, whose phases $\psi_j - i$ lie in $(-i, -i + 1]$ and strictly decrease.
		Lemma~\ref{lem:refinement}, applied to the triangle $\tau^{\leqslant i - 1} E \to \tau^{\leqslant i} E \to H^i(E)[-i] \to (\tau^{\leqslant i - 1} E)[1]$ and this filtration, replaces the step $\tau^{\leqslant i - 1} E \to \tau^{\leqslant i} E$ by a sequence of morphisms whose cones are these factors.
		Doing this for every $i$ we obtain a filtration of $E$ whose factors are nonzero objects of the $\calP(\psi)$, with phases in $(-i, -i + 1]$ strictly decreasing within each block and the blocks ordered by increasing $i$;
		hence the phases strictly decrease overall, and we have an HN filtration of $E$ with respect to $\calP$.
		So $\calP$ is a slicing of $\calD$.

		\step{$\Psi$ is well defined: the central charge and the support property}
		Let $E \in \calP(\psi)$ be nonzero, $\psi = \varphi + n$ with $\varphi \in (0, 1]$ and $n \in \ZZ$, so that $E = E_0[n]$ with $E_0 \in \calA$ semistable of phase $\varphi$ with respect to $(\calA, Z_{\calA})$.
		Since $[E] = (-1)^n [E_0]$ in $\Kgroup(\calD)$ (Section~2.9), we get
		\[
		Z(E) = (-1)^n Z(E_0) = (-1)^n Z_{\calA}(E_0) = (-1)^n m \, e^{i \pi \varphi} = m \, e^{i \pi (\varphi + n)} = m \, e^{i \pi \psi}
		\]
		with $m = |Z_{\calA}(E_0)| > 0$, which is \condref{cond:ST}.
		Moreover $\| v(E) \| = \| v(E_0) \| \leqslant C \, |Z(E_0)| = C \, |Z(E)|$ with the constant $C$ of \condref{cond:SP-ab}, which is \condref{cond:SP}.
		So $\Psi(\calA, Z) = (Z, \calP) \in \operatorname{Stab}_v(\calD)$, and $\Psi$ is well defined.

		\step{the heart of $\Psi(\calA, Z)$ is $\calA$}
		Keep the notation of Steps~5 and~6.
		By Proposition~\ref{prop:discrete_slicing}(3), $\calP((0, 1])$ is the heart of a bounded t-structure.
		Every nonzero $E \in \calA$ has an HN filtration with respect to $(\calA, Z_{\calA})$, which is a filtration of $E$ with factors in $\calP(\psi_j)$, $\psi_j \in (0, 1]$;
		so $\calA \subset \calP((0, 1])$ by Proposition~\ref{prop:extension_closure}, and Corollary~\ref{cor:hearts_inclusion} gives $\calA = \calP((0, 1])$.

		\step{$\Phi$ and $\Psi$ are inverse to each other}
		Let $\sigma = (Z, \calP) \in \operatorname{Stab}_v(\calD)$, $(\calA, Z) = \Phi(\sigma)$ with $\calA = \calP((0, 1])$, and $\calP' := \calP_{\calA, Z}$, so that $\Psi(\Phi(\sigma)) = (Z, \calP')$.
		For $\varphi \in (0, 1]$ we have $\calP(\varphi) \subset \calP((0, 1]) = \calA$, so by assertion (1) the nonzero objects of $\calP'(\varphi)$, i.e.\ the $(\calA, Z_{\calA})$-semistable objects of phase $\varphi$, are exactly the nonzero objects of $\calP(\varphi)$.
		Hence $\calP'(\varphi) = \calP(\varphi)$ for $\varphi \in (0, 1]$, and then for all $\varphi \in \RR$ by \condref{cond:SL1}.
		Thus $\Psi \circ \Phi = \id$.
		Conversely, for $(\calA, Z) \in \operatorname{Stab}^{\mathrm{ab}}_v(\calD)$ we have $\Phi(\Psi(\calA, Z)) = (\calP_{\calA, Z}((0, 1]), Z) = (\calA, Z)$ by Step~7.
		Thus $\Phi \circ \Psi = \id$, and $\Phi$, $\Psi$ are mutually inverse bijections.
	\end{proof}

	Because of Theorem~\ref{thm:bijection} we shall use the notations $\sigma = (Z, \calP)$ and $\sigma = (\calA, Z)$ interchangeably, and we shall simply speak of a \emph{stability condition} $\sigma$ on $\calD$.
	Via $\Phi$, the HN filtration of an object $E$ of the heart $\calA$ with respect to $\sigma$ is its HN filtration in the abelian category $\calA$, and the $\sigma$-semistable objects are the shifts of the $(\calA, Z_{\calA})$-semistable objects, the shift by $[n]$ adding $n$ to the phase.

	\begin{corollary}
		\label{cor:rotated_heart}
		Let $\sigma = (Z, \calP)$ be a Bridgeland stability condition on $\calD$ and $\psi \in \RR$.
		Then $\calP_{\psi}(\varphi) := \calP(\varphi + \psi)$ defines a slicing $\calP_{\psi}$, and $\sigma_{\psi} := (e^{-i \pi \psi} Z, \calP_{\psi})$ is a Bridgeland stability condition on $\calD$ with heart $\calA_{\psi} := \calP((\psi, \psi + 1])$.
		Consequently $(\calA_{\psi}, e^{-i \pi \psi} Z_{\calA_{\psi}})$ is a stability condition on the abelian category $\calA_{\psi}$, whose semistable objects of phase $\varphi \in (0, 1]$ are exactly the nonzero objects of $\calP(\varphi + \psi)$.
	\end{corollary}

	\begin{proof}
		Conditions \condref{cond:SL1}--\condref{cond:SL3} for $\calP_{\psi}$ are those for $\calP$ with all phases shifted by $\psi$, which does not affect them.
		For $0 \neq E \in \calP_{\psi}(\varphi) = \calP(\varphi + \psi)$ we have $e^{-i \pi \psi} Z(E) = m(E) \, e^{i \pi (\varphi + \psi)} e^{-i \pi \psi} = m(E) \, e^{i \pi \varphi}$, which is \condref{cond:ST} for $\sigma_{\psi}$, and \condref{cond:SP} holds for $\sigma_{\psi}$ with the same constant as for $\sigma$, since $|e^{-i \pi \psi} Z(E)| = |Z(E)|$ and the semistable objects are the same.
		The heart of $\calP_{\psi}$ is $\calP_{\psi}((0, 1]) = \calP((\psi, \psi + 1])$, and the last assertion is Theorem~\ref{thm:bijection}(1) applied to $\sigma_{\psi}$.
	\end{proof}

	\begin{remark}
		We record some variants and complements of Theorem~\ref{thm:bijection}.
		\begin{enumerate}
			\item Corollary~\ref{cor:rotated_heart} shows that the heart $\calP((0, 1])$ plays no distinguished role among the hearts $\calP((\psi, \psi + 1])$:
			cutting the slicing at $\psi$ instead of $0$ corresponds to replacing $Z$ by $e^{-i \pi \psi} Z$.
			This is a first instance of the action of $\CC$ on $\operatorname{Stab}_v(\calD)$, to be discussed in the next section.
			\item \emph{Bounded t-structures instead of hearts.}
			By Lemma~\ref{lem:cohomology_filtration}(3) a bounded t-structure is determined by its heart, and by Proposition~\ref{prop:bounded_heart} the hearts of bounded t-structures are characterized intrinsically;
			so $\operatorname{Stab}^{\mathrm{ab}}_v(\calD)$ may equally be described as the set of pairs $(t, Z)$, where $t = (\Dle{0}, \Dge{0})$ is a bounded t-structure on $\calD$ and $(\Dle{0} \cap \Dge{0}, Z_{\Dle{0} \cap \Dge{0}})$ is a stability condition satisfying \condref{cond:SP-ab}.
			In these terms the heart map reads $\Phi(Z, \calP) = \big( (\calP(> 0), \calP(\leqslant 1)), Z \big)$ by Proposition~\ref{prop:slicing_t}(2).
			\item \emph{Fibres over hearts.}
			Composing $\Phi$ with $(\calA, Z) \mapsto \calA$ gives a map from $\operatorname{Stab}_v(\calD)$ to the set of hearts of bounded t-structures on $\calD$, $\sigma \mapsto \calP((0, 1])$.
			Its fibre over $\calA$ is the set of homomorphisms $Z \colon \Lambda \to \CC$ such that $Z_{\calA}$ is a stability function on $\calA$ with the Harder--Narasimhan property and the support property;
			in particular the heart alone does not determine $\sigma$.
			Example~\ref{ex:algebras_bridgeland} below describes this fibre for the heart $\operatorname{mod}\text{-}A$ of a finite-dimensional algebra.
			\item \emph{Without the support property.}
			The support property enters the proof of Theorem~\ref{thm:bijection} only in Steps~4 and~6, where it is transported from one side to the other without change.
			Hence the same maps give a bijection between the pairs $(Z, \calP)$ satisfying only \condref{cond:ST} and the pairs $(\calA, Z)$ with $(\calA, Z_{\calA})$ a stability condition on $\calA$, for an arbitrary abelian group $\Lambda$;
			this is the original formulation \cite[Prop.~5.3]{Bri07}.
		\end{enumerate}
	\end{remark}

	\subsection{First consequences}

	\begin{proposition}
		\label{prop:slice_abelian}
		Let $\sigma = (Z, \calP)$ be a stability condition on $\calD$ and $\varphi \in \RR$.
		\begin{enumerate}
			\item $\calP(\varphi)$ is an abelian category, and its short exact sequences are exactly the triangles of $\calD$ with all three vertices in $\calP(\varphi)$.
			Moreover, for every $\psi$ with $\varphi \in (\psi, \psi + 1]$, the subcategory $\calP(\varphi)$ is closed under kernels, cokernels and extensions in the heart $\calP((\psi, \psi + 1])$.
			\item $\calP(\varphi)$ has finite length, i.e.\ it is noetherian and artinian (\cite[Exercise~5.9]{MS17}).
		\end{enumerate}
	\end{proposition}

	\begin{proof}
		(1) Let $\psi$ be such that $\varphi \in (\psi, \psi + 1]$ and put $\calA_{\psi} := \calP((\psi, \psi + 1])$.
		By Corollary~\ref{cor:rotated_heart}, $(\calA_{\psi}, e^{-i \pi \psi} Z_{\calA_{\psi}})$ is a stability condition on $\calA_{\psi}$ whose semistable objects of phase $\varphi - \psi \in (0, 1]$ are the nonzero objects of $\calP(\varphi)$.
		So $\calP(\varphi)$ is the subcategory $\calP_{Z'}(\varphi - \psi)$ of Lemma~\ref{lem:semistable_abelian} for $Z' := e^{-i \pi \psi} Z_{\calA_{\psi}}$, hence an abelian category closed under kernels, cokernels and extensions in $\calA_{\psi}$, with exact inclusion.
		Its short exact sequences are therefore the short exact sequences of $\calA_{\psi}$ with terms in $\calP(\varphi)$, i.e.\ the triangles with vertices in $\calP(\varphi)$, by Lemma~\ref{lem:triangles_vs_ses}.

		(2) Let $0 \neq E \in \calP(\varphi)$, and let $F \subsetneq F'$ be subobjects of $E$ in $\calP(\varphi)$.
		Then $F' / F$ is a nonzero object of $\calP(\varphi)$ by (1), so $Z(F') = Z(F) + Z(F' / F)$ with $Z(F' / F) \in \RR_{>0} e^{i \pi \varphi}$, i.e.\ $m(F') > m(F)$;
		in particular $m(F) \leqslant m(E)$ for every subobject $F$ of $E$ in $\calP(\varphi)$.
		By the support property \condref{cond:SP}, the classes $v(F)$ of these subobjects satisfy $\| v(F) \| \leqslant C \, m(F) \leqslant C \, m(E)$, so they lie in the intersection of the lattice $\Lambda$ with a bounded subset of $\Lambda_{\RR}$, which is a finite set.
		Hence the masses of subobjects of $E$ in $\calP(\varphi)$ take only finitely many values, every strictly ascending or strictly descending chain of such subobjects has length bounded by the number of these values, and $\calP(\varphi)$ is noetherian and artinian.
	\end{proof}

	The simple objects of the abelian category $\calP(\varphi)$ are called \emph{$\sigma$-stable} (of phase $\varphi$).
	By Lemma~\ref{lem:semistable_abelian}(4) and Theorem~\ref{thm:bijection}, for $\varphi \in (0, 1]$ an object is $\sigma$-stable of phase $\varphi$ if and only if it is $(\calA, Z_{\calA})$-stable of phase $\varphi$, where $\calA = \calP((0, 1])$;
	in general $E$ is $\sigma$-stable if and only if $E[-n]$ is $\sigma$-stable for some (equivalently, every) $n \in \ZZ$.
	Since $\calP(\varphi)$ has finite length by Proposition~\ref{prop:slice_abelian}(2), every $\sigma$-semistable object of phase $\varphi$ has a Jordan--H\"older filtration in $\calP(\varphi)$ with $\sigma$-stable factors of phase $\varphi$, unique up to permutation and isomorphism.

	\begin{corollary}[Schur's lemma]
		\label{cor:schur_bridgeland}
		Let $\sigma$ be a stability condition on $\calD$ and let $E, F$ be $\sigma$-semistable objects.
		\begin{enumerate}
			\item If $\varphi(E) > \varphi(F)$, then $\Hom(E, F) = 0$.
			\item If $\varphi(E) = \varphi(F)$, both $E$ and $F$ are $\sigma$-stable and $f \colon E \to F$ is nonzero, then $f$ is an isomorphism.
			In particular $\End(E)$ is a division ring for every $\sigma$-stable object $E$.
		\end{enumerate}
	\end{corollary}

	\begin{proof}
		(1) is \condref{cond:SL2}.
		For (2), shifting we may assume $\varphi(E) = \varphi(F) \in (0, 1]$, so that $E$ and $F$ are $(\calA, Z_{\calA})$-stable objects of the same phase in $\calA = \calP((0, 1])$, and Lemma~\ref{lem:schur}(2) applies.
		A nonzero endomorphism of $E$ is then an isomorphism, so $\End(E)$ is a division ring.
	\end{proof}

	\begin{example}[curves, {\cite[Example~5.4]{Bri07}}, {\cite[Example~5.12]{MS17}}]
		\label{ex:curves_bridgeland}
		Let $C$ be a smooth projective curve over a field $\kk$, $\calD := \D^{\mathrm{b}}(\mathrm{Coh}(C))$ and $\calA := \mathrm{Coh}(C)$, the heart of the standard t-structure (Example~\ref{ex:standard_t}).
		Rank and degree are additive on short exact sequences of coherent sheaves, so they define a homomorphism
		\[
		v := (\operatorname{rk}, \deg) \colon \Kgroup(\calD) \cong \Kgroup(\calA) \longrightarrow \Lambda := \ZZ^2 ,
		\]
		where we let $\Lambda$ be the image of $(\operatorname{rk}, \deg)$, a subgroup of finite index in $\ZZ^2$ which equals $\ZZ^2$ as soon as $C$ has a $\kk$-rational point $p$, since then $v(\oo_C) = (1, 0)$ and $v(\oo_p) = (0, 1)$.
		Let $Z \colon \Lambda \to \CC$, $Z(r, d) := -d + i r$.
		Then $Z_{\calA}(E) = -\deg E + i \operatorname{rk} E$, and $(\calA, Z_{\calA})$ is a stability condition by Corollary~\ref{cor:HN_curves}.
		The support property \condref{cond:SP-ab} holds with $C = 1$ for the Euclidean norm on $\Lambda_{\RR} = \RR^2$, because $|Z(r, d)| = \sqrt{r^2 + d^2} = \| (r, d) \|$;
		in other words $Z \colon \Lambda_{\RR} \to \CC$ is an isomorphism of real vector spaces, so no semistable object is needed to verify it.
		Hence $(\calA, Z) \in \operatorname{Stab}^{\mathrm{ab}}_v(\calD)$, and Theorem~\ref{thm:bijection} gives a stability condition $\sigma_C = \Psi(\calA, Z) = (Z, \calP)$ on $\D^{\mathrm{b}}(\mathrm{Coh}(C))$, whose semistable objects of phase $\varphi + n$, $\varphi \in (0, 1]$, are the shifts $E[n]$ of the slope semistable sheaves $E$ of phase $\varphi$, i.e.\ the torsion sheaves if $\varphi = 1$ and the semistable vector bundles of slope $-\cot(\pi \varphi)$ if $\varphi < 1$.
	\end{example}

	\begin{example}[finite-dimensional algebras, {\cite[Example~5.5]{Bri07}}, {\cite[Exercise~5.13]{MS17}}]
		\label{ex:algebras_bridgeland}
		Let $A$ be a finite-dimensional algebra over a field $\kk$ with simple modules $S_1, \ldots, S_m$ up to isomorphism, $\calA := \operatorname{mod}\text{-}A$ and $\calD := \D^{\mathrm{b}}(\calA)$.
		By Example~\ref{ex:finite_dim_algebra} the classes $[S_1], \ldots, [S_m]$ form a basis of $\Kgroup(\calA) \cong \Kgroup(\calD)$, so we may take $\Lambda := \Kgroup(\calD) \cong \ZZ^m$ and $v := \id$, and a homomorphism $Z \colon \Lambda \to \CC$ is the same as a vector 
		\[
		(z_1, \ldots, z_m) = (Z(S_1), \ldots, Z(S_m)) \in \CC^m.
		\]
		By Example~\ref{ex:finite_dim_algebra}, $(\calA, Z_{\calA})$ is a stability condition whenever $z_1, \ldots, z_m \in \HH$.
		The support property \condref{cond:SP-ab} holds for every object of $\calA$, not only for the semistable ones.
		Indeed, for $M \in \calA$ we have $v(M) = (n_1, \ldots, n_m)$ with $n_j = [M : S_j] \geqslant 0$ and $Z(M) = \sum_j n_j z_j$;
		the continuous function $(t_1, \ldots, t_m) \mapsto |\sum_j t_j z_j|$ is positive on the compact simplex $\{ t_j \geqslant 0, \sum_j t_j = 1 \}$, since a sum of elements of $\HH$ with non-negative coefficients, not all zero, lies in $\HH$ and is nonzero;
		if $c > 0$ is its minimum, homogeneity gives 
		\[
		|Z(M)| \geqslant c \sum_j n_j = c \, \| v(M) \|_1,
		\] 
		so \condref{cond:SP-ab} holds with $C = 1 / c$ for the norm $\| \cdot \|_1$.
		Hence Theorem~\ref{thm:bijection} produces a family of stability conditions on $\D^{\mathrm{b}}(\operatorname{mod}\text{-}A)$ with heart $\operatorname{mod}\text{-}A$ parametrized by $\HH^m$.
		Conversely, if $\sigma = (Z, \calP)$ is a stability condition with heart $\calP((0, 1]) = \calA$, then $Z(S_j) \in \HH$ for all $j$ by \condref{cond:Z}, so $\HH^m$ is exactly the set of stability conditions with this heart.
		Corollary~\ref{cor:rotated_heart} produces further stability conditions whose hearts $\calP((\psi, \psi + 1])$ are different from $\operatorname{mod}\text{-}A$;
		this is a first glimpse of the structure of the set of all stability conditions as a space, which is the subject of the next section.
	\end{example}

	\begin{remark}
		Let $X$ be a smooth projective variety of dimension $\geqslant 2$ over a field.
		Example~\ref{ex:surfaces} shows that there is no stability function on $\mathrm{Coh}(X)$ factoring through the Chern character, hence, by Theorem~\ref{thm:bijection}, no stability condition on $\D^{\mathrm{b}}(\mathrm{Coh}(X))$ with heart $\mathrm{Coh}(X)$ and central charge factoring through the Chern character.
		Stability conditions on surfaces are constructed with other hearts, obtained from $\mathrm{Coh}(X)$ by tilting at a torsion pair;
		we shall come back to this later.
	\end{remark}
